    41→    41→

    44→    44→
    45→    45→export default function TrackWizard() {
    46→    46→  const { selectedBoardId, setBoardTasks, setIsTrackWizardOpen, setSelectedTaskId } = useKanbanStore();
    47→    47→  const [step, setStep] = useState(0); // 0=name, 1=instruments, 2=stages, 3=configure
    48→    48→  const [trackName, setTrackName] = useState('');
    49→    49→  const [selectedInstruments, setSelectedInstruments] = useState<string[]>([]);
    50→    50→  const [selectedStages, setSelectedStages] = useState<string[]>([]);
    51→    51→  const [customInstrument, setCustomInstrument] = useState('');
    52→    52→  const [wizardData, setWizardData] = useState<WizardData | null>(null);
    53→    53→  const [loading, setLoading] = useState(false);
    54→    54→  const [allInstruments, setAllInstruments] = useState<string[]>([]);
    55→    55→
    56→    56→  // Step 3: editable stages
    57→    57→  const [editableStages, setEditableStages] = useState<EditableStage[]>([]);
    58→    58→  const [expandedStageId, setExpandedStageId] = useState<string | null>(null);
    59→    59→  const [addingSubtaskForStage, setAddingSubtaskForStage] = useState<string | null>(null);
    60→    60→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
    61→    61→  const [newSubtaskDesc, setNewSubtaskDesc] = useState('');
    62→    62→  const [newStageLabel, setNewStageLabel] = useState('');
    63→    63→  const [newStageDesc, setNewStageDesc] = useState('');
    64→    64→  const [showNewStageForm, setShowNewStageForm] = useState(false);
    65→    65→
    66→    66→  useEffect(() => {
    67→    67→    fetch('/api/tracks/wizard')
    68→    68→      .then(r => r.json())
    69→    69→      .then(data => {
    70→    70→        setWizardData(data);
    71→    71→        setAllInstruments([...data.instruments]);
    72→    72→      });
    73→    73→  }, []);
    74→    74→
    75→    75→  // Build editable stages when entering step 3
    76→    76→  const buildEditableStages = useCallback(() => {
    77→    77→    if (!wizardData) return [];
    78→    78→    return selectedStages.map(stageKey => {
    79→    79→      const stage = wizardData.stages.find(s => s.key === stageKey);
    80→    80→      if (!stage) return null;

    82→    82→      const subtaskTitles = isInstrumentBased && selectedInstruments.length > 0
    83→    83→        ? selectedInstruments.map(i => `Запись: ${i}`)
    84→    84→        : stageKey === 'Сонграйтинг' ? ['Написание текста', 'Демо']
    85→    85→        : stageKey === 'Аранжировка' ? ['Гармоническая структура', 'Тембровая расстановка']
    86→    86→        : stageKey === 'Редактура' ? ['Компинг', 'Тюнинг вокала', 'Тайминг']
    87→    87→        : stageKey === 'Сведение' ? ['Баланс и эквализация', 'Пространственная обработка', 'Эффекты и автоматизация']
    88→    88→        : stageKey === 'Мастеринг' ? ['Финальная громкость и лимитирование', 'Стерео-обработка']
    89→    89→        : [];
    90→    90→      return {
    91→    91→        id: uid(),
    92→    92→        emoji: stage.emoji,
    93→    93→        label: stage.label,
    94→    94→        description: '',
    95→    95→        subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
    96→    96→      };
    97→    97→    }).filter(Boolean) as EditableStage[];
    98→    98→  }, [wizardData, selectedStages, selectedInstruments]);
    99→    99→
   100→   100→  const goToStep3 = () => {
   101→   101→    const stages = buildEditableStages();
   102→   102→    setEditableStages(stages);
   103→   103→    setExpandedStageId(stages.length > 0 ? stages[0].id : null);
   104→   104→    setStep(3);
   105→   105→  };
   106→   106→
   107→   107→  const toggleInstrument = (inst: string) => {
   108→   108→    setSelectedInstruments(prev =>
   109→   109→      prev.includes(inst) ? prev.filter(i => i !== inst) : [...prev, inst]
   110→   110→    );
   111→   111→  };
   112→   112→
   113→   113→  const addCustomInstrument = () => {
   114→   114→    const trimmed = customInstrument.trim();
   115→   115→    if (!trimmed || allInstruments.includes(trimmed)) return;
   116→   116→    setAllInstruments(prev => [...prev, trimmed]);
   117→   117→    setSelectedInstruments(prev => [...prev, trimmed]);
   118→   118→    setCustomInstrument('');
   119→   119→  };
   120→   120→
   121→   121→  const toggleStage = (key: string) => {
   122→   122→    setSelectedStages(prev =>
   123→   123→      prev.includes(key) ? prev.filter(s => s !== key) : [...prev, key]
   124→   124→    );
   125→   125→  };
   126→   126→
   127→   127→  const selectAllStages = () => {
   128→   128→    if (!wizardData) return;
   129→   129→    if (selectedStages.length === wizardData.stages.length) {
   130→   130→      setSelectedStages([]);
   131→   131→    } else {
   132→   132→      setSelectedStages(wizardData.stages.map(s => s.key));
   133→   133→    }
   134→   134→  };
   135→   135→
   136→   136→  // Step 3 helpers
   137→   137→  const updateStage = (stageId: string, updates: Partial<Pick<EditableStage, 'label' | 'description' | 'emoji'>>) => {
   138→   138→    setEditableStages(prev => prev.map(s => s.id === stageId ? { ...s, ...updates } : s));
   139→   139→  };
   140→   140→
   141→   141→  const removeStage = (stageId: string) => {
   142→   142→    setEditableStages(prev => prev.filter(s => s.id !== stageId));
   143→   143→  };
   144→   144→
   145→   145→  const addNewStage = () => {
   146→   146→    if (!newStageLabel.trim()) return;
   147→   147→    const newStage: EditableStage = {
   148→   148→      id: uid(),
   149→   149→      emoji: DEFAULT_STAGE_EMOJIS[editableStages.length % DEFAULT_STAGE_EMOJIS.length],
   150→   150→      label: newStageLabel.trim(),
   151→   151→      description: newStageDesc.trim(),
   152→   152→      subtasks: [],
   153→   153→    };
   154→   154→    setEditableStages(prev => [...prev, newStage]);
   155→   155→    setNewStageLabel('');
   156→   156→    setNewStageDesc('');
   157→   157→    setShowNewStageForm(false);
   158→   158→    setExpandedStageId(newStage.id);
   159→   159→  };
   160→   160→
   161→   161→  const updateSubtask = (stageId: string, subtaskId: string, updates: Partial<Pick<EditableSubtask, 'title' | 'description'>>) => {
   162→   162→    setEditableStages(prev => prev.map(s => {
   163→   163→      if (s.id !== stageId) return s;
   164→   164→      return {
   165→   165→        ...s,
   166→   166→        subtasks: s.subtasks.map(st => st.id === subtaskId ? { ...st, ...updates } : st),
   167→   167→      };
   168→   168→    }));
   169→   169→  };
   170→   170→
   171→   171→  const removeSubtask = (stageId: string, subtaskId: string) => {
   172→   172→    setEditableStages(prev => prev.map(s => {
   173→   173→      if (s.id !== stageId) return s;
   174→   174→      return { ...s, subtasks: s.subtasks.filter(st => st.id !== subtaskId) };
   175→   175→    }));
   176→   176→  };
   177→   177→
   178→   178→  const addSubtask = (stageId: string) => {
   179→   179→    if (!newSubtaskTitle.trim()) return;
   180→   180→    setEditableStages(prev => prev.map(s => {
   181→   181→      if (s.id !== stageId) return s;
   182→   182→      return {
   183→   183→        ...s,
   184→   184→        subtasks: [...s.subtasks, { id: uid(), title: newSubtaskTitle.trim(), description: newSubtaskDesc.trim() }],
   185→   185→      };
   186→   186→    }));
   187→   187→    setNewSubtaskTitle('');
   188→   188→    setNewSubtaskDesc('');
   189→   189→    setAddingSubtaskForStage(null);
   190→   190→  };
   191→   191→
   192→   192→  const handleCreate = useCallback(async () => {
   193→   193→    if (!trackName.trim() || !selectedBoardId) return;
   194→   194→    setLoading(true);
   195→   195→    try {
   196→   196→      const res = await fetch('/api/tracks/wizard', {
   197→   197→        method: 'POST',
   198→   198→        headers: { 'Content-Type': 'application/json' },
   199→   199→        body: JSON.stringify({
   200→   200→          title: trackName.trim(),
   201→   201→          instruments: selectedInstruments,
   202→   202→          boardId: selectedBoardId,
   203→   203→          customStages: editableStages.map(s => ({
   204→   204→            emoji: s.emoji,
   205→   205→            label: s.label,
   206→   206→            description: s.description,
   207→   207→            subtasks: s.subtasks.map(st => ({
   208→   208→              title: st.title,
   209→   209→              description: st.description,
   210→   210→            })),
   211→   211→          })),
   212→   212→        }),
   213→   213→      });
   214→   214→      const data = await res.json();
   215→   215→      // Reload tasks for the board
   216→   216→      const taskRes = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   217→   217→      const taskData = await taskRes.json();
   218→   218→      setBoardTasks(taskData.tasks);
   219→   219→      // Select the new track
   220→   220→      if (data.trackId) setSelectedTaskId(data.trackId);
   221→   221→      setIsTrackWizardOpen(false);
   222→   222→    } finally {
   223→   223→      setLoading(false);
   224→   224→    }
   225→   225→  }, [trackName, selectedInstruments, selectedBoardId, editableStages, setBoardTasks, setSelectedTaskId, setIsTrackWizardOpen]);
   226→   226→
   227→   227→  const canProceed = () => {
   228→   228→    if (step === 0) return trackName.trim().length > 0;
   229→   229→    if (step === 1) return selectedInstruments.length > 0;
   230→   230→    if (step === 2) return selectedStages.length > 0;
   231→   231→    if (step === 3) return editableStages.length > 0;
   232→   232→    return true;
   233→   233→  };
   234→   234→
   235→   235→  const handleClose = () => setIsTrackWizardOpen(false);
   236→   236→
   237→   237→  const STEPS = ['Название', 'Инструменты', 'Этапы', 'Настройка'];
   238→   238→
   239→   239→  return (
   240→   240→    <div className="flex flex-col h-full">
   241→   241→      {/* Header */}
   242→   242→      <div className="border-b border-slate-800/50 p-4 shrink-0">
   243→   243→        <div className="flex items-center justify-between mb-3">
   244→   244→          <div className="flex items-center gap-2">
   245→   245→            <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-purple-500 to-pink-500 flex items-center justify-center">
   246→   246→              <Music className="w-4 h-4 text-white" />
   247→   247→            </div>
   248→   248→            <div>
   249→   249→              <h3 className="text-xs font-bold text-white uppercase tracking-wider">Конструктор трека</h3>
   250→   250→              <p className="text-[10px] text-slate-500">Мастер создания пайплайна</p>
   251→   251→            </div>
   252→   252→          </div>
   253→   253→          <button onClick={handleClose} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors">
   254→   254→            <X className="w-4 h-4" />
   255→   255→          </button>
   256→   256→        </div>
   257→   257→
   258→   258→        {/* Progress steps */}
   259→   259→        <div className="flex items-center gap-1">
   260→   260→          {STEPS.map((s, i) => (
   261→   261→            <div key={s} className="flex items-center flex-1">
   262→   262→              <button
   263→   263→                onClick={() => i < step && setStep(i)}
   264→   264→                className={cn(
   265→   265→                  'flex items-center gap-1.5 text-[10px] font-medium px-2 py-1 rounded-md transition-all duration-200',
   266→   266→                  i === step && 'bg-cyan-500/15 text-cyan-400',
   267→   267→                  i < step && 'text-slate-400 hover:bg-slate-800/60 cursor-pointer',
   268→   268→                  i > step && 'text-slate-600',
   269→   269→                )}
   270→   270→              >
   271→   271→                {i < step ? <Check className="w-3 h-3" /> : <span className="w-4 h-4 rounded-full border text-center text-[9px] flex items-center justify-center" style={{
   272→   272→                  borderColor: i === step ? '#00d9ff' : i < step ? '#00d9ff' : '#334155',
   273→   273→                }}>{i + 1}</span>}
   274→   274→                <span className="hidden lg:inline">{s}</span>
   275→   275→              </button>
   276→   276→              {i < STEPS.length - 1 && (
   277→   277→                <div className="flex-1 h-px mx-1" style={{ backgroundColor: i < step ? '#00d9ff30' : '#1e293b' }} />
   278→   278→              )}
   279→   279→            </div>
   280→   280→          ))}
   281→   281→        </div>
   282→   282→      </div>
   283→   283→
   284→   284→      {/* Content */}
   285→   285→      <div className="flex-1 overflow-y-auto p-4">
   286→   286→        {/* Step 0: Name */}
   287→   287→        {step === 0 && (
   288→   288→          <div className="space-y-3">
   289→   289→            <div className="space-y-1.5">
   290→   290→              <label className="text-[11px] text-slate-400 font-medium">Название трека</label>
   291→   291→              <Input
   292→   292→                value={trackName}
   293→   293→                onChange={(e) => setTrackName(e.target.value)}
   294→   294→                onKeyDown={(e) => e.key === 'Enter' && canProceed() && setStep(1)}
   295→   295→                placeholder="Лирика, Эксперимент..."
   296→   296→                autoFocus
   297→   297→                className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-10 focus:border-purple-500/50"
   298→   298→              />
   299→   299→            </div>
   300→   300→            <div className="rounded-lg bg-slate-900/40 border border-slate-800/30 p-3">
   301→   301→              <div className="flex items-center gap-2 mb-2">
   302→   302→                <Sparkles className="w-3.5 h-3.5 text-purple-400" />
   303→   303→                <span className="text-[11px] text-purple-400 font-medium">Подсказка</span>
   304→   304→              </div>
   305→   305→              <p className="text-[10px] text-slate-500 leading-relaxed">
   306→   306→                После ввода названия вы выберете инструменты и этапы производства.
   307→   307→                Система автоматически создаст структуру задач для вашего трека.
   308→   308→              </p>
   309→   309→            </div>
   310→   310→          </div>
   311→   311→        )}
   312→   312→
   313→   313→        {/* Step 1: Instruments */}
   314→   314→        {step === 1 && (
   315→   315→          <div className="space-y-3">
   316→   316→            <div className="flex items-center justify-between">
   317→   317→              <span className="text-[11px] text-slate-400 font-medium">
   318→   318→                Инструменты <span className="text-slate-600">({selectedInstruments.length} выбрано)</span>
   319→   319→              </span>
   320→   320→            </div>
   321→   321→            <div className="flex flex-wrap gap-1.5">
   322→   322→              {allInstruments.map((inst) => {
   323→   323→                const IconComp = CUSTOM_ICONS[inst];
   324→   324→                const isSelected = selectedInstruments.includes(inst);
   325→   325→                return (
   326→   326→                  <button
   327→   327→                    key={inst}
   328→   328→                    onClick={() => toggleInstrument(inst)}
   329→   329→                    className={cn(
   330→   330→                      'flex items-center gap-1.5 text-[11px] px-2.5 py-1.5 rounded-lg border transition-all duration-150',
   331→   331→                      isSelected
   332→   332→                        ? 'bg-cyan-500/15 border-cyan-500/40 text-cyan-300 shadow-sm shadow-cyan-500/10'
   333→   333→                        : 'bg-slate-900/60 border-slate-800/50 text-slate-400 hover:border-slate-700 hover:text-slate-300',
   334→   334→                    )}
   335→   335→                  >
   336→   336→                    {IconComp ? <IconComp className="w-3 h-3" /> : <Music className="w-3 h-3" />}
   337→   337→                    {inst}
   338→   338→                    {isSelected && <Check className="w-3 h-3 ml-0.5" />}
   339→   339→                  </button>
   340→   340→                );
   341→   341→              })}
   342→   342→            </div>
   343→   343→            {/* Add custom instrument */}
   344→   344→            <div className="flex gap-2 pt-1">
   345→   345→              <Input
   346→   346→                value={customInstrument}
   347→   347→                onChange={(e) => setCustomInstrument(e.target.value)}
   348→   348→                onKeyDown={(e) => e.key === 'Enter' && addCustomInstrument()}
   349→   349→                placeholder="Свой инструмент..."
   350→   350→                className="bg-slate-900/80 border-slate-700/50 text-[11px] text-slate-200 placeholder:text-slate-600 h-7 focus:border-purple-500/50"
   351→   351→              />
   352→   352→              <Button size="sm" onClick={addCustomInstrument} disabled={!customInstrument.trim()} variant="ghost" className="h-7 text-[10px] text-slate-400 hover:text-purple-400 gap-1">
   353→   353→                <Plus className="w-3 h-3" />
   354→   354→              </Button>
   355→   355→            </div>
   356→   356→            <div className="rounded-lg bg-slate-900/40 border border-slate-800/30 p-3">
   357→   357→              <div className="flex items-center gap-2 mb-1">
   358→   358→                <Sparkles className="w-3.5 h-3.5 text-amber-400" />
   359→   359→                <span className="text-[11px] text-amber-400 font-medium">Подсказка</span>
   360→   360→              </div>
   361→   361→              <p className="text-[10px] text-slate-500 leading-relaxed">
   362→   362→                Выбранные инструменты определят подзадачи на этапе &laquo;Запись&raquo;.
   363→   363→                Для каждого инструмента создастся отдельная задача.
   364→   364→              </p>
   365→   365→            </div>
   366→   366→          </div>
   367→   367→        )}
   368→   368→
   369→   369→        {/* Step 2: Stages */}
   370→   370→        {step === 2 && (
   371→   371→          <div className="space-y-3">
   372→   372→            <div className="flex items-center justify-between">
   373→   373→              <span className="text-[11px] text-slate-400 font-medium">
   374→   374→                Этапы производства <span className="text-slate-600">({selectedStages.length})</span>
   375→   375→              </span>
   376→   376→              <button
   377→   377→                onClick={selectAllStages}
   378→   378→                className="text-[10px] text-cyan-500/70 hover:text-cyan-400 transition-colors"
   379→   379→              >
   380→   380→                {selectedStages.length === (wizardData?.stages.length || 0) ? 'Снять все' : 'Выбрать все'}
   381→   381→              </button>
   382→   382→            </div>
   383→   383→            {wizardData?.stages.map(stage => {
   384→   384→              const isSelected = selectedStages.includes(stage.key);
   385→   385→              const subtaskHint = stage.hasInstrumentBased && selectedInstruments.length > 0
   386→   386→                ? `${selectedInstruments.length} задач по инструментам`
   387→   387→                : `${stage.defaultSubtaskCount} задач`;
   388→   388→              return (
   389→   389→                <button
   390→   390→                  key={stage.key}
   391→   391→                  onClick={() => toggleStage(stage.key)}
   392→   392→                  className={cn(
   393→   393→                    'w-full flex items-center gap-3 p-3 rounded-lg border transition-all duration-150 text-left group',
   394→   394→                    isSelected
   395→   395→                      ? 'bg-slate-800/60 border-cyan-500/30'
   396→   396→                      : 'bg-slate-900/40 border-slate-800/30 hover:border-slate-700/60',
   397→   397→                  )}
   398→   398→                >
   399→   399→                  <div className={cn(
   400→   400→                    'w-5 h-5 rounded-md border-2 flex items-center justify-center transition-all duration-150',
