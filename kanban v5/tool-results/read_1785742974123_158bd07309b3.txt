    94→/* ── Track Detail View (3-level hierarchy) ──────────── */
    95→
    96→const EMOJI_RE = /^(\p{Emoji_Presentation}|\p{Extended_Pictographic})/u;
    97→const DESC_LIMIT = 300;
    98→
    99→function TrackDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   100→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   101→  const [stages, setStages] = useState<TaskChild[]>(task.children || []);
   102→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(
   103→    Object.fromEntries((task.children || []).map(c => [c.id, true]))
   104→  );
   105→  const [expandedDescIds, setExpandedDescIds] = useState<Record<string, boolean>>({});
   106→  const [editingDescId, setEditingDescId] = useState<string | null>(null);
   107→  const [descDraft, setDescDraft] = useState('');
   108→  const [addingGrandchild, setAddingGrandchild] = useState<string | null>(null);
   109→  const [newGcTitle, setNewGcTitle] = useState('');
   110→  const [addingStage, setAddingStage] = useState(false);
   111→  const [newStageTitle, setNewStageTitle] = useState('');
   112→
   113→  const config = task.trackConfig ? JSON.parse(task.trackConfig) : null;
   114→  const allSubtasks = stages.flatMap(s => s.children || []);
   115→  const progress = getProgress(allSubtasks);
   116→  const progressColor = progress === 100 ? 'bg-emerald-500' : progress > 50 ? 'bg-cyan-500' : progress > 0 ? 'bg-amber-500' : 'bg-slate-700';
   117→
   118→  const reloadTasks = useCallback(async () => {
   119→    if (!selectedBoardId) return;
   120→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   121→    const data = await res.json();
   122→    setBoardTasks(data.tasks);
   123→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   124→    if (updated) setStages(updated.children || []);
   125→  }, [selectedBoardId, setBoardTasks, task.id]);
   126→
   127→  const toggleGrandchildStatus = async (e: React.MouseEvent, id: string) => {
   128→    e.stopPropagation();
   129→    const order: TaskStatus[] = ['todo', 'in-progress', 'review', 'done'];
   130→    const all = stages.flatMap(s => s.children || []);
   131→    const gc = all.find(c => c.id === id);
   132→    if (!gc) return;
   133→    const idx = order.indexOf(gc.status as TaskStatus);
   134→    const next = order[(idx + 1) % order.length];
   135→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id, status: next }) });
   136→    await reloadTasks();
   137→  };
   138→
   139→  const toggleStageExpanded = (stageId: string) => {
   140→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   141→  };
   142→
   143→  const toggleDescExpanded = (id: string) => {
   144→    setExpandedDescIds(prev => ({ ...prev, [id]: !prev[id] }));
   145→  };
   146→
   147→  const startEditDesc = (id: string, current: string) => {
   148→    setEditingDescId(id);
   149→    setDescDraft(current || '');
   150→  };
   151→
   152→  const saveDesc = async (id: string) => {
   153→    await fetch('/api/tasks', {
   154→      method: 'PUT',
   155→      headers: { 'Content-Type': 'application/json' },
   156→      body: JSON.stringify({ id, description: descDraft.trim() || null }),
   157→    });
   158→    setEditingDescId(null);
   159→    setDescDraft('');
   160→    await reloadTasks();
   161→  };
   162→
   163→  const cancelEditDesc = () => {
   164→    setEditingDescId(null);
   165→    setDescDraft('');
   166→  };
   167→
   168→  const deleteGrandchild = async (e: React.MouseEvent, id: string) => {
   169→    e.stopPropagation();
   170→    await fetch(`/api/tasks?id=${id}`, { method: 'DELETE' });
   171→    await reloadTasks();
   172→  };
   173→
   174→  const addGrandchild = async (stageId: string) => {
   175→    if (!newGcTitle.trim()) return;
   176→    await fetch('/api/tasks', {
   177→      method: 'POST',
   178→      headers: { 'Content-Type': 'application/json' },
   179→      body: JSON.stringify({ title: newGcTitle.trim(), parentId: stageId }),
   180→    });
   181→    setNewGcTitle('');
   182→    setAddingGrandchild(null);
   183→    await reloadTasks();
   184→  };
   185→
   186→  const addStage = async () => {
   187→    if (!newStageTitle.trim()) return;
   188→    await fetch('/api/tasks', {
   189→      method: 'POST',
   190→      headers: { 'Content-Type': 'application/json' },
   191→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: selectedBoardId }),
   192→    });
   193→    setNewStageTitle('');
   194→    setAddingStage(false);
   195→    await reloadTasks();
   196→  };
   197→
   198→  const deleteStage = async (e: React.MouseEvent, stageId: string) => {
   199→    e.stopPropagation();
   200→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   201→    await reloadTasks();
   202→  };
   203→
   204→  const handleDelete = async () => {
   205→    await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   206→    setSelectedTaskId(null);
   207→    if (selectedBoardId) {
   208→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   209→      const data = await res.json();
   210→      setBoardTasks(data.tasks);
   211→    }
   212→  };
   213→
   214→  return (
   215→    <div className="flex-1 overflow-y-auto">
   216→      {/* Header */}
   217→      <div className="border-b border-slate-800/50 p-4">
   218→        <div className="flex items-center gap-1.5 mb-2">
   219→          <span className="text-[9px] text-purple-400 bg-purple-500/15 px-1.5 py-0.5 rounded font-medium uppercase tracking-wider">
   220→            Трек
   221→          </span>
   222→          <div className="flex-1" />
   223→          <button onClick={() => setEditingTask(task)} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors">
   224→            <Pencil className="w-3.5 h-3.5" />
   225→          </button>
   226→          <button onClick={handleDelete} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors">
   227→            <Trash2 className="w-3.5 h-3.5" />
   228→          </button>
   229→        </div>
   230→        <h3 className="text-sm font-semibold text-white leading-tight mb-1.5">{task.title}</h3>
   231→        {task.description && (
   232→          <p className="text-[11px] text-slate-500 mb-1.5 line-clamp-2">{task.description}</p>
   233→        )}
   234→        {config && (
   235→          <div className="flex flex-wrap gap-1">
   236→            {config.instruments?.map((inst: string) => (
   237→              <span key={inst} className="text-[9px] bg-slate-800/80 text-slate-400 px-1.5 py-0.5 rounded">{inst}</span>
   238→            ))}
   239→          </div>
   240→        )}
   241→      </div>
   242→
   243→      {/* Progress */}
   244→      <div className="border-b border-slate-800/30 px-4 py-3">
   245→        <div className="flex items-center justify-between mb-1.5">
   246→          <span className="text-[11px] text-slate-400 font-medium">Общий прогресс</span>
   247→          <span className={cn('text-[11px] font-bold',
   248→            progressColor === 'bg-emerald-500' ? 'text-emerald-400' : progressColor === 'bg-cyan-500' ? 'text-cyan-400' : progressColor === 'bg-amber-500' ? 'text-amber-400' : 'text-slate-600'
   249→          )}>{progress}%</span>
   250→        </div>
   251→        <div className="relative h-2 bg-slate-800/80 rounded-full overflow-hidden">
   252→          <div className={cn('h-full rounded-full transition-all duration-700 ease-out', progressColor)} style={{ width: `${progress}%` }} />
   253→          {progress > 0 && progress < 100 && (
   254→            <div
   255→              className="absolute inset-0 rounded-full animate-pulse opacity-20"
   256→              style={{
   257→                background: `linear-gradient(90deg, transparent, ${progressColor === 'bg-emerald-500' ? '#10b981' : progressColor === 'bg-cyan-500' ? '#00d9ff' : progressColor === 'bg-amber-500' ? '#f59e0b' : '#475569'}, transparent)`,
   258→              }}
   259→            />
   260→          )}
   261→        </div>
   262→        <div className="flex items-center justify-between mt-1">
   263→          <span className="text-[9px] text-slate-600">{allSubtasks.filter(s => s.status === 'done').length} из {allSubtasks.length} задач</span>
   264→          {progress === 100 && <span className="text-[9px] text-emerald-400 font-medium">✓ Завершено</span>}
   265→        </div>
   266→      </div>
   267→
   268→      {/* Stages as collapsible groups */}
   269→      <div className="px-3 py-2 space-y-1">
   270→        {stages.map((stage) => {
   271→          const grandchildren = stage.children || [];
   272→          const stageProgress = getProgress(grandchildren);
   273→          const stageDone = stage.status === 'done';
   274→          const isExpanded = expandedStages[stage.id] !== false;
   275→          const isCustom = !EMOJI_RE.test(stage.title);
   276→          const StatusIcon = SUBTASK_STATUS_ICON[stage.status] || Circle;
   277→          const hasDesc = !!stage.description;
   278→          const descExpanded = expandedDescIds[stage.id] === true;
   279→          const isEditingDesc = editingDescId === stage.id;
   280→
   281→          return (
   282→            <div key={stage.id} className="rounded-lg border border-slate-800/30 overflow-hidden">
   283→              {/* Stage header — full panel click to expand */}
   284→              <button
   285→                onClick={() => toggleStageExpanded(stage.id)}
   286→                className="w-full flex items-center gap-2 px-2.5 py-2 hover:bg-slate-900/60 transition-colors text-left group/stage"
   287→              >
   288→                <StatusIcon className={cn('w-3.5 h-3.5 flex-shrink-0', SUBTASK_STATUS_COLOR[stage.status])} />
   289→                {isCustom && <Cat className="w-3 h-3 flex-shrink-0 text-amber-400/60" />}
   290→                <span className={cn('flex-1 text-[11px] font-medium min-w-0',
   291→                  stageDone ? 'text-slate-500 line-through' : 'text-slate-300'
   292→                )} onDoubleClick={(e) => { e.stopPropagation(); useKanbanStore.getState().setDescriptionEditorItem({ id: stage.id, title: stage.title, description: stage.description, parentId: task.id, parentTitle: task.title }); }}>
   293→                  {stage.title}
   294→                </span>
   295→                {/* Description icon */}
   296→                <button
   297→                  onClick={(e) => { e.stopPropagation(); if (hasDesc) toggleDescExpanded(stage.id); else startEditDesc(stage.id, stage.description || ''); }}
   298→                  className={cn('flex-shrink-0 transition-opacity',
   299→                    hasDesc ? 'text-slate-500 hover:text-cyan-400' : 'text-slate-700 opacity-0 group-hover/stage:opacity-100 hover:text-cyan-400'
   300→                  )}
   301→                >
   302→                  <MessageSquare className="w-3 h-3" />
   303→                </button>
   304→                {grandchildren.length > 0 && (
   305→                  <span className="text-[9px] text-slate-600 mr-1">{stageDone ? '✓' : `${stageProgress}%`}</span>
   306→                )}
   307→                <button
   308→                  onClick={(e) => { e.stopPropagation(); deleteStage(e, stage.id); }}
   309→                  className="flex-shrink-0 p-0.5 rounded opacity-0 group-hover/stage:opacity-100 hover:bg-rose-500/10 text-slate-600 hover:text-rose-400 transition-all"
   310→                >
   311→                  <Trash2 className="w-3 h-3" />
   312→                </button>
   313→                {isExpanded
   314→                  ? <ChevronUp className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   315→                  : <ChevronDown className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   316→                }
   317→              </button>
   318→
   319→              {/* Expanded content */}
   320→              {isExpanded && (
   321→                <div>
   322→                  {/* Description card */}
   323→                  {hasDesc && !isEditingDesc && (
   324→                    <div className="mx-2.5 mb-1.5">
   325→                      <div className="bg-slate-900/60 border border-slate-800/40 rounded-md p-2">
   326→                        {descExpanded ? (
   327→                          <>
   328→                            <p className="text-[10px] text-slate-400 whitespace-pre-wrap leading-relaxed">
   329→                              {stage.description && stage.description.length > DESC_LIMIT
   330→                                ? stage.description.slice(0, DESC_LIMIT) + '...'
   331→                                : stage.description}
   332→                            </p>
   333→                            <div className="flex items-center gap-2 mt-1.5">
   334→                              <button
   335→                                onClick={() => startEditDesc(stage.id, stage.description || '')}
   336→                                className="text-[9px] text-cyan-500/70 hover:text-cyan-400 transition-colors"
   337→                              >Редактировать</button>
   338→                              <button
   339→                                onClick={() => toggleDescExpanded(stage.id)}
   340→                                className="text-[9px] text-slate-600 hover:text-slate-400 transition-colors"
   341→                              >Свернуть</button>
   342→                            </div>
   343→                          </>
   344→                        ) : (
   345→                          <p className="text-[10px] text-slate-500 line-clamp-2">
   346→                            {stage.description}
   347→                          </p>
   348→                        )}
   349→                      </div>
   350→                    </div>
   351→                  )}
   352→                  {/* Editing description */}
   353→                  {isEditingDesc && (
   354→                    <div className="mx-2.5 mb-1.5">
   355→                      <div className="bg-slate-900/60 border border-cyan-500/30 rounded-md p-2 space-y-1.5">
   356→                        <Textarea
   357→                          value={descDraft}
   358→                          onChange={(e) => setDescDraft(e.target.value)}
   359→                          placeholder="Описание этапа..."
   360→                          maxLength={DESC_LIMIT}
   361→                          className="bg-slate-800/80 border-slate-700/50 text-[10px] text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none"
   362→                          autoFocus
   363→                        />
   364→                        <div className="flex items-center justify-between">
   365→                          <span className="text-[8px] text-slate-600">{descDraft.length}/{DESC_LIMIT}</span>
   366→                          <div className="flex items-center gap-1">
   367→                            <button onClick={cancelEditDesc} className="text-[9px] text-slate-500 hover:text-slate-300 px-1.5 py-0.5 rounded hover:bg-slate-800 transition-colors">Отмена</button>
   368→                            <button
   369→                              onClick={() => saveDesc(stage.id)}
   370→                              className="text-[9px] text-cyan-400 bg-cyan-500/10 px-1.5 py-0.5 rounded hover:bg-cyan-500/20 transition-colors"
   371→                            >Сохранить</button>
   372→                          </div>
   373→                        </div>
   374→                      </div>
   375→                    </div>
   376→                  )}
   377→
   378→                  {/* Progress bar */}
   379→                  {grandchildren.length > 0 && (
   380→                    <div className="px-2.5 pb-1.5">
   381→                      <div className="h-1 bg-slate-800/60 rounded-full overflow-hidden">
   382→                        <div
   383→                          className={cn('h-full rounded-full transition-all duration-500',
   384→                            stageProgress === 100 ? 'bg-emerald-500' : stageProgress > 50 ? 'bg-cyan-500' : stageProgress > 0 ? 'bg-amber-500' : 'bg-slate-700'
   385→                          )}
   386→                          style={{ width: `${stageProgress}%` }}
   387→                        />
   388→                      </div>
   389→                    </div>
   390→                  )}
   391→
   392→                  {/* Grandchildren list */}
   393→                  <div className="space-y-0.5 px-2 pb-2">
   394→                    {grandchildren.map((gc) => {
   395→                      const GcIcon = SUBTASK_STATUS_ICON[gc.status] || Circle;
   396→                      const gcHasDesc = !!gc.description;
   397→                      const gcDescExpanded = expandedDescIds[gc.id] === true;
   398→                      const gcIsEditingDesc = editingDescId === gc.id;
   399→                      return (
   400→                        <div key={gc.id} className="rounded-md border border-transparent hover:border-slate-800/30 overflow-hidden">
   401→                          <div
   402→                            className="flex items-center gap-2 px-2 py-1 rounded-md hover:bg-slate-800/40 transition-colors group/gc cursor-pointer"
   403→                            onClick={() => {
   404→                              useKanbanStore.getState().setDescriptionEditorItem({
   405→                                id: gc.id, title: gc.title, description: gc.description,
   406→                                parentId: stage.id, parentTitle: stage.title,
   407→                              });
   408→                            }}
   409→                          >
   410→                            <button
   411→                              onClick={(e) => toggleGrandchildStatus(e, gc.id)}
   412→                              className={cn('flex-shrink-0 hover:opacity-70 transition-opacity', SUBTASK_STATUS_COLOR[gc.status] || 'text-slate-500')}
   413→                            >
   414→                              <GcIcon className="w-3 h-3" />
   415→                            </button>
   416→                            <span className={cn('flex-1 text-[10px] min-w-0',
   417→                              gc.status === 'done' ? 'text-slate-600 line-through' : 'text-slate-400'
   418→                            )}>
   419→                              {gc.title}
   420→                            </span>
   421→                            {/* GC description icon */}
   422→                            <button
   423→                              onClick={(e) => { e.stopPropagation(); if (gcHasDesc) toggleDescExpanded(gc.id); else startEditDesc(gc.id, gc.description || ''); }}
   424→                              className={cn('flex-shrink-0 transition-opacity',
   425→                                gcHasDesc ? 'text-slate-600 hover:text-cyan-400' : 'text-slate-700 opacity-0 group-hover/gc:opacity-100 hover:text-cyan-400'
   426→                              )}
   427→                            >
   428→                              <MessageSquare className="w-2.5 h-2.5" />
   429→                            </button>
   430→                            <button
   431→                              onClick={(e) => deleteGrandchild(e, gc.id)}
   432→                              className="flex-shrink-0 p-0.5 rounded opacity-0 group-hover/gc:opacity-100 hover:bg-rose-500/10 text-slate-700 hover:text-rose-400 transition-all"
   433→                            >
   434→                              <Trash2 className="w-2.5 h-2.5" />
   435→                            </button>
   436→                          </div>
   437→                          {/* GC expanded description */}
   438→                          {gcHasDesc && !gcIsEditingDesc && gcDescExpanded && (
   439→                            <div className="px-2 pb-1.5 pt-0.5">
   440→                              <div className="bg-slate-900/40 border border-slate-800/30 rounded p-1.5">

   731→      setPriority(task.priority); setAssignee(task.assignee || ''); setCategory(task.category);

   756→        await fetch('/api/tasks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title, description: description || null, status, priority, assignee: assignee || null, category, parentId: null, boardId: selectedBoardId, deadline }) });
   757→      }
   758→      await reloadTasks(); setEditingTask(null); setIsCreating(false);
   759→      if (mode === 'edit' && task) setSelectedTaskId(task.id);
   760→    } finally { setSaving(false); }
   761→  };
   762→
   763→  const handleDelete = async () => {
   764→    if (!task) return;
   765→    setSaving(true);
   766→    try {
   767→      await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   768→      await reloadTasks(); setEditingTask(null); setSelectedTaskId(null);
   769→    } finally { setSaving(false); }
   770→  };

   774→      <div className="p-4 space-y-3">
   775→        <div className="flex items-center justify-between">

   782→          <div className="grid grid-cols-2 gap-2">
   783→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Статус</Label><Select value={status} onValueChange={(v) => setStatus(v as TaskStatus)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{STATUSES.map((s) => (<SelectItem key={s.value} value={s.value}><span className={s.color}>{s.label}</span></SelectItem>))}</SelectContent></Select></div>

   785→          </div>

   787→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Категория</Label><Select value={category} onValueChange={(v) => setCategory(v as TaskCategory)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{CATEGORIES.map((c) => (<SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>))}</SelectContent></Select></div>
   788→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Ответственный</Label><Input value={assignee} onChange={(e) => setAssignee(e.target.value)} placeholder="Имя..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 placeholder:text-slate-600 h-8 focus:border-cyan-500/50" /></div>
   789→          </div>
   790→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Дедлайн</Label><DeadlinePicker value={deadline} onChange={setDeadline} size="md" /></div>
   791→        </div>
   792→        <div className="flex gap-2 pt-1">
   793→          <Button onClick={handleSave} disabled={!title.trim() || saving} className="flex-1 bg-cyan-600 hover:bg-cyan-700 text-white h-8 text-xs"><Save className="w-3 h-3 mr-1" />{saving ? '...' : mode === 'edit' ? 'Сохранить' : 'Создать'}</Button>
   794→          {mode === 'edit' && <Button onClick={handleDelete} variant="destructive" disabled={saving} className="h-8 text-xs">Удалить</Button>}
   795→        </div>
   796→      </div>
   797→    </div>
   798→  );
   799→}
