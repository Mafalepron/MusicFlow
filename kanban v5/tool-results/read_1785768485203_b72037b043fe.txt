   101→   101→function TrackDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   102→   102→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   103→   103→  const [stages, setStages] = useState<TaskChild[]>(task.children || []);
   104→   104→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(
   105→   105→    Object.fromEntries((task.children || []).map(c => [c.id, true]))
   106→   106→  );
   107→   107→  const [expandedDescIds, setExpandedDescIds] = useState<Record<string, boolean>>({});
   108→   108→  const [editingDescId, setEditingDescId] = useState<string | null>(null);
   109→   109→  const [descDraft, setDescDraft] = useState('');
   110→   110→  const [addingGrandchild, setAddingGrandchild] = useState<string | null>(null);
   111→   111→  const [newGcTitle, setNewGcTitle] = useState('');
   112→   112→  const [addingStage, setAddingStage] = useState(false);
   113→   113→  const [newStageTitle, setNewStageTitle] = useState('');
   114→   114→
   115→   115→  const config = task.trackConfig ? JSON.parse(task.trackConfig) : null;
   116→   116→  const allSubtasks = stages.flatMap(s => s.children || []);
   117→   117→  const progress = getProgress(allSubtasks);
   118→   118→  const boardColor = board?.color || '#00d9ff';
   119→   119→  const bc = boardColorStyles(boardColor);
   120→   120→
   121→   121→  const getProgressColor = () => {
   122→   122→    if (progress === 100) return '#10b981';
   123→   123→    if (progress > 50) return boardColor;
   124→   124→    if (progress > 0) return '#f59e0b';
   125→   125→    return '#334155';
   126→   126→  };
   127→   127→  const getProgressTextColor = () => {
   128→   128→    if (progress === 100) return '#34d399';
   129→   129→    if (progress > 50) return boardColor;
   130→   130→    if (progress > 0) return '#f59e0b';
   131→   131→    return '#475569';
   132→   132→  };
   133→   133→
   134→   134→  const reloadTasks = useCallback(async () => {
   135→   135→    if (!selectedBoardId) return;
   136→   136→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   137→   137→    const data = await res.json();
   138→   138→    setBoardTasks(data.tasks);
   139→   139→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   140→   140→    if (updated) setStages(updated.children || []);
   141→   141→  }, [selectedBoardId, setBoardTasks, task.id]);
   142→   142→
   143→   143→  const toggleGrandchildStatus = async (e: React.MouseEvent, id: string) => {
   144→   144→    e.stopPropagation();
   145→   145→    const order: TaskStatus[] = ['todo', 'in-progress', 'review', 'done'];
   146→   146→    const all = stages.flatMap(s => s.children || []);
   147→   147→    const gc = all.find(c => c.id === id);
   148→   148→    if (!gc) return;
   149→   149→    const idx = order.indexOf(gc.status as TaskStatus);
   150→   150→    const next = order[(idx + 1) % order.length];
   151→   151→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id, status: next }) });
   152→   152→    await reloadTasks();
   153→   153→  };
   154→   154→
   155→   155→  const toggleStageExpanded = (stageId: string) => {
   156→   156→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   157→   157→  };
   158→   158→
   159→   159→  const toggleDescExpanded = (id: string) => {
   160→   160→    setExpandedDescIds(prev => ({ ...prev, [id]: !prev[id] }));
   161→   161→  };
   162→   162→
   163→   163→  const startEditDesc = (id: string, current: string) => {
   164→   164→    setEditingDescId(id);
   165→   165→    setDescDraft(current || '');
   166→   166→  };
   167→   167→
   168→   168→  const saveDesc = async (id: string) => {
   169→   169→    await fetch('/api/tasks', {
   170→   170→      method: 'PUT',
   171→   171→      headers: { 'Content-Type': 'application/json' },
   172→   172→      body: JSON.stringify({ id, description: descDraft.trim() || null }),
   173→   173→    });
   174→   174→    setEditingDescId(null);
   175→   175→    setDescDraft('');
   176→   176→    await reloadTasks();
   177→   177→  };
   178→   178→
   179→   179→  const cancelEditDesc = () => {
   180→   180→    setEditingDescId(null);
   181→   181→    setDescDraft('');
   182→   182→  };
   183→   183→
   184→   184→  const deleteGrandchild = async (e: React.MouseEvent, id: string) => {
   185→   185→    e.stopPropagation();
   186→   186→    await fetch(`/api/tasks?id=${id}`, { method: 'DELETE' });
   187→   187→    await reloadTasks();
   188→   188→  };
   189→   189→
   190→   190→  const addGrandchild = async (stageId: string) => {
   191→   191→    if (!newGcTitle.trim()) return;
   192→   192→    await fetch('/api/tasks', {
   193→   193→      method: 'POST',
   194→   194→      headers: { 'Content-Type': 'application/json' },
   195→   195→      body: JSON.stringify({ title: newGcTitle.trim(), parentId: stageId }),
   196→   196→    });
   197→   197→    setNewGcTitle('');
   198→   198→    setAddingGrandchild(null);
   199→   199→    await reloadTasks();
   200→   200→  };
   201→   201→
   202→   202→  const addStage = async () => {
   203→   203→    if (!newStageTitle.trim()) return;
   204→   204→    await fetch('/api/tasks', {
   205→   205→      method: 'POST',
   206→   206→      headers: { 'Content-Type': 'application/json' },
   207→   207→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: selectedBoardId }),
   208→   208→    });
   209→   209→    setNewStageTitle('');
   210→   210→    setAddingStage(false);
   211→   211→    await reloadTasks();
   212→   212→  };
   213→   213→
   214→   214→  const deleteStage = async (e: React.MouseEvent, stageId: string) => {
   215→   215→    e.stopPropagation();
   216→   216→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   217→   217→    await reloadTasks();
   218→   218→  };
   219→   219→
   220→   220→  const handleDelete = async () => {
   221→   221→    await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   222→   222→    setSelectedTaskId(null);
   223→   223→    if (selectedBoardId) {
   224→   224→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   225→   225→      const data = await res.json();
   226→   226→      setBoardTasks(data.tasks);
   227→   227→    }
   228→   228→  };
   229→   229→
   230→   230→  return (
   231→   231→    <div className="flex-1 overflow-y-auto">
   232→   232→      {/* Header */}
   233→   233→      <div className="border-b border-slate-800/50 p-4">
   234→   234→        <div className="flex items-center gap-1.5 mb-2">
   235→   235→          <span className="text-[9px] px-1.5 py-0.5 rounded font-medium uppercase tracking-wider"
   236→   236→            style={{ color: boardColor, backgroundColor: boardColor + '15' }}>
   237→   237→            Трек
   238→   238→          </span>
   239→   239→          <div className="flex-1" />
   240→   240→          <button onClick={() => setEditingTask(task)} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors">
   241→   241→            <Pencil className="w-3.5 h-3.5" />
   242→   242→          </button>
   243→   243→          <button onClick={handleDelete} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors">
   244→   244→            <Trash2 className="w-3.5 h-3.5" />
   245→   245→          </button>
   246→   246→        </div>
   247→   247→        <h3 className="text-sm font-semibold text-white leading-tight mb-1.5">{task.title}</h3>
   248→   248→        {task.description && (
   249→   249→          <p className="text-[11px] text-slate-500 mb-1.5 line-clamp-2">{task.description}</p>
   250→   250→        )}
   251→   251→        {config && (
   252→   252→          <div className="flex flex-wrap gap-1">
   253→   253→            {config.instruments?.map((inst: string) => (
   254→   254→              <span key={inst} className="text-[9px] bg-slate-800/80 text-slate-400 px-1.5 py-0.5 rounded">{inst}</span>
   255→   255→            ))}
   256→   256→          </div>
   257→   257→        )}
   258→   258→      </div>
   259→   259→
   260→   260→      {/* Progress */}
   261→   261→      <div className="border-b border-slate-800/30 px-4 py-3">
   262→   262→        <div className="flex items-center justify-between mb-1.5">
   263→   263→          <span className="text-[11px] text-slate-400 font-medium">Общий прогресс</span>
   264→   264→          <span className="text-[11px] font-bold" style={{ color: getProgressTextColor() }}>{progress}%</span>
   265→   265→        </div>
   266→   266→        <div className="relative h-2 bg-slate-800/80 rounded-full overflow-hidden">
   267→   267→          <div className="h-full rounded-full transition-all duration-700 ease-out" style={{ width: `${progress}%`, backgroundColor: getProgressColor() }} />
   268→   268→          {progress > 0 && progress < 100 && (
   269→   269→            <div
   270→   270→              className="absolute inset-0 rounded-full animate-pulse opacity-20"
   271→   271→              style={{
   272→   272→                background: `linear-gradient(90deg, transparent, ${getProgressColor()}, transparent)`,
   273→   273→              }}
   274→   274→            />
   275→   275→          )}
   276→   276→        </div>
   277→   277→        <div className="flex items-center justify-between mt-1">
   278→   278→          <span className="text-[9px] text-slate-600">{allSubtasks.filter(s => s.status === 'done').length} из {allSubtasks.length} задач</span>
   279→   279→          {progress === 100 && <span className="text-[9px] text-emerald-400 font-medium">✓ Завершено</span>}
   280→   280→        </div>
   281→   281→      </div>
   282→   282→
   283→   283→      {/* Stages as collapsible groups */}
   284→   284→      <div className="px-3 py-2 space-y-1">
   285→   285→        {stages.map((stage) => {
   286→   286→          const grandchildren = stage.children || [];
   287→   287→          const stageProgress = getProgress(grandchildren);
   288→   288→          const stageDone = stage.status === 'done';
   289→   289→          const isExpanded = expandedStages[stage.id] !== false;
   290→   290→          const isCustom = !EMOJI_RE.test(stage.title);
   291→   291→          const StatusIcon = SUBTASK_STATUS_ICON[stage.status] || Circle;
   292→   292→          const hasDesc = !!stage.description;
   293→   293→          const descExpanded = expandedDescIds[stage.id] === true;
   294→   294→          const isEditingDesc = editingDescId === stage.id;
   295→   295→
   296→   296→          return (
   297→   297→            <div key={stage.id} className="rounded-lg border border-slate-800/30 overflow-hidden">
   298→   298→              {/* Stage header — full panel click to expand */}
   299→   299→              <button
   300→   300→                onClick={() => toggleStageExpanded(stage.id)}
   301→   301→                className="w-full flex items-center gap-2 px-2.5 py-2 hover:bg-slate-900/60 transition-colors text-left group/stage"
   302→   302→              >
   303→   303→                <StatusIcon className={cn('w-3.5 h-3.5 flex-shrink-0', SUBTASK_STATUS_COLOR[stage.status])} />
   304→   304→                {isCustom && <Cat className="w-3 h-3 flex-shrink-0 text-amber-400/60" />}
   305→   305→                <span className={cn('flex-1 text-[11px] font-medium min-w-0',
   306→   306→                  stageDone ? 'text-slate-500 line-through' : 'text-slate-300'
   307→   307→                )} onDoubleClick={(e) => { e.stopPropagation(); useKanbanStore.getState().setDescriptionEditorItem({ id: stage.id, title: stage.title, description: stage.description, parentId: task.id, parentTitle: task.title }); }}>
   308→   308→                  {stage.title}
   309→   309→                </span>
   310→   310→                {/* Description icon */}
   311→   311→                <button
   312→   312→                  onClick={(e) => { e.stopPropagation(); if (hasDesc) toggleDescExpanded(stage.id); else startEditDesc(stage.id, stage.description || ''); }}
   313→   313→                  className={cn('flex-shrink-0 transition-opacity',
   314→   314→                    hasDesc ? '' : ''
   315→   315→                  )}
   316→   316→                  style={{ color: hasDesc ? '#64748b' : 'transparent' }}
   317→   317→                  onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.color = boardColor; (e.currentTarget as HTMLElement).style.opacity = '1'; }}
   318→   318→                  onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.color = hasDesc ? '#64748b' : 'transparent'; (e.currentTarget as HTMLElement).style.opacity = hasDesc ? '1' : '0'; }}
   319→   319→                >
   320→   320→                  <MessageSquare className="w-3 h-3" />
   321→   321→                </button>
   322→   322→                {grandchildren.length > 0 && (
   323→   323→                  <span className="text-[9px] text-slate-600 mr-1">{stageDone ? '✓' : `${stageProgress}%`}</span>
   324→   324→                )}
   325→   325→                <button
   326→   326→                  onClick={(e) => { e.stopPropagation(); deleteStage(e, stage.id); }}
   327→   327→                  className="flex-shrink-0 p-0.5 rounded opacity-0 group-hover/stage:opacity-100 hover:bg-rose-500/10 text-slate-600 hover:text-rose-400 transition-all"
   328→   328→                >
   329→   329→                  <Trash2 className="w-3 h-3" />
   330→   330→                </button>
   331→   331→                {isExpanded
   332→   332→                  ? <ChevronUp className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   333→   333→                  : <ChevronDown className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   334→   334→                }
   335→   335→              </button>
   336→   336→
   337→   337→              {/* Expanded content */}
   338→   338→              {isExpanded && (
   339→   339→                <div>
   340→   340→                  {/* Description card */}
   341→   341→                  {hasDesc && !isEditingDesc && (
   342→   342→                    <div className="mx-2.5 mb-1.5">
   343→   343→                      <div className="bg-slate-900/60 border border-slate-800/40 rounded-md p-2">
   344→   344→                        {descExpanded ? (
   345→   345→                          <>
   346→   346→                            <p className="text-[10px] text-slate-400 whitespace-pre-wrap leading-relaxed">
   347→   347→                              {stage.description && stage.description.length > DESC_LIMIT
   348→   348→                                ? stage.description.slice(0, DESC_LIMIT) + '...'
   349→   349→                                : stage.description}
   350→   350→                            </p>
   351→   351→                            <div className="flex items-center gap-2 mt-1.5">
   352→   352→                              <button
   353→   353→                                onClick={() => startEditDesc(stage.id, stage.description || '')}
   354→   354→                                className="text-[9px] transition-colors"
   355→   355→                                style={{ color: boardColor + '80' }}
   356→   356→                                onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.color = boardColor; }}
   357→   357→                                onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.color = boardColor + '80'; }}
   358→   358→                              >Редактировать</button>
   359→   359→                              <button
   360→   360→                                onClick={() => toggleDescExpanded(stage.id)}
   361→   361→                                className="text-[9px] text-slate-600 hover:text-slate-400 transition-colors"
   362→   362→                              >Свернуть</button>
   363→   363→                            </div>
   364→   364→                          </>
   365→   365→                        ) : (
   366→   366→                          <p className="text-[10px] text-slate-500 line-clamp-2">
   367→   367→                            {stage.description}
   368→   368→                          </p>
   369→   369→                        )}
   370→   370→                      </div>
   371→   371→                    </div>
   372→   372→                  )}
   373→   373→                  {/* Editing description */}
   374→   374→                  {isEditingDesc && (
   375→   375→                    <div className="mx-2.5 mb-1.5">
   376→   376→                      <div className="bg-slate-900/60 rounded-md p-2 space-y-1.5" style={{ border: `1px solid ${boardColor}30` }}>
   377→   377→                        <Textarea
   378→   378→                          value={descDraft}
   379→   379→                          onChange={(e) => setDescDraft(e.target.value)}
   380→   380→                          placeholder="Описание этапа..."
   381→   381→                          maxLength={DESC_LIMIT}
   382→   382→                          className="bg-slate-800/80 border-slate-700/50 text-[10px] text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none"
   383→   383→                          autoFocus
   384→   384→                        />
   385→   385→                        <div className="flex items-center justify-between">
   386→   386→                          <span className="text-[8px] text-slate-600">{descDraft.length}/{DESC_LIMIT}</span>

   572→   572→function TaskDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   573→   573→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   574→   574→  const [subtasks, setSubtasks] = useState<TaskChild[]>(task.children || []);
   575→   575→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
   576→   576→  const [newSubtaskDeadline, setNewSubtaskDeadline] = useState<string | null>(null);
   577→   577→  const [addingSubtask, setAddingSubtask] = useState(false);
   578→   578→  const [expandedSubtasks, setExpandedSubtasks] = useState(true);
   579→   579→
   580→   580→  const boardColor = board?.color || '#00d9ff';
   581→   581→  const progress = getProgress(subtasks);
   582→   582→
   583→   583→  const getProgressColor = () => {
   584→   584→    if (progress === 100) return '#10b981';
   585→   585→    if (progress > 50) return boardColor;
   586→   586→    if (progress > 0) return '#f59e0b';
   587→   587→    return '#334155';
   588→   588→  };
   589→   589→  const getProgressTextColor = () => {
   590→   590→    if (progress === 100) return '#34d399';
   591→   591→    if (progress > 50) return boardColor;
   592→   592→    if (progress > 0) return '#f59e0b';
   593→   593→    return '#475569';
   594→   594→  };
   595→   595→
   596→   596→  const reloadTasks = useCallback(async () => {
   597→   597→    if (!selectedBoardId) return;
   598→   598→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   599→   599→    const data = await res.json();
   600→   600→    setBoardTasks(data.tasks);
   601→   601→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   602→   602→    if (updated) setSubtasks(updated.children || []);
   603→   603→  }, [selectedBoardId, setBoardTasks, task.id]);
   604→   604→
   605→   605→  const cycleSubtaskStatus = async (e: React.MouseEvent, subtaskId: string) => {
   606→   606→    e.stopPropagation();
   607→   607→    const order: TaskStatus[] = ['todo', 'in-progress', 'review', 'done'];
   608→   608→    const sub = subtasks.find(s => s.id === subtaskId);
   609→   609→    if (!sub) return;
   610→   610→    const idx = order.indexOf(sub.status as TaskStatus);
   611→   611→    const next = order[(idx + 1) % order.length];
   612→   612→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: subtaskId, status: next }) });
   613→   613→    await reloadTasks();
   614→   614→  };
   615→   615→
   616→   616→  const deleteSubtask = async (e: React.MouseEvent, subtaskId: string) => {
   617→   617→    e.stopPropagation();
   618→   618→    await fetch(`/api/tasks?id=${subtaskId}`, { method: 'DELETE' });
   619→   619→    await reloadTasks();
   620→   620→  };
   621→   621→
   622→   622→  const addSubtask = async () => {
   623→   623→    if (!newSubtaskTitle.trim()) return;
   624→   624→    await fetch('/api/tasks', {
   625→   625→      method: 'POST', headers: { 'Content-Type': 'application/json' },

   627→   627→    });

   630→   630→  };

   635→   635→  };

   661→   661→            task.status === 'todo' ? '' :
   662→   662→            task.status === 'in-progress' ? 'text-orange-400 bg-orange-500/10' :

   672→   672→          <button onClick={handleDeleteTask} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors"><Trash2 className="w-3.5 h-3.5" /></button>

   719→   719→        </button>
   720→   720→        {expandedSubtasks && (
   721→   721→          <div className="space-y-1.5">
   722→   722→            {subtasks.length === 0 && !addingSubtask && <p className="text-[10px] text-slate-600 py-2 text-center">Нет подзадач</p>}
   723→   723→            {subtasks.map((sub) => {
   724→   724→              const Icon = SUBTASK_STATUS_ICON[sub.status] || Circle;
   725→   725→              return (
   726→   726→                <div
   727→   727→                  key={sub.id}
   728→   728→                  className="group/sub flex items-center gap-2 px-2 py-1.5 rounded-lg hover:bg-slate-900/80 transition-colors cursor-pointer"
   729→   729→                  onClick={() => {

   731→   731→                      id: sub.id, title: sub.title, description: sub.description,
   732→   732→                      parentId: task.id, parentTitle: task.title,
   733→   733→                    });
   734→   734→                  }}
   735→   735→                >
   736→   736→                  <button onClick={(e) => cycleSubtaskStatus(e, sub.id)} className={cn('flex-shrink-0 transition-opacity hover:opacity-70', SUBTASK_STATUS_COLOR[sub.status] || 'text-slate-500')}><Icon className="w-3.5 h-3.5" /></button>
   737→   737→                  <span className={cn('flex-1 text-[11px] leading-tight min-w-0', sub.status === 'done' ? 'text-slate-500 line-through' : 'text-slate-300')}>{sub.title}</span>

   741→   741→                  <button onClick={(e) => deleteSubtask(e, sub.id)} className="p-0.5 rounded opacity-0 group-hover/sub:opacity-100 hover:bg-slate-800 text-slate-600 hover:text-rose-400 transition-all flex-shrink-0"><X className="w-3 h-3" /></button>
   742→   742→                </div>
   743→   743→              );

   755→   755→                </div>

   761→   761→        )}
   762→   762→      </div>
   763→   763→    </div>
   764→   764→  );

   768→   768→
   769→   769→function TaskForm({ mode, task, boardColor }: { mode: 'create' | 'edit'; task?: Task; boardColor?: string }) {
   770→   770→  const { selectedBoardId, setBoardTasks, selectedProjectId, setEditingTask, setIsCreating, setSelectedTaskId } = useKanbanStore();
   771→   771→  const color = boardColor || '#00d9ff';
   772→   772→  const [title, setTitle] = useState('');
   773→   773→  const [description, setDescription] = useState('');
   774→   774→  const [status, setStatus] = useState<TaskStatus>('todo');
   775→   775→  const [priority, setPriority] = useState<TaskPriority>('medium');
   776→   776→  const [assignee, setAssignee] = useState('');
   777→   777→  const [category, setCategory] = useState<TaskCategory>('general');
   778→   778→  const [deadline, setDeadline] = useState<string | null>(null);
   779→   779→  const [saving, setSaving] = useState(false);
   780→   780→
   781→   781→  useEffect(() => {
   782→   782→    if (mode === 'edit' && task) {
   783→   783→      setTitle(task.title); setDescription(task.description || ''); setStatus(task.status);
   784→   784→      setPriority(task.priority); setAssignee(task.assignee || ''); setCategory(task.category);
   785→   785→      setDeadline(task.deadline || null);
   786→   786→    }
   787→   787→  }, [mode, task]);
   788→   788→
   789→   789→  const reloadTasks = useCallback(async () => {
   790→   790→    if (selectedBoardId) {
   791→   791→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   792→   792→      const data = await res.json();
   793→   793→      setBoardTasks(data.tasks);
   794→   794→    }
   795→   795→    if (selectedProjectId) {
   796→   796→      const bRes = await fetch(`/api/boards?projectId=${selectedProjectId}`);
   797→   797→      const bData = await bRes.json();
   798→   798→      useKanbanStore.getState().setBoards(bData.boards);
   799→   799→    }
   800→   800→  }, [selectedBoardId, selectedProjectId, setBoardTasks]);
   801→   801→
   802→   802→  const handleSave = async () => {
   803→   803→    if (!title.trim()) return;
   804→   804→    setSaving(true);
   805→   805→    try {
   806→   806→      if (mode === 'edit' && task) {
   807→   807→        await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: task.id, title, description: description || null, status, priority, assignee: assignee || null, category, boardId: selectedBoardId, deadline }) });
   808→   808→      } else {
   809→   809→        await fetch('/api/tasks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title, description: description || null, status, priority, assignee: assignee || null, category, parentId: null, boardId: selectedBoardId, deadline }) });
   810→   810→      }
   811→   811→      await reloadTasks(); setEditingTask(null); setIsCreating(false);
   812→   812→      if (mode === 'edit' && task) setSelectedTaskId(task.id);
   813→   813→    } finally { setSaving(false); }
   814→   814→  };
   815→   815→
   816→   816→  const handleDelete = async () => {
   817→   817→    if (!task) return;
   818→   818→    setSaving(true);
   819→   819→    try {
   820→   820→      await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   821→   821→      await reloadTasks(); setEditingTask(null); setSelectedTaskId(null);
   822→   822→    } finally { setSaving(false); }
   823→   823→  };
   824→   824→
   825→   825→  return (
   826→   826→    <div className="flex-1 overflow-y-auto">
   827→   827→      <div className="p-4 space-y-3">
   828→   828→        <div className="flex items-center justify-between">
   829→   829→          <h3 className="text-xs font-semibold uppercase tracking-wider" style={{ color }}>{mode === 'edit' ? 'Редактировать' : 'Новая задача'}</h3>
   830→   830→          <button onClick={() => { setEditingTask(null); setIsCreating(false); }} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300"><X className="h-3.5 w-3.5" /></button>
   831→   831→        </div>
   832→   832→        <div className="space-y-2.5">
   833→   833→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Название</Label><Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Задача..." className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-8" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   834→   834→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Описание</Label><Textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Описание..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   835→   835→          <div className="grid grid-cols-2 gap-2">
   836→   836→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Статус</Label><Select value={status} onValueChange={(v) => setStatus(v as TaskStatus)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{STATUSES.map((s) => (<SelectItem key={s.value} value={s.value}><span className={s.color}>{s.label}</span></SelectItem>))}</SelectContent></Select></div>
   837→   837→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Приоритет</Label><Select value={priority} onValueChange={(v) => setPriority(v as TaskPriority)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{PRIORITIES.map((p) => (<SelectItem key={p.value} value={p.value}><div className="flex items-center gap-1.5"><div className={cn('w-1.5 h-1.5 rounded-full', p.dot)} />{p.label}</div></SelectItem>))}</SelectContent></Select></div>
   838→   838→          </div>
   839→   839→          <div className="grid grid-cols-2 gap-2">
   840→   840→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Категория</Label><Select value={category} onValueChange={(v) => setCategory(v as TaskCategory)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{CATEGORIES.map((c) => (<SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>))}</SelectContent></Select></div>
   841→   841→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Ответственный</Label><Input value={assignee} onChange={(e) => setAssignee(e.target.value)} placeholder="Имя..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 placeholder:text-slate-600 h-8" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   842→   842→          </div>
   843→   843→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Дедлайн</Label><DeadlinePicker value={deadline} onChange={setDeadline} size="md" /></div>
   844→   844→        </div>
   845→   845→        <div className="flex gap-2 pt-1">
   846→   846→          <Button onClick={handleSave} disabled={!title.trim() || saving} className="flex-1 text-white h-8 text-xs" style={{ backgroundColor: color, boxShadow: `0 4px 14px ${color}30` }}><Save className="w-3 h-3 mr-1" />{saving ? '...' : mode === 'edit' ? 'Сохранить' : 'Создать'}</Button>
   847→   847→          {mode === 'edit' && <Button onClick={handleDelete} variant="destructive" disabled={saving} className="h-8 text-xs">Удалить</Button>}
   848→   848→        </div>
   849→   849→      </div>
   850→   850→    </div>
   851→   851→  );
   852→   852→}

   854→   854→function getProgress(children: { status: string }[]): number {
   855→   855→  if (children.length === 0) return 0;
   856→   856→  return Math.round((children.filter(c => c.status === 'done').length / children.length) * 100);
