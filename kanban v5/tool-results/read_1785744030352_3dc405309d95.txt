    99→    99→function TrackDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   100→   100→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   101→   101→  const [stages, setStages] = useState<TaskChild[]>(task.children || []);
   102→   102→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(
   103→   103→    Object.fromEntries((task.children || []).map(c => [c.id, true]))
   104→   104→  );
   105→   105→  const [expandedDescIds, setExpandedDescIds] = useState<Record<string, boolean>>({});
   106→   106→  const [editingDescId, setEditingDescId] = useState<string | null>(null);
   107→   107→  const [descDraft, setDescDraft] = useState('');
   108→   108→  const [addingGrandchild, setAddingGrandchild] = useState<string | null>(null);
   109→   109→  const [newGcTitle, setNewGcTitle] = useState('');
   110→   110→  const [addingStage, setAddingStage] = useState(false);
   111→   111→  const [newStageTitle, setNewStageTitle] = useState('');
   112→   112→
   113→   113→  const config = task.trackConfig ? JSON.parse(task.trackConfig) : null;
   114→   114→  const allSubtasks = stages.flatMap(s => s.children || []);
   115→   115→  const progress = getProgress(allSubtasks);
   116→   116→  const progressColor = progress === 100 ? 'bg-emerald-500' : progress > 50 ? 'bg-cyan-500' : progress > 0 ? 'bg-amber-500' : 'bg-slate-700';
   117→   117→
   118→   118→  const reloadTasks = useCallback(async () => {
   119→   119→    if (!selectedBoardId) return;
   120→   120→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   121→   121→    const data = await res.json();
   122→   122→    setBoardTasks(data.tasks);
   123→   123→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   124→   124→    if (updated) setStages(updated.children || []);
   125→   125→  }, [selectedBoardId, setBoardTasks, task.id]);
   126→   126→
   127→   127→  const toggleGrandchildStatus = async (e: React.MouseEvent, id: string) => {
   128→   128→    e.stopPropagation();

   130→   130→    const all = stages.flatMap(s => s.children || []);
   131→   131→    const gc = all.find(c => c.id === id);
   132→   132→    if (!gc) return;
   133→   133→    const idx = order.indexOf(gc.status as TaskStatus);
   134→   134→    const next = order[(idx + 1) % order.length];
   135→   135→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id, status: next }) });
   136→   136→    await reloadTasks();
   137→   137→  };
   138→   138→
   139→   139→  const toggleStageExpanded = (stageId: string) => {
   140→   140→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   141→   141→  };
   142→   142→
   143→   143→  const toggleDescExpanded = (id: string) => {
   144→   144→    setExpandedDescIds(prev => ({ ...prev, [id]: !prev[id] }));
   145→   145→  };
   146→   146→
   147→   147→  const startEditDesc = (id: string, current: string) => {
   148→   148→    setEditingDescId(id);
   149→   149→    setDescDraft(current || '');
   150→   150→  };
   151→   151→
   152→   152→  const saveDesc = async (id: string) => {
   153→   153→    await fetch('/api/tasks', {
   154→   154→      method: 'PUT',
   155→   155→      headers: { 'Content-Type': 'application/json' },
   156→   156→      body: JSON.stringify({ id, description: descDraft.trim() || null }),
   157→   157→    });
   158→   158→    setEditingDescId(null);
   159→   159→    setDescDraft('');
   160→   160→    await reloadTasks();
   161→   161→  };
   162→   162→
   163→   163→  const cancelEditDesc = () => {
   164→   164→    setEditingDescId(null);
   165→   165→    setDescDraft('');
   166→   166→  };
   167→   167→
   168→   168→  const deleteGrandchild = async (e: React.MouseEvent, id: string) => {
   169→   169→    e.stopPropagation();
   170→   170→    await fetch(`/api/tasks?id=${id}`, { method: 'DELETE' });
   171→   171→    await reloadTasks();
   172→   172→  };
   173→   173→
   174→   174→  const addGrandchild = async (stageId: string) => {
   175→   175→    if (!newGcTitle.trim()) return;
   176→   176→    await fetch('/api/tasks', {
   177→   177→      method: 'POST',
   178→   178→      headers: { 'Content-Type': 'application/json' },
   179→   179→      body: JSON.stringify({ title: newGcTitle.trim(), parentId: stageId }),
   180→   180→    });
   181→   181→    setNewGcTitle('');
   182→   182→    setAddingGrandchild(null);
   183→   183→    await reloadTasks();
   184→   184→  };
   185→   185→
   186→   186→  const addStage = async () => {
   187→   187→    if (!newStageTitle.trim()) return;
   188→   188→    await fetch('/api/tasks', {
   189→   189→      method: 'POST',
   190→   190→      headers: { 'Content-Type': 'application/json' },
   191→   191→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: selectedBoardId }),
   192→   192→    });
   193→   193→    setNewStageTitle('');
   194→   194→    setAddingStage(false);
   195→   195→    await reloadTasks();
   196→   196→  };
   197→   197→
   198→   198→  const deleteStage = async (e: React.MouseEvent, stageId: string) => {
   199→   199→    e.stopPropagation();
   200→   200→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   201→   201→    await reloadTasks();
   202→   202→  };
   203→   203→
   204→   204→  const handleDelete = async () => {
   205→   205→    await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   206→   206→    setSelectedTaskId(null);
   207→   207→    if (selectedBoardId) {
   208→   208→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   209→   209→      const data = await res.json();
   210→   210→      setBoardTasks(data.tasks);
   211→   211→    }
   212→   212→  };
   213→   213→
   214→   214→  return (
   215→   215→    <div className="flex-1 overflow-y-auto">
   216→   216→      {/* Header */}
   217→   217→      <div className="border-b border-slate-800/50 p-4">
   218→   218→        <div className="flex items-center gap-1.5 mb-2">
   219→   219→          <span className="text-[9px] text-purple-400 bg-purple-500/15 px-1.5 py-0.5 rounded font-medium uppercase tracking-wider">
   220→   220→            Трек

   224→   224→            <Pencil className="w-3.5 h-3.5" />
   225→   225→          </button>
   226→   226→          <button onClick={handleDelete} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors">
   227→   227→            <Trash2 className="w-3.5 h-3.5" />
   228→   228→          </button>
   229→   229→        </div>
   230→   230→        <h3 className="text-sm font-semibold text-white leading-tight mb-1.5">{task.title}</h3>
   231→   231→        {task.description && (
   232→   232→          <p className="text-[11px] text-slate-500 mb-1.5 line-clamp-2">{task.description}</p>
   233→   233→        )}
   234→   234→        {config && (
   235→   235→          <div className="flex flex-wrap gap-1">
   236→   236→            {config.instruments?.map((inst: string) => (
   237→   237→              <span key={inst} className="text-[9px] bg-slate-800/80 text-slate-400 px-1.5 py-0.5 rounded">{inst}</span>
   238→   238→            ))}
   239→   239→          </div>
   240→   240→        )}
   241→   241→      </div>
   242→   242→
   243→   243→      {/* Progress */}
   244→   244→      <div className="border-b border-slate-800/30 px-4 py-3">
   245→   245→        <div className="flex items-center justify-between mb-1.5">
   246→   246→          <span className="text-[11px] text-slate-400 font-medium">Общий прогресс</span>
   247→   247→          <span className={cn('text-[11px] font-bold',
   248→   248→            progressColor === 'bg-emerald-500' ? 'text-emerald-400' : progressColor === 'bg-cyan-500' ? 'text-cyan-400' : progressColor === 'bg-amber-500' ? 'text-amber-400' : 'text-slate-600'
   249→   249→          )}>{progress}%</span>
   250→   250→        </div>
   251→   251→        <div className="relative h-2 bg-slate-800/80 rounded-full overflow-hidden">
   252→   252→          <div className={cn('h-full rounded-full transition-all duration-700 ease-out', progressColor)} style={{ width: `${progress}%` }} />
   253→   253→          {progress > 0 && progress < 100 && (
   254→   254→            <div
   255→   255→              className="absolute inset-0 rounded-full animate-pulse opacity-20"
   256→   256→              style={{
   257→   257→                background: `linear-gradient(90deg, transparent, ${progressColor === 'bg-emerald-500' ? '#10b981' : progressColor === 'bg-cyan-500' ? '#00d9ff' : progressColor === 'bg-amber-500' ? '#f59e0b' : '#475569'}, transparent)`,
   258→   258→              }}
   259→   259→            />
   260→   260→          )}
   261→   261→        </div>
   262→   262→        <div className="flex items-center justify-between mt-1">
   263→   263→          <span className="text-[9px] text-slate-600">{allSubtasks.filter(s => s.status === 'done').length} из {allSubtasks.length} задач</span>
   264→   264→          {progress === 100 && <span className="text-[9px] text-emerald-400 font-medium">✓ Завершено</span>}
   265→   265→        </div>
   266→   266→      </div>
   267→   267→
   268→   268→      {/* Stages as collapsible groups */}
   269→   269→      <div className="px-3 py-2 space-y-1">
   270→   270→        {stages.map((stage) => {
   271→   271→          const grandchildren = stage.children || [];
   272→   272→          const stageProgress = getProgress(grandchildren);
   273→   273→          const stageDone = stage.status === 'done';
   274→   274→          const isExpanded = expandedStages[stage.id] !== false;
   275→   275→          const isCustom = !EMOJI_RE.test(stage.title);
   276→   276→          const StatusIcon = SUBTASK_STATUS_ICON[stage.status] || Circle;
   277→   277→          const hasDesc = !!stage.description;
   278→   278→          const descExpanded = expandedDescIds[stage.id] === true;
   279→   279→          const isEditingDesc = editingDescId === stage.id;
   280→   280→
   281→   281→          return (
   282→   282→            <div key={stage.id} className="rounded-lg border border-slate-800/30 overflow-hidden">
   283→   283→              {/* Stage header — full panel click to expand */}
   284→   284→              <button
   285→   285→                onClick={() => toggleStageExpanded(stage.id)}
   286→   286→                className="w-full flex items-center gap-2 px-2.5 py-2 hover:bg-slate-900/60 transition-colors text-left group/stage"
   287→   287→              >
   288→   288→                <StatusIcon className={cn('w-3.5 h-3.5 flex-shrink-0', SUBTASK_STATUS_COLOR[stage.status])} />
   289→   289→                {isCustom && <Cat className="w-3 h-3 flex-shrink-0 text-amber-400/60" />}
   290→   290→                <span className={cn('flex-1 text-[11px] font-medium min-w-0',
   291→   291→                  stageDone ? 'text-slate-500 line-through' : 'text-slate-300'
   292→   292→                )} onDoubleClick={(e) => { e.stopPropagation(); useKanbanStore.getState().setDescriptionEditorItem({ id: stage.id, title: stage.title, description: stage.description, parentId: task.id, parentTitle: task.title }); }}>
   293→   293→                  {stage.title}
   294→   294→                </span>
   295→   295→                {/* Description icon */}
   296→   296→                <button
   297→   297→                  onClick={(e) => { e.stopPropagation(); if (hasDesc) toggleDescExpanded(stage.id); else startEditDesc(stage.id, stage.description || ''); }}
   298→   298→                  className={cn('flex-shrink-0 transition-opacity',
   299→   299→                    hasDesc ? 'text-slate-500 hover:text-cyan-400' : 'text-slate-700 opacity-0 group-hover/stage:opacity-100 hover:text-cyan-400'
   300→   300→                  )}
   301→   301→                >
   302→   302→                  <MessageSquare className="w-3 h-3" />
   303→   303→                </button>
   304→   304→                {grandchildren.length > 0 && (
   305→   305→                  <span className="text-[9px] text-slate-600 mr-1">{stageDone ? '✓' : `${stageProgress}%`}</span>
   306→   306→                )}
   307→   307→                <button
   308→   308→                  onClick={(e) => { e.stopPropagation(); deleteStage(e, stage.id); }}
   309→   309→                  className="flex-shrink-0 p-0.5 rounded opacity-0 group-hover/stage:opacity-100 hover:bg-rose-500/10 text-slate-600 hover:text-rose-400 transition-all"
   310→   310→                >
   311→   311→                  <Trash2 className="w-3 h-3" />
   312→   312→                </button>
   313→   313→                {isExpanded
   314→   314→                  ? <ChevronUp className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   315→   315→                  : <ChevronDown className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   316→   316→                }
   317→   317→              </button>
   318→   318→
   319→   319→              {/* Expanded content */}
   320→   320→              {isExpanded && (
   321→   321→                <div>
   322→   322→                  {/* Description card */}
   323→   323→                  {hasDesc && !isEditingDesc && (
   324→   324→                    <div className="mx-2.5 mb-1.5">
   325→   325→                      <div className="bg-slate-900/60 border border-slate-800/40 rounded-md p-2">
   326→   326→                        {descExpanded ? (
   327→   327→                          <>
   328→   328→                            <p className="text-[10px] text-slate-400 whitespace-pre-wrap leading-relaxed">
   329→   329→                              {stage.description && stage.description.length > DESC_LIMIT
   330→   330→                                ? stage.description.slice(0, DESC_LIMIT) + '...'
   331→   331→                                : stage.description}
   332→   332→                            </p>
   333→   333→                            <div className="flex items-center gap-2 mt-1.5">
   334→   334→                              <button
   335→   335→                                onClick={() => startEditDesc(stage.id, stage.description || '')}
   336→   336→                                className="text-[9px] text-cyan-500/70 hover:text-cyan-400 transition-colors"
   337→   337→                              >Редактировать</button>
   338→   338→                              <button
   339→   339→                                onClick={() => toggleDescExpanded(stage.id)}
   340→   340→                                className="text-[9px] text-slate-600 hover:text-slate-400 transition-colors"
   341→   341→                              >Свернуть</button>
   342→   342→                            </div>
   343→   343→                          </>
   344→   344→                        ) : (
   345→   345→                          <p className="text-[10px] text-slate-500 line-clamp-2">
   346→   346→                            {stage.description}
   347→   347→                          </p>
   348→   348→                        )}
   349→   349→                      </div>
   350→   350→                    </div>
   351→   351→                  )}
   352→   352→                  {/* Editing description */}
   353→   353→                  {isEditingDesc && (
   354→   354→                    <div className="mx-2.5 mb-1.5">
   355→   355→                      <div className="bg-slate-900/60 border border-cyan-500/30 rounded-md p-2 space-y-1.5">
   356→   356→                        <Textarea
   357→   357→                          value={descDraft}
   358→   358→                          onChange={(e) => setDescDraft(e.target.value)}
   359→   359→                          placeholder="Описание этапа..."
   360→   360→                          maxLength={DESC_LIMIT}
   361→   361→                          className="bg-slate-800/80 border-slate-700/50 text-[10px] text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none"
   362→   362→                          autoFocus
   363→   363→                        />
   364→   364→                        <div className="flex items-center justify-between">
   365→   365→                          <span className="text-[8px] text-slate-600">{descDraft.length}/{DESC_LIMIT}</span>
   366→   366→                          <div className="flex items-center gap-1">
   367→   367→                            <button onClick={cancelEditDesc} className="text-[9px] text-slate-500 hover:text-slate-300 px-1.5 py-0.5 rounded hover:bg-slate-800 transition-colors">Отмена</button>
   368→   368→                            <button
   369→   369→                              onClick={() => saveDesc(stage.id)}
   370→   370→                              className="text-[9px] text-cyan-400 bg-cyan-500/10 px-1.5 py-0.5 rounded hover:bg-cyan-500/20 transition-colors"
   371→   371→                            >Сохранить</button>
   372→   372→                          </div>
   373→   373→                        </div>
   374→   374→                      </div>
   375→   375→                    </div>
   376→   376→                  )}
   377→   377→
   378→   378→                  {/* Progress bar */}
   379→   379→                  {grandchildren.length > 0 && (
   380→   380→                    <div className="px-2.5 pb-1.5">
   381→   381→                      <div className="h-1 bg-slate-800/60 rounded-full overflow-hidden">

   542→   542→function TaskDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   543→   543→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   544→   544→  const [subtasks, setSubtasks] = useState<TaskChild[]>(task.children || []);
   545→   545→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
   546→   546→  const [newSubtaskDeadline, setNewSubtaskDeadline] = useState<string | null>(null);
   547→   547→  const [addingSubtask, setAddingSubtask] = useState(false);
   548→   548→  const [expandedSubtasks, setExpandedSubtasks] = useState(true);
   549→   549→
   550→   550→  const progress = getProgress(subtasks);
   551→   551→  const progressColor = progress === 100 ? 'bg-emerald-500' : progress > 50 ? 'bg-cyan-500' : progress > 0 ? 'bg-amber-500' : 'bg-slate-700';

   553→   553→  const reloadTasks = useCallback(async () => {
   554→   554→    if (!selectedBoardId) return;
   555→   555→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   556→   556→    const data = await res.json();
   557→   557→    setBoardTasks(data.tasks);
   558→   558→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   559→   559→    if (updated) setSubtasks(updated.children || []);

   566→   566→    if (!sub) return;

   569→   569→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: subtaskId, status: next }) });
   570→   570→    await reloadTasks();

   576→   576→    await reloadTasks();
   577→   577→  };

   591→   591→    await reloadTasks();
   592→   592→  };

   597→   597→  };

   607→   607→  };

   616→   616→          )}
   617→   617→          <span className={cn('text-[10px] font-medium px-1.5 py-0.5 rounded',

   627→   627→          <button onClick={handleDeleteTask} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors"><Trash2 className="w-3.5 h-3.5" /></button>
   628→   628→        </div>

   708→   708→                </div>
   709→   709→                <div className="flex gap-1.5 pt-0.5">
   710→   710→                  <button onClick={addSubtask} disabled={!newSubtaskTitle.trim()} className="flex-1 text-[10px] bg-cyan-600 hover:bg-cyan-700 disabled:opacity-40 disabled:hover:bg-cyan-600 text-white rounded px-2 py-1.5 transition-colors font-medium">Добавить</button>
   711→   711→                  <button onClick={() => { setAddingSubtask(false); setNewSubtaskTitle(''); setNewSubtaskDeadline(null); }} className="text-[10px] text-slate-500 hover:text-slate-300 px-2 py-1.5 transition-colors">Отмена</button>
   712→   712→                </div>
   713→   713→              </div>
   714→   714→            ) : (

   718→   718→        )}
   719→   719→      </div>
   720→   720→    </div>
   721→   721→  );

   726→   726→function TaskForm({ mode, task }: { mode: 'create' | 'edit'; task?: Task }) {
   727→   727→  const { selectedBoardId, setBoardTasks, selectedProjectId, setEditingTask, setIsCreating, setSelectedTaskId } = useKanbanStore();
   728→   728→  const [title, setTitle] = useState('');
   729→   729→  const [description, setDescription] = useState('');
   730→   730→  const [status, setStatus] = useState<TaskStatus>('todo');
   731→   731→  const [priority, setPriority] = useState<TaskPriority>('medium');
   732→   732→  const [assignee, setAssignee] = useState('');
   733→   733→  const [category, setCategory] = useState<TaskCategory>('general');
   734→   734→  const [deadline, setDeadline] = useState<string | null>(null);
   735→   735→  const [saving, setSaving] = useState(false);
   736→   736→
   737→   737→  useEffect(() => {
   738→   738→    if (mode === 'edit' && task) {
   739→   739→      setTitle(task.title); setDescription(task.description || ''); setStatus(task.status);
   740→   740→      setPriority(task.priority); setAssignee(task.assignee || ''); setCategory(task.category);
   741→   741→      setDeadline(task.deadline || null);
   742→   742→    }
   743→   743→  }, [mode, task]);
   744→   744→
   745→   745→  const reloadTasks = useCallback(async () => {
   746→   746→    if (selectedBoardId) {
   747→   747→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   748→   748→      const data = await res.json();
   749→   749→      setBoardTasks(data.tasks);
   750→   750→    }
   751→   751→    if (selectedProjectId) {
   752→   752→      const bRes = await fetch(`/api/boards?projectId=${selectedProjectId}`);
   753→   753→      const bData = await bRes.json();
   754→   754→      useKanbanStore.getState().setBoards(bData.boards);
   755→   755→    }
   756→   756→  }, [selectedBoardId, selectedProjectId, setBoardTasks]);
   757→   757→
   758→   758→  const handleSave = async () => {
   759→   759→    if (!title.trim()) return;
   760→   760→    setSaving(true);
   761→   761→    try {
   762→   762→      if (mode === 'edit' && task) {
   763→   763→        await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: task.id, title, description: description || null, status, priority, assignee: assignee || null, category, boardId: selectedBoardId, deadline }) });
   764→   764→      } else {
   765→   765→        await fetch('/api/tasks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title, description: description || null, status, priority, assignee: assignee || null, category, parentId: null, boardId: selectedBoardId, deadline }) });
   766→   766→      }
   767→   767→      await reloadTasks(); setEditingTask(null); setIsCreating(false);
   768→   768→      if (mode === 'edit' && task) setSelectedTaskId(task.id);
   769→   769→    } finally { setSaving(false); }
   770→   770→  };
   771→   771→
   772→   772→  const handleDelete = async () => {
   773→   773→    if (!task) return;
   774→   774→    setSaving(true);
   775→   775→    try {
   776→   776→      await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   777→   777→      await reloadTasks(); setEditingTask(null); setSelectedTaskId(null);
   778→   778→    } finally { setSaving(false); }
   779→   779→  };
   780→   780→
   781→   781→  return (
   782→   782→    <div className="flex-1 overflow-y-auto">
   783→   783→      <div className="p-4 space-y-3">
   784→   784→        <div className="flex items-center justify-between">
   785→   785→          <h3 className="text-xs font-semibold text-slate-300 uppercase tracking-wider">{mode === 'edit' ? 'Редактировать' : 'Новая задача'}</h3>
   786→   786→          <button onClick={() => { setEditingTask(null); setIsCreating(false); }} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300"><X className="h-3.5 w-3.5" /></button>
   787→   787→        </div>
   788→   788→        <div className="space-y-2.5">
   789→   789→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Название</Label><Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Задача..." className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-8 focus:border-cyan-500/50" /></div>
   790→   790→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Описание</Label><Textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Описание..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none focus:border-cyan-500/50" /></div>
   791→   791→          <div className="grid grid-cols-2 gap-2">
   792→   792→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Статус</Label><Select value={status} onValueChange={(v) => setStatus(v as TaskStatus)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{STATUSES.map((s) => (<SelectItem key={s.value} value={s.value}><span className={s.color}>{s.label}</span></SelectItem>))}</SelectContent></Select></div>
   793→   793→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Приоритет</Label><Select value={priority} onValueChange={(v) => setPriority(v as TaskPriority)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{PRIORITIES.map((p) => (<SelectItem key={p.value} value={p.value}><div className="flex items-center gap-1.5"><div className={cn('w-1.5 h-1.5 rounded-full', p.dot)} />{p.label}</div></SelectItem>))}</SelectContent></Select></div>
   794→   794→          </div>
   795→   795→          <div className="grid grid-cols-2 gap-2">
   796→   796→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Категория</Label><Select value={category} onValueChange={(v) => setCategory(v as TaskCategory)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{CATEGORIES.map((c) => (<SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>))}</SelectContent></Select></div>
   797→   797→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Ответственный</Label><Input value={assignee} onChange={(e) => setAssignee(e.target.value)} placeholder="Имя..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 placeholder:text-slate-600 h-8 focus:border-cyan-500/50" /></div>
   798→   798→          </div>
   799→   799→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Дедлайн</Label><DeadlinePicker value={deadline} onChange={setDeadline} size="md" /></div>
   800→   800→        </div>
   801→   801→        <div className="flex gap-2 pt-1">
   802→   802→          <Button onClick={handleSave} disabled={!title.trim() || saving} className="flex-1 bg-cyan-600 hover:bg-cyan-700 text-white h-8 text-xs"><Save className="w-3 h-3 mr-1" />{saving ? '...' : mode === 'edit' ? 'Сохранить' : 'Создать'}</Button>
   803→   803→          {mode === 'edit' && <Button onClick={handleDelete} variant="destructive" disabled={saving} className="h-8 text-xs">Удалить</Button>}
   804→   804→        </div>
   805→   805→      </div>
   806→   806→    </div>
   807→   807→  );
   808→   808→}

   810→   810→function getProgress(children: { status: string }[]): number {
