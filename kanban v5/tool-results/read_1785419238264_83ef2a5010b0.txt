   609→   609→function getProgress(children: { status: string }[]): number {
   610→   610→  if (children.length === 0) return 0;
   611→   611→  const done = children.filter(c => c.status === 'done').length;
   612→   612→  return Math.round((done / children.length) * 100);
   613→   613→}

   615→   615→function isOverdue(deadline: string): boolean {
   616→   616→  const d = new Date(deadline);
   617→   617→  const today = new Date();
   618→   618→  today.setHours(0, 0, 0, 0);
   619→   619→  d.setHours(0, 0, 0, 0);
   620→   620→  return d < today;
   621→   621→}

   623→   623→function formatDate(deadline: string): string {
   624→   624→  const d = new Date(deadline);
   625→   625→  const day = String(d.getDate()).padStart(2, '0');
   626→   626→  const month = String(d.getMonth() + 1).padStart(2, '0');
   627→   627→  return `${day}.${month}`;
   628→   628→}
   629→   629→
   630→   630→function toLocalDateStr(isoStr: string): string {
   631→   631→  const d = new Date(isoStr);
   632→   632→  const year = d.getFullYear();
   633→   633→  const month = String(d.getMonth() + 1).padStart(2, '0');
   634→   634→  const day = String(d.getDate()).padStart(2, '0');
   635→   635→  return `${year}-${month}-${day}`;
   636→   636→}
