   525→
   526→function TaskDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   527→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   528→  const [subtasks, setSubtasks] = useState<TaskChild[]>(task.children || []);
   529→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
   530→  const [newSubtaskDeadline, setNewSubtaskDeadline] = useState<string | null>(null);
   531→  const [addingSubtask, setAddingSubtask] = useState(false);
   532→  const [expandedSubtasks, setExpandedSubtasks] = useState(true);
   533→
   534→  const progress = getProgress(subtasks);
   535→  const progressColor = progress === 100 ? 'bg-emerald-500' : progress > 50 ? 'bg-cyan-500' : progress > 0 ? 'bg-amber-500' : 'bg-slate-700';
   536→
   537→  const reloadTasks = useCallback(async () => {
   538→    if (!selectedBoardId) return;
   539→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   540→    const data = await res.json();
   541→    setBoardTasks(data.tasks);
   542→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   543→    if (updated) setSubtasks(updated.children || []);
   544→  }, [selectedBoardId, setBoardTasks, task.id]);
   545→
   546→  const cycleSubtaskStatus = async (e: React.MouseEvent, subtaskId: string) => {
   547→    e.stopPropagation();
   548→    const order: TaskStatus[] = ['todo', 'in-progress', 'review', 'done'];
   549→    const sub = subtasks.find(s => s.id === subtaskId);
   550→    if (!sub) return;
   551→    const idx = order.indexOf(sub.status as TaskStatus);
   552→    const next = order[(idx + 1) % order.length];
   553→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: subtaskId, status: next }) });
   554→    await reloadTasks();
   555→  };
   556→
   557→  const deleteSubtask = async (e: React.MouseEvent, subtaskId: string) => {
   558→    e.stopPropagation();
   559→    await fetch(`/api/tasks?id=${subtaskId}`, { method: 'DELETE' });
   560→    await reloadTasks();
   561→  };
   562→
   563→  const addSubtask = async () => {
   564→    if (!newSubtaskTitle.trim()) return;
   565→    await fetch('/api/tasks', {
   566→      method: 'POST', headers: { 'Content-Type': 'application/json' },
   567→      body: JSON.stringify({ title: newSubtaskTitle.trim(), parentId: task.id, status: 'todo', ...(newSubtaskDeadline ? { deadline: newSubtaskDeadline } : {}) }),
   568→    });
   569→    setNewSubtaskTitle(''); setNewSubtaskDeadline(null); setAddingSubtask(false);
   570→    await reloadTasks();
   571→  };
   572→
   573→  const updateTaskDeadline = async (deadline: string | null) => {
   574→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: task.id, deadline }) });
   575→    await reloadTasks();
   576→  };
   577→
   578→  const updateSubtaskDeadline = async (subtaskId: string, deadline: string | null) => {
   579→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: subtaskId, deadline }) });
   580→    await reloadTasks();
   581→  };
   582→
   583→  const handleDeleteTask = async () => {
   584→    await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   585→    setSelectedTaskId(null);
   586→    if (selectedBoardId) {
   587→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   588→      const data = await res.json();
   589→      setBoardTasks(data.tasks);
   590→    }
   591→  };
   592→
   593→  return (
   594→    <div className="flex-1 overflow-y-auto">
   595→      <div className="border-b border-slate-800/50 p-4">
   596→        <div className="flex items-center gap-1.5 mb-2.5">
   597→          {board && (
   598→            <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded font-medium"
   599→              style={{ color: board.color, backgroundColor: board.color + '15' }}>{board.title}</span>
   600→          )}
   601→          <span className={cn('text-[10px] font-medium px-1.5 py-0.5 rounded',
   602→            task.status === 'todo' ? 'text-cyan-400 bg-cyan-500/10' :
   603→            task.status === 'in-progress' ? 'text-orange-400 bg-orange-500/10' :
   604→            task.status === 'review' ? 'text-rose-400 bg-rose-500/10' :
   605→            'text-emerald-400 bg-emerald-500/10'
   606→          )}>
   607→            {STATUSES.find(s => s.value === task.status)?.label || task.status}
   608→          </span>
   609→          <div className="flex-1" />
   610→          <button onClick={() => setEditingTask(task)} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors"><Pencil className="w-3.5 h-3.5" /></button>
   611→          <button onClick={handleDeleteTask} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors"><Trash2 className="w-3.5 h-3.5" /></button>
   612→        </div>
   613→        <h3 className={cn('text-sm font-semibold leading-tight mb-1', task.status === 'done' ? 'text-slate-500 line-through' : 'text-slate-100')}>{task.title}</h3>
   614→        {task.description && <p className="text-[11px] text-slate-400 leading-relaxed mb-2">{task.description}</p>}
   615→        <div className="flex items-center gap-3 text-[10px] text-slate-500">
   616→          {task.assignee && <span>👤 {task.assignee}</span>}
   617→          <span className={cn('flex items-center gap-1',
   618→            task.priority === 'high' ? 'text-rose-400' : task.priority === 'medium' ? 'text-amber-400' : 'text-slate-500'
   619→          )}>
   620→            {task.priority === 'high' ? '🔴' : task.priority === 'medium' ? '🟡' : '⚪'}
   621→            {PRIORITIES.find(p => p.value === task.priority)?.label || task.priority}
   622→          </span>
   623→        </div>
   624→      </div>
   625→      <div className="border-b border-slate-800/30 px-4 py-3">
   626→        <DeadlinePicker value={task.deadline || null} onChange={updateTaskDeadline} isDone={task.status === 'done'} size="md" />
   627→      </div>
   628→      {subtasks.length > 0 && (
   629→        <div className="border-b border-slate-800/30 px-4 py-3">
   630→          <div className="flex items-center justify-between mb-1.5">
   631→            <span className="text-[11px] text-slate-400 font-medium">Прогресс</span>
   632→            <span className={cn('text-[11px] font-bold',
   633→              progressColor === 'bg-emerald-500' ? 'text-emerald-400' : progressColor === 'bg-cyan-500' ? 'text-cyan-400' : progressColor === 'bg-amber-500' ? 'text-amber-400' : 'text-slate-600'
   634→            )}>{progress}%</span>
   635→          </div>
   636→          <div className="relative h-2 bg-slate-800/80 rounded-full overflow-hidden">
   637→            <div className={cn('h-full rounded-full transition-all duration-700 ease-out', progressColor)} style={{ width: `${progress}%` }} />
   638→          </div>
   639→          <div className="flex items-center justify-between mt-1">
   640→            <span className="text-[9px] text-slate-600">{subtasks.filter(s => s.status === 'done').length} из {subtasks.length} подзадач</span>
   641→            {progress === 100 && <span className="text-[9px] text-emerald-400 font-medium">✓ Завершено</span>}
   642→          </div>
   643→        </div>
   644→      )}
   645→      <div className="px-4 py-3">
   646→        <button onClick={() => setExpandedSubtasks(!expandedSubtasks)} className="flex items-center gap-1.5 mb-2 group w-full text-left">
   647→          {expandedSubtasks
   648→            ? <ChevronUp className="w-3.5 h-3.5 text-slate-600 group-hover:text-slate-400 transition-colors" />
   649→            : <ChevronDown className="w-3.5 h-3.5 text-slate-600 group-hover:text-slate-400 transition-colors" />}
   650→          <span className="text-[11px] font-semibold text-slate-400 uppercase tracking-wider">Подзадачи</span>
   651→          {subtasks.length > 0 && <span className="text-[9px] text-slate-600 bg-slate-800/50 px-1.5 py-0.5 rounded ml-1">{subtasks.length}</span>}
   652→        </button>
   653→        {expandedSubtasks && (
   654→          <div className="space-y-1.5">
   655→            {subtasks.length === 0 && !addingSubtask && <p className="text-[10px] text-slate-600 py-2 text-center">Нет подзадач</p>}
   656→            {subtasks.map((sub) => {
   657→              const Icon = SUBTASK_STATUS_ICON[sub.status] || Circle;
   658→              return (
   659→                <div key={sub.id} className="group/sub flex items-center gap-2 px-2 py-1.5 rounded-lg hover:bg-slate-900/80 transition-colors">
   660→                  <button onClick={(e) => cycleSubtaskStatus(e, sub.id)} className={cn('flex-shrink-0 transition-opacity hover:opacity-70', SUBTASK_STATUS_COLOR[sub.status] || 'text-slate-500')}><Icon className="w-3.5 h-3.5" /></button>
   661→                  <span className={cn('flex-1 text-[11px] leading-tight min-w-0', sub.status === 'done' ? 'text-slate-500 line-through' : 'text-slate-300')}>{sub.title}</span>
   662→                  <div className="relative flex-shrink-0" onClick={(e) => e.stopPropagation()}>
   663→                    <DeadlinePicker value={sub.deadline || null} onChange={(d) => updateSubtaskDeadline(sub.id, d)} isDone={sub.status === 'done'} size="sm" inline />
   664→                  </div>
   665→                  <button onClick={(e) => deleteSubtask(e, sub.id)} className="p-0.5 rounded opacity-0 group-hover/sub:opacity-100 hover:bg-slate-800 text-slate-600 hover:text-rose-400 transition-all flex-shrink-0"><X className="w-3 h-3" /></button>
   666→                </div>
   667→              );
   668→            })}
   669→            {addingSubtask ? (
   670→              <div className="space-y-2 bg-slate-900/60 border border-slate-800/50 rounded-lg p-2.5 mt-1">
   671→                <Input value={newSubtaskTitle} onChange={(e) => setNewSubtaskTitle(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && addSubtask()} placeholder="Название подзадачи..." autoFocus className="bg-slate-800/80 border-slate-700/50 text-[11px] text-slate-200 placeholder:text-slate-600 h-7 focus:border-cyan-500/50" />
   672→                <div className="flex items-center gap-2">
   673→                  <span className="text-[10px] text-slate-600 flex-shrink-0">Дедлайн:</span>
   674→                  <DeadlinePicker value={newSubtaskDeadline} onChange={setNewSubtaskDeadline} size="sm" />
   675→                </div>
   676→                <div className="flex gap-1.5 pt-0.5">
   677→                  <button onClick={addSubtask} disabled={!newSubtaskTitle.trim()} className="flex-1 text-[10px] bg-cyan-600 hover:bg-cyan-700 disabled:opacity-40 disabled:hover:bg-cyan-600 text-white rounded px-2 py-1.5 transition-colors font-medium">Добавить</button>
   678→                  <button onClick={() => { setAddingSubtask(false); setNewSubtaskTitle(''); setNewSubtaskDeadline(null); }} className="text-[10px] text-slate-500 hover:text-slate-300 px-2 py-1.5 transition-colors">Отмена</button>
   679→                </div>
   680→              </div>
   681→            ) : (
   682→              <button onClick={() => setAddingSubtask(true)} className="flex items-center gap-1.5 px-2 py-1 text-[10px] text-slate-600 hover:text-cyan-400 transition-colors rounded-lg hover:bg-slate-900/50 w-full"><Plus className="w-3 h-3" /> Добавить подзадачу</button>
   683→            )}
   684→          </div>
   685→        )}
   686→      </div>
   687→    </div>
   688→  );
   689→}

   693→function TaskForm({ mode, task }: { mode: 'create' | 'edit'; task?: Task }) {
   694→  const { selectedBoardId, setBoardTasks, selectedProjectId, setEditingTask, setIsCreating, setSelectedTaskId } = useKanbanStore();
   695→  const [title, setTitle] = useState('');
   696→  const [description, setDescription] = useState('');
   697→  const [status, setStatus] = useState<TaskStatus>('todo');
   698→  const [priority, setPriority] = useState<TaskPriority>('medium');
   699→  const [assignee, setAssignee] = useState('');
   700→  const [category, setCategory] = useState<TaskCategory>('general');
   701→  const [deadline, setDeadline] = useState<string | null>(null);
   702→  const [saving, setSaving] = useState(false);
   703→
   704→  useEffect(() => {
   705→    if (mode === 'edit' && task) {
   706→      setTitle(task.title); setDescription(task.description || ''); setStatus(task.status);
   707→      setPriority(task.priority); setAssignee(task.assignee || ''); setCategory(task.category);
   708→      setDeadline(task.deadline || null);
   709→    }
   710→  }, [mode, task]);
   711→
   712→  const reloadTasks = useCallback(async () => {
   713→    if (selectedBoardId) {
   714→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   715→      const data = await res.json();
   716→      setBoardTasks(data.tasks);
   717→    }
   718→    if (selectedProjectId) {
   719→      const bRes = await fetch(`/api/boards?projectId=${selectedProjectId}`);
   720→      const bData = await bRes.json();
   721→      useKanbanStore.getState().setBoards(bData.boards);
   722→    }
   723→  }, [selectedBoardId, selectedProjectId, setBoardTasks]);
   724→
   725→  const handleSave = async () => {
   726→    if (!title.trim()) return;
   727→    setSaving(true);
   728→    try {
   729→      if (mode === 'edit' && task) {
   730→        await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: task.id, title, description: description || null, status, priority, assignee: assignee || null, category, boardId: selectedBoardId, deadline }) });
   731→      } else {
   732→        await fetch('/api/tasks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title, description: description || null, status, priority, assignee: assignee || null, category, parentId: null, boardId: selectedBoardId, deadline }) });
   733→      }
   734→      await reloadTasks(); setEditingTask(null); setIsCreating(false);
   735→      if (mode === 'edit' && task) setSelectedTaskId(task.id);
   736→    } finally { setSaving(false); }
   737→  };
   738→
   739→  const handleDelete = async () => {
   740→    if (!task) return;
   741→    setSaving(true);
   742→    try {
   743→      await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   744→      await reloadTasks(); setEditingTask(null); setSelectedTaskId(null);
   745→    } finally { setSaving(false); }
   746→  };
   747→
   748→  return (
   749→    <div className="flex-1 overflow-y-auto">
   750→      <div className="p-4 space-y-3">
   751→        <div className="flex items-center justify-between">
   752→          <h3 className="text-xs font-semibold text-slate-300 uppercase tracking-wider">{mode === 'edit' ? 'Редактировать' : 'Новая задача'}</h3>
   753→          <button onClick={() => { setEditingTask(null); setIsCreating(false); }} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300"><X className="h-3.5 w-3.5" /></button>
   754→        </div>
   755→        <div className="space-y-2.5">
   756→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Название</Label><Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Задача..." className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-8 focus:border-cyan-500/50" /></div>
   757→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Описание</Label><Textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Описание..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none focus:border-cyan-500/50" /></div>
   758→          <div className="grid grid-cols-2 gap-2">
   759→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Статус</Label><Select value={status} onValueChange={(v) => setStatus(v as TaskStatus)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{STATUSES.map((s) => (<SelectItem key={s.value} value={s.value}><span className={s.color}>{s.label}</span></SelectItem>))}</SelectContent></Select></div>
   760→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Приоритет</Label><Select value={priority} onValueChange={(v) => setPriority(v as TaskPriority)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{PRIORITIES.map((p) => (<SelectItem key={p.value} value={p.value}><div className="flex items-center gap-1.5"><div className={cn('w-1.5 h-1.5 rounded-full', p.dot)} />{p.label}</div></SelectItem>))}</SelectContent></Select></div>
   761→          </div>
   762→          <div className="grid grid-cols-2 gap-2">
   763→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Категория</Label><Select value={category} onValueChange={(v) => setCategory(v as TaskCategory)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{CATEGORIES.map((c) => (<SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>))}</SelectContent></Select></div>
   764→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Ответственный</Label><Input value={assignee} onChange={(e) => setAssignee(e.target.value)} placeholder="Имя..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 placeholder:text-slate-600 h-8 focus:border-cyan-500/50" /></div>
   765→          </div>
   766→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Дедлайн</Label><DeadlinePicker value={deadline} onChange={setDeadline} size="md" /></div>
   767→        </div>
   768→        <div className="flex gap-2 pt-1">
   769→          <Button onClick={handleSave} disabled={!title.trim() || saving} className="flex-1 bg-cyan-600 hover:bg-cyan-700 text-white h-8 text-xs"><Save className="w-3 h-3 mr-1" />{saving ? '...' : mode === 'edit' ? 'Сохранить' : 'Создать'}</Button>
   770→          {mode === 'edit' && <Button onClick={handleDelete} variant="destructive" disabled={saving} className="h-8 text-xs">Удалить</Button>}
   771→        </div>
   772→      </div>
   773→    </div>
   774→  );
   775→}
   776→
   777→function getProgress(children: { status: string }[]): number {
   778→  if (children.length === 0) return 0;
   779→  return Math.round((children.filter(c => c.status === 'done').length / children.length) * 100);
   780→}
