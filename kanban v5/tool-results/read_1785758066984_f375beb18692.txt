     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { useKanbanStore } from '@/store/kanban-store';
     5→import { Button } from '@/components/ui/button';
     6→import { Input } from '@/components/ui/input';
     7→import { Textarea } from '@/components/ui/textarea';
     8→import {
     9→  ChevronRight, ChevronLeft, Check, Music, Guitar, Mic2,
    10→  Sparkles, X, Loader2, Plus, Trash2, FileText,
    11→} from 'lucide-react';
    12→import { cn } from '@/lib/utils';
    13→
    14→interface WizardData {
    15→  instruments: string[];
    16→  stages: { key: string; label: string; emoji: string; hasInstrumentBased: boolean; defaultSubtaskCount: number }[];
    17→}
    18→
    19→interface EditableSubtask {
    20→  id: string;
    21→  title: string;
    22→  description: string;
    23→}
    24→
    25→interface EditableStage {
    26→  id: string;
    27→  emoji: string;
    28→  label: string;
    29→  description: string;
    30→  subtasks: EditableSubtask[];
    31→  sourceKey?: string; // template key if added from template, undefined if custom
    32→}
    33→
    34→const CUSTOM_ICONS: Record<string, typeof Music> = {
    35→  'Вокал (основной)': Mic2,
    36→  'Бэк-вокал': Mic2,
    37→  'Гитара (акустическая)': Guitar,
    38→  'Гитара (электро)': Guitar,
    39→};
    40→
    41→const DEFAULT_STAGE_EMOJIS = ['📁', '🎯', '⚙️', '🔧', '🎨', '📊', '💡', '📋', '🧩', '🚀'];
    42→
    43→const STAGE_SUBTASKS: Record<string, string[]> = {
    44→  'Сонграйтинг': ['Написание текста', 'Демо'],
    45→  'Аранжировка': ['Гармоническая структура', 'Тембровая расстановка'],
    46→  'Редактура': ['Компинг', 'Тюнинг вокала', 'Тайминг'],
    47→  'Сведение': ['Баланс и эквализация', 'Пространственная обработка', 'Эффекты и автоматизация'],
    48→  'Мастеринг': ['Финальная громкость и лимитирование', 'Стерео-обработка'],
    49→};
    50→
    51→let _uid = 0;
    52→const uid = () => `es_${++_uid}_${Date.now()}`;
    53→
    54→export default function TrackWizard() {
    55→  const { selectedBoardId, setBoardTasks, setIsTrackWizardOpen, setSelectedTaskId, setTrackWizardStep } = useKanbanStore();
    56→  const [step, setStep] = useState(0); // 0=name, 1=instruments, 2=stages+configure
    57→
    58→  // Sync wizard step to store for onboarding
    59→  useEffect(() => { setTrackWizardStep(step); return () => setTrackWizardStep(-1); }, [step, setTrackWizardStep]);
    60→
    61→  const [trackName, setTrackName] = useState('');
    62→  const [trackDescription, setTrackDescription] = useState('');
    63→  const [selectedInstruments, setSelectedInstruments] = useState<string[]>([]);
    64→  const [customInstrument, setCustomInstrument] = useState('');
    65→  const [wizardData, setWizardData] = useState<WizardData | null>(null);
    66→  const [loading, setLoading] = useState(false);
    67→  const [allInstruments, setAllInstruments] = useState<string[]>([]);
    68→
    69→  // Editable stages (shared state for merged step 2)
    70→  const [editableStages, setEditableStages] = useState<EditableStage[]>([]);
    71→  const [expandedStageId, setExpandedStageId] = useState<string | null>(null);
    72→  const [addingSubtaskForStage, setAddingSubtaskForStage] = useState<string | null>(null);
    73→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
    74→  const [newSubtaskDesc, setNewSubtaskDesc] = useState('');
    75→  const [newStageLabel, setNewStageLabel] = useState('');
    76→  const [newStageDesc, setNewStageDesc] = useState('');
    77→  const [showNewStageForm, setShowNewStageForm] = useState(false);
    78→  const [showTemplates, setShowTemplates] = useState(true);
    79→
    80→  useEffect(() => {
    81→    fetch('/api/tracks/wizard')
    82→      .then(r => r.json())
    83→      .then(data => {
    84→        setWizardData(data);
    85→        setAllInstruments([...data.instruments]);
    86→      });
    87→  }, []);
    88→
    89→  // --- Template helpers ---
    90→  const isTemplateAdded = (key: string) => editableStages.some(s => s.sourceKey === key);
    91→
    92→  const toggleTemplate = (key: string) => {
    93→    if (isTemplateAdded(key)) {
    94→      // Remove template stage
    95→      setEditableStages(prev => prev.filter(s => s.sourceKey !== key));
    96→    } else {
    97→      // Add template stage with default subtasks
    98→      const tpl = wizardData?.stages.find(s => s.key === key);
    99→      if (!tpl) return;
   100→      const subtaskTitles = tpl.hasInstrumentBased && selectedInstruments.length > 0
   101→        ? selectedInstruments.map(i => `Запись: ${i}`)
   102→        : STAGE_SUBTASKS[key] || [];
   103→      const newStage: EditableStage = {
   104→        id: uid(),
   105→        emoji: tpl.emoji,
   106→        label: tpl.label,
   107→        description: '',
   108→        subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
   109→        sourceKey: key,
   110→      };
   111→      setEditableStages(prev => [...prev, newStage]);
   112→      setExpandedStageId(newStage.id);
   113→    }
   114→  };
   115→
   116→  const addAllTemplates = () => {
   117→    if (!wizardData) return;
   118→    const allAdded = wizardData.stages.every(s => isTemplateAdded(s.key));
   119→    if (allAdded) {
   120→      // Remove all template stages
   121→      setEditableStages(prev => prev.filter(s => !s.sourceKey));
   122→    } else {
   123→      // Add all missing templates
   124→      const newStages: EditableStage[] = [];
   125→      for (const tpl of wizardData.stages) {
   126→        if (!isTemplateAdded(tpl.key)) {
   127→          const subtaskTitles = tpl.hasInstrumentBased && selectedInstruments.length > 0
   128→            ? selectedInstruments.map(i => `Запись: ${i}`)
   129→            : STAGE_SUBTASKS[tpl.key] || [];
   130→          newStages.push({
   131→            id: uid(),
   132→            emoji: tpl.emoji,
   133→            label: tpl.label,
   134→            description: '',
   135→            subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
   136→            sourceKey: tpl.key,
   137→          });
   138→        }
   139→      }
   140→      setEditableStages(prev => [...prev, ...newStages]);
   141→    }
   142→  };
   143→
   144→  // Re-generate instrument-based subtasks when instruments change
   145→  const updateInstrumentSubtasks = useCallback(() => {
   146→    if (!wizardData) return;
   147→    const recordStage = wizardData.stages.find(s => s.key === 'Запись');
   148→    if (!recordStage || !isTemplateAdded('Запись')) return;
   149→    const instSubtasks = selectedInstruments.map(i => `Запись: ${i}`);
   150→    setEditableStages(prev => prev.map(s => {
   151→      if (s.sourceKey !== 'Запись') return s;
   152→      // Keep existing subtasks that still match an instrument, add new ones
   153→      const existing = s.subtasks.filter(st => instSubtasks.includes(st.title));
   154→      const finalSubs: EditableSubtask[] = instSubtasks.map(title => {
   155→        const ex = existing.find(e => e.title === title);
   156→        return ex || { id: uid(), title, description: '' };
   157→      });
   158→      return { ...s, subtasks: finalSubs };
   159→    }));
   160→  }, [wizardData, selectedInstruments]);
   161→
   162→  // When entering step 2, ensure instrument subtasks are synced
   163→  const goToStep2 = () => {
   164→    updateInstrumentSubtasks();
   165→    setStep(2);
   166→  };
   167→
   168→  // --- Instrument helpers ---
   169→  const toggleInstrument = (inst: string) => {
   170→    setSelectedInstruments(prev =>
   171→      prev.includes(inst) ? prev.filter(i => i !== inst) : [...prev, inst]
   172→    );
   173→  };
   174→
   175→  const addCustomInstrument = () => {
   176→    const trimmed = customInstrument.trim();
   177→    if (!trimmed || allInstruments.includes(trimmed)) return;
   178→    setAllInstruments(prev => [...prev, trimmed]);
   179→    setSelectedInstruments(prev => [...prev, trimmed]);
   180→    setCustomInstrument('');
   181→  };
   182→
   183→  // --- Stage editing helpers ---
   184→  const updateStage = (stageId: string, updates: Partial<Pick<EditableStage, 'label' | 'description' | 'emoji'>>) => {
   185→    setEditableStages(prev => prev.map(s => s.id === stageId ? { ...s, ...updates } : s));
   186→  };
   187→
   188→  const removeStage = (stageId: string) => {
   189→    setEditableStages(prev => prev.filter(s => s.id !== stageId));
   190→  };
   191→
   192→  const addNewStage = () => {
   193→    if (!newStageLabel.trim()) return;
   194→    const newStage: EditableStage = {
   195→      id: uid(),
   196→      emoji: DEFAULT_STAGE_EMOJIS[editableStages.length % DEFAULT_STAGE_EMOJIS.length],
   197→      label: newStageLabel.trim(),
   198→      description: newStageDesc.trim(),
   199→      subtasks: [],
   200→    };
   201→    setEditableStages(prev => [...prev, newStage]);
   202→    setNewStageLabel('');
   203→    setNewStageDesc('');
   204→    setShowNewStageForm(false);
   205→    setExpandedStageId(newStage.id);
   206→  };
   207→
   208→  const updateSubtask = (stageId: string, subtaskId: string, updates: Partial<Pick<EditableSubtask, 'title' | 'description'>>) => {
   209→    setEditableStages(prev => prev.map(s => {
   210→      if (s.id !== stageId) return s;
   211→      return { ...s, subtasks: s.subtasks.map(st => st.id === subtaskId ? { ...st, ...updates } : st) };
   212→    }));
   213→  };
   214→
   215→  const removeSubtask = (stageId: string, subtaskId: string) => {
   216→    setEditableStages(prev => prev.map(s => {
   217→      if (s.id !== stageId) return s;
   218→      return { ...s, subtasks: s.subtasks.filter(st => st.id !== subtaskId) };
   219→    }));
   220→  };
   221→
   222→  const addSubtask = (stageId: string) => {
   223→    if (!newSubtaskTitle.trim()) return;
   224→    setEditableStages(prev => prev.map(s => {
   225→      if (s.id !== stageId) return s;
   226→      return { ...s, subtasks: [...s.subtasks, { id: uid(), title: newSubtaskTitle.trim(), description: newSubtaskDesc.trim() }] };
   227→    }));
   228→    setNewSubtaskTitle('');
   229→    setNewSubtaskDesc('');
   230→    setAddingSubtaskForStage(null);
   231→  };
   232→
   233→  // --- Create ---
   234→  const handleCreate = useCallback(async () => {
   235→    if (!trackName.trim() || !selectedBoardId) return;
   236→    setLoading(true);
   237→    try {
   238→      const res = await fetch('/api/tracks/wizard', {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify({
   242→          title: trackName.trim(),
   243→          description: trackDescription.trim(),
   244→          instruments: selectedInstruments,
   245→          boardId: selectedBoardId,
   246→          customStages: editableStages.map(s => ({
   247→            emoji: s.emoji,
   248→            label: s.label,
   249→            description: s.description,
   250→            subtasks: s.subtasks.map(st => ({ title: st.title, description: st.description })),
   251→          })),
   252→        }),
   253→      });
   254→      const data = await res.json();
   255→      const taskRes = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   256→      const taskData = await taskRes.json();
   257→      setBoardTasks(taskData.tasks);
   258→      if (data.trackId) setSelectedTaskId(data.trackId);
   259→      setIsTrackWizardOpen(false);
   260→    } finally {
   261→      setLoading(false);
   262→    }
   263→  }, [trackName, trackDescription, selectedInstruments, selectedBoardId, editableStages, setBoardTasks, setSelectedTaskId, setIsTrackWizardOpen]);
   264→
   265→  const canProceed = () => {
   266→    if (step === 0) return trackName.trim().length > 0;
   267→    if (step === 1) return selectedInstruments.length > 0;
   268→    if (step === 2) return editableStages.length > 0;
   269→    return true;
   270→  };
   271→
   272→  const handleClose = () => setIsTrackWizardOpen(false);
   273→
   274→  const STEPS = ['Название', 'Инструменты', 'Этапы'];
   275→
   276→  return (
   277→    <div className="flex flex-col h-full">
   278→      {/* Header */}
   279→      <div className="border-b border-slate-800/50 p-4 shrink-0">
   280→        <div className="flex items-center justify-between mb-3">
   281→          <div className="flex items-center gap-2">
   282→            <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-purple-500 to-pink-500 flex items-center justify-center">
   283→              <Music className="w-4 h-4 text-white" />
   284→            </div>
   285→            <div>
   286→              <h3 className="text-xs font-bold text-white uppercase tracking-wider">Конструктор трека</h3>
   287→              <p className="text-[10px] text-slate-500">Мастер создания пайплайна</p>
   288→            </div>
   289→          </div>
   290→          <button onClick={handleClose} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors">
   291→            <X className="w-4 h-4" />
   292→          </button>
   293→        </div>
   294→
   295→        {/* Progress steps */}
   296→        <div className="flex items-center gap-1">
   297→          {STEPS.map((s, i) => (
   298→            <div key={s} className="flex items-center flex-1">
   299→              <button
   300→                onClick={() => i < step && setStep(i)}
   301→                className={cn(
   302→                  'flex items-center gap-1.5 text-[10px] font-medium px-2 py-1 rounded-md transition-all duration-200',
   303→                  i === step && 'bg-cyan-500/15 text-cyan-400',
   304→                  i < step && 'text-slate-400 hover:bg-slate-800/60 cursor-pointer',
   305→                  i > step && 'text-slate-600',
   306→                )}
   307→              >
   308→                {i < step ? <Check className="w-3 h-3" /> : <span className="w-4 h-4 rounded-full border text-center text-[9px] flex items-center justify-center" style={{
   309→                  borderColor: i === step ? '#00d9ff' : i < step ? '#00d9ff' : '#334155',
   310→                }}>{i + 1}</span>}
   311→                <span className="hidden lg:inline">{s}</span>
   312→              </button>
   313→              {i < STEPS.length - 1 && (
   314→                <div className="flex-1 h-px mx-1" style={{ backgroundColor: i < step ? '#00d9ff30' : '#1e293b' }} />
   315→              )}
   316→            </div>
   317→          ))}
   318→        </div>
   319→      </div>
   320→
   321→      {/* Content */}
   322→      <div className="flex-1 overflow-y-auto p-4">
   323→        {/* Step 0: Name */}
   324→        {step === 0 && (
   325→          <div className="space-y-3">
   326→            <div className="space-y-1.5">
   327→              <label className="text-[11px] text-slate-400 font-medium">Название трека</label>
   328→              <Input
   329→                value={trackName}
   330→                onChange={(e) => setTrackName(e.target.value)}
   331→                onKeyDown={(e) => e.key === 'Enter' && canProceed() && setStep(1)}
   332→                placeholder="Лирика, Эксперимент..."
   333→                autoFocus
   334→                className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-10 focus:border-purple-500/50"
   335→              />
   336→            </div>
   337→            <div className="space-y-1.5">
   338→              <label className="text-[11px] text-slate-400 font-medium">Описание</label>
   339→              <Textarea
   340→                value={trackDescription}
   341→                onChange={(e) => setTrackDescription(e.target.value)}
   342→                placeholder="Краткое описание идеи трека, настроение, жанр..."
   343→                rows={3}
   344→                className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 focus:border-purple-500/50 resize-none"
   345→              />
   346→            </div>
   347→            <div className="rounded-lg bg-slate-900/40 border border-slate-800/30 p-3">
   348→              <div className="flex items-center gap-2 mb-2">
   349→                <Sparkles className="w-3.5 h-3.5 text-purple-400" />
   350→                <span className="text-[11px] text-purple-400 font-medium">Подсказка</span>
   351→              </div>
   352→              <p className="text-[10px] text-slate-500 leading-relaxed">
   353→                После ввода названия и описания вы выберете инструменты и этапы производства.
   354→                Система автоматически создаст структуру задач для вашего трека.
   355→              </p>
   356→            </div>
   357→          </div>
   358→        )}
   359→
   360→        {/* Step 1: Instruments */}
   361→        {step === 1 && (
   362→          <div className="space-y-3">
   363→            <div className="flex items-center justify-between">
   364→              <span className="text-[11px] text-slate-400 font-medium">
   365→                Инструменты <span className="text-slate-600">({selectedInstruments.length} выбрано)</span>
   366→              </span>
   367→            </div>
   368→            <div className="flex flex-wrap gap-1.5">
   369→              {allInstruments.map((inst) => {
   370→                const IconComp = CUSTOM_ICONS[inst];
   371→                const isSelected = selectedInstruments.includes(inst);
   372→                return (
   373→                  <button
   374→                    key={inst}
   375→                    onClick={() => toggleInstrument(inst)}
   376→                    className={cn(
   377→                      'flex items-center gap-1.5 text-[11px] px-2.5 py-1.5 rounded-lg border transition-all duration-150',
   378→                      isSelected
   379→                        ? 'bg-cyan-500/15 border-cyan-500/40 text-cyan-300 shadow-sm shadow-cyan-500/10'
   380→                        : 'bg-slate-900/60 border-slate-800/50 text-slate-400 hover:border-slate-700 hover:text-slate-300',
   381→                    )}
   382→                  >
   383→                    {IconComp ? <IconComp className="w-3 h-3" /> : <Music className="w-3 h-3" />}
   384→                    {inst}
   385→                    {isSelected && <Check className="w-3 h-3 ml-0.5" />}
   386→                  </button>
   387→                );
   388→              })}
   389→            </div>
   390→            <div className="flex gap-2 pt-1">
   391→              <Input
   392→                value={customInstrument}
   393→                onChange={(e) => setCustomInstrument(e.target.value)}
   394→                onKeyDown={(e) => e.key === 'Enter' && addCustomInstrument()}
   395→                placeholder="Свой инструмент..."
   396→                className="bg-slate-900/80 border-slate-700/50 text-[11px] text-slate-200 placeholder:text-slate-600 h-7 focus:border-purple-500/50"
   397→              />
   398→              <Button size="sm" onClick={addCustomInstrument} disabled={!customInstrument.trim()} variant="ghost" className="h-7 text-[10px] text-slate-400 hover:text-purple-400 gap-1">
   399→                <Plus className="w-3 h-3" />
   400→              </Button>
   401→            </div>
   402→            <div className="rounded-lg bg-slate-900/40 border border-slate-800/30 p-3">
   403→              <div className="flex items-center gap-2 mb-1">
   404→                <Sparkles className="w-3.5 h-3.5 text-amber-400" />
   405→                <span className="text-[11px] text-amber-400 font-medium">Подсказка</span>
   406→              </div>
   407→              <p className="text-[10px] text-slate-500 leading-relaxed">
   408→                Выбранные инструменты определят подзадачи на этапе &laquo;Запись&raquo;.
   409→                Для каждого инструмента создастся отдельная задача.
   410→              </p>
   411→            </div>
   412→          </div>
   413→        )}
   414→
   415→        {/* Step 2: Stages + Configure (merged) */}
   416→        {step === 2 && (
   417→          <div className="space-y-3">
   418→            {/* Templates section */}
   419→            <div className="rounded-lg border border-slate-800/40 bg-slate-900/30 overflow-hidden">
   420→              <button
   421→                onClick={() => setShowTemplates(v => !v)}
   422→                className="w-full flex items-center justify-between px-3 py-2 hover:bg-slate-800/30 transition-colors"
   423→              >
   424→                <div className="flex items-center gap-2">
   425→                  <Sparkles className="w-3.5 h-3.5 text-amber-400" />
   426→                  <span className="text-[11px] text-slate-300 font-medium">Шаблоны этапов</span>
   427→                  <span className="text-[9px] text-slate-600">
   428→                    {wizardData?.stages.filter(s => isTemplateAdded(s.key)).length || 0}/{wizardData?.stages.length || 0}
   429→                  </span>
   430→                </div>
   431→                <ChevronRight className={cn('w-3.5 h-3.5 text-slate-500 transition-transform', showTemplates && 'rotate-90')} />
   432→              </button>
   433→
   434→              {showTemplates && (
   435→                <div className="px-3 pb-3 space-y-2.5">
   436→                  <div className="flex flex-wrap gap-1.5">
   437→                    {wizardData?.stages.map(tpl => {
   438→                      const added = isTemplateAdded(tpl.key);
   439→                      return (
   440→                        <button
   441→                          key={tpl.key}
   442→                          onClick={() => toggleTemplate(tpl.key)}
   443→                          className={cn(
   444→                            'flex items-center gap-1.5 text-[10px] px-2 py-1.5 rounded-lg border transition-all duration-150',
   445→                            added
   446→                              ? 'bg-cyan-500/15 border-cyan-500/30 text-cyan-300'
   447→                              : 'bg-slate-900/60 border-slate-800/50 text-slate-500 hover:border-slate-700 hover:text-slate-300',
   448→                          )}
   449→                        >
   450→                          <span className="text-sm">{tpl.emoji}</span>
   451→                          {tpl.label}
   452→                          {tpl.hasInstrumentBased && selectedInstruments.length > 0 && (
   453→                            <span className="text-[8px] text-purple-400/60 bg-purple-500/10 px-1 py-0.5 rounded">{selectedInstruments.length}</span>
   454→                          )}
   455→                          {added && <Check className="w-3 h-3" />}
   456→                        </button>
   457→                      );
   458→                    })}
   459→                  </div>
   460→                  <div className="flex items-center gap-2 pt-1">
   461→                    <button
   462→                      onClick={addAllTemplates}
   463→                      className="text-[9px] text-cyan-500/70 hover:text-cyan-400 transition-colors"
