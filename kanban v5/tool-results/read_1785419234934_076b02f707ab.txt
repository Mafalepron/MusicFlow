   395→function TaskForm({ mode, task }: { mode: 'create' | 'edit'; task?: Task }) {
   396→  const { selectedBoardId, setBoardTasks, selectedProjectId, setEditingTask, setIsCreating, setSelectedTaskId } = useKanbanStore();
   397→
   398→  const [title, setTitle] = useState('');
   399→  const [description, setDescription] = useState('');
   400→  const [status, setStatus] = useState<TaskStatus>('todo');
   401→  const [priority, setPriority] = useState<TaskPriority>('medium');
   402→  const [assignee, setAssignee] = useState('');
   403→  const [category, setCategory] = useState<TaskCategory>('general');
   404→  const [deadline, setDeadline] = useState('');
   405→  const [saving, setSaving] = useState(false);
   406→
   407→  useEffect(() => {
   408→    if (mode === 'edit' && task) {
   409→      setTitle(task.title);
   410→      setDescription(task.description || '');
   411→      setStatus(task.status);
   412→      setPriority(task.priority);
   413→      setAssignee(task.assignee || '');
   414→      setCategory(task.category);
   415→      setDeadline(task.deadline ? toLocalDateStr(task.deadline) : '');
   416→    }
   417→  }, [mode, task]);
   418→
   419→  const reloadTasks = useCallback(async () => {
   420→    if (selectedBoardId) {
   421→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   422→      const data = await res.json();
   423→      setBoardTasks(data.tasks);
   424→    }
   425→    if (selectedProjectId) {
   426→      const bRes = await fetch(`/api/boards?projectId=${selectedProjectId}`);
   427→      const bData = await bRes.json();
   428→      useKanbanStore.getState().setBoards(bData.boards);
   429→    }
   430→  }, [selectedBoardId, selectedProjectId, setBoardTasks]);
   431→
   432→  const handleSave = async () => {
   433→    if (!title.trim()) return;
   434→    setSaving(true);
   435→    try {
   436→      if (mode === 'edit' && task) {
   437→        await fetch('/api/tasks', {
   438→          method: 'PUT',
   439→          headers: { 'Content-Type': 'application/json' },
   440→          body: JSON.stringify({
   441→            id: task.id, title, description: description || null,
   442→            status, priority, assignee: assignee || null, category,
   443→            boardId: selectedBoardId,
   444→            deadline: deadline || null,
   445→          }),
   446→        });
   447→      } else {
   448→        await fetch('/api/tasks', {
   449→          method: 'POST',
   450→          headers: { 'Content-Type': 'application/json' },
   451→          body: JSON.stringify({
   452→            title, description: description || null, status, priority,
   453→            assignee: assignee || null, category,
   454→            parentId: selectedProjectId, boardId: selectedBoardId,
   455→            deadline: deadline || null,
   456→          }),
   457→        });
   458→      }
   459→      await reloadTasks();
   460→      setEditingTask(null);
   461→      setIsCreating(false);
   462→      if (mode === 'edit' && task) setSelectedTaskId(task.id);
   463→    } finally {
   464→      setSaving(false);
   465→    }
   466→  };
   467→
   468→  const handleDelete = async () => {
   469→    if (!task) return;
   470→    setSaving(true);
   471→    try {
   472→      await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   473→      await reloadTasks();
   474→      setEditingTask(null);
   475→      setSelectedTaskId(null);
   476→    } finally {
   477→      setSaving(false);
   478→    }
   479→  };
   480→
   481→  return (
   482→    <div className="flex-1 overflow-y-auto">
   483→      <div className="p-4 space-y-3">
   484→        <div className="flex items-center justify-between">
   485→          <h3 className="text-xs font-semibold text-slate-300 uppercase tracking-wider">
   486→            {mode === 'edit' ? 'Редактировать' : 'Новая задача'}
   487→          </h3>
   488→          <button
   489→            onClick={() => { setEditingTask(null); setIsCreating(false); }}

   609→function getProgress(children: { status: string }[]): number {
   610→  if (children.length === 0) return 0;
   611→  const done = children.filter(c => c.status === 'done').length;
   612→  return Math.round((done / children.length) * 100);
   613→}

   615→function isOverdue(deadline: string): boolean {
   616→  const d = new Date(deadline);
   617→  const today = new Date();
   618→  today.setHours(0, 0, 0, 0);
   619→  d.setHours(0, 0, 0, 0);
   620→  return d < today;
   621→}

   623→function formatDate(deadline: string): string {
   624→  const d = new Date(deadline);
   625→  const day = String(d.getDate()).padStart(2, '0');
   626→  const month = String(d.getMonth() + 1).padStart(2, '0');
   627→  return `${day}.${month}`;
   628→}

   630→function toLocalDateStr(isoStr: string): string {
   631→  const d = new Date(isoStr);
   632→  const year = d.getFullYear();
   633→  const month = String(d.getMonth() + 1).padStart(2, '0');
   634→  const day = String(d.getDate()).padStart(2, '0');
   635→  return `${year}-${month}-${day}`;
   636→}
