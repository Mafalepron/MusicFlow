   385→                >
   386→                  Отмена
   387→                </button>

   389→                  onClick={() => void saveDesc()}

   391→                  style={{ color: '#000', backgroundColor: '#FCEE0A', boxShadow: '0 0 8px rgba(252,238,10,0.3)' }}
   392→                >
   393→                  Сохранить
   394→                </button>
   395→              </div>
   396→            </div>
   397→          </div>
   398→        ) : task.description ? (
   399→          <p
   400→            className="text-[11px] text-slate-400 leading-relaxed whitespace-pre-wrap cursor-pointer hover:text-slate-300 transition-colors"
   401→            onClick={() => { setDescDraft(task.description || ''); setIsEditingDesc(true); }}
   402→          >
   403→            {task.description}
   404→          </p>
   405→        ) : (
   406→          <button
   407→            onClick={() => { setDescDraft(''); setIsEditingDesc(true); }}
   408→            className="text-[11px] text-slate-600 hover:text-slate-400 transition-colors"
   409→          >
   410→            Нажмите, чтобы добавить описание...

   432→/* ── Track Text / Lyrics Section ──────────────────────── */
   433→
   434→function TrackTextSection({
   435→  task,
   436→  boardColor,
   437→  onUpdate,
   438→}: {
   439→  task: Task;
   440→  boardColor: string;
   441→  onUpdate: (patch: Record<string, unknown>) => Promise<void>;
   442→}) {
   443→  const config = task.trackConfig ? JSON.parse(task.trackConfig) : null;
   444→  const trackText: string = config?.text || '';
   445→  const [isEditing, setIsEditing] = useState(false);
   446→  const [isExpanded, setIsExpanded] = useState(false);
   447→  const [draft, setDraft] = useState(trackText);
   448→
   449→  const save = async () => {
   450→    const newConfig = { ...config, text: draft.trim() };
   451→    await onUpdate({ trackConfig: JSON.stringify(newConfig) });
   452→    setIsEditing(false);
   453→  };
   454→
   455→  const cancel = () => {
   456→    setDraft(trackText);
   457→    setIsEditing(false);
   458→  };
   459→
   460→  return (
   461→    <div className="px-4 py-3">
   462→      <div className="flex items-center justify-between mb-2">
   463→        <span
   464→          className="text-[9px] uppercase tracking-widest font-medium"
   465→          style={{ color: '#FCEE0A', textShadow: '0 0 6px rgba(252,238,10,0.3)' }}
   466→        >
   467→          Текст трека
