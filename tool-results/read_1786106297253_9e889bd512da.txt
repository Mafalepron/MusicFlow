    94→/* ── Priority Bars (visual indicator) ──────────────────── */
    95→
    96→function PriorityBars({ priority, size = 'sm' }: { priority: string; size?: 'sm' | 'xs' }) {
    97→  const level = PRIORITY_LEVELS[priority] || 2;
    98→  const hex = PRIORITY_HEX[priority] || '#64748b';
    99→  const barW = size === 'xs' ? 'w-[2.5px]' : 'w-[3px]';
   100→  const gap = size === 'xs' ? 'gap-[1.5px]' : 'gap-[2px]';
   101→
   102→  return (
   103→    <div className={cn('flex items-end', gap)} style={{ height: size === 'xs' ? '10px' : '12px' }}>
   104→      {[1, 2, 3].map(i => (
   105→        <div
   106→          key={i}
   107→          className={cn(barW, 'rounded-sm transition-all duration-200')}
   108→          style={{
   109→            height: i === 1 ? '30%' : i === 2 ? '65%' : '100%',
   110→            backgroundColor: i <= level ? hex : hexToRgba(hex, 0.15),
   111→            boxShadow: i <= level ? `0 0 3px ${hex}60` : 'none',
   112→          }}
   113→        />
   114→      ))}
   115→    </div>
   116→  );
   117→}

   119→/* ── Priority Selector (popover) ──────────────────────── */

   124→  size = 'sm',

   206→        </div>

   214→export default function DescriptionBottomPanel() {
   215→  const {
   216→    boardTasks, selectedTaskId, selectedBoardId,
   217→    selectedStageForPanel, setSelectedStageForPanel,
   218→    setBoardTasks, boards,
   219→  } = useKanbanStore();
   220→
   221→  const [isCollapsed, setIsCollapsed] = useState(false);
   222→  const [members, setMembers] = useState<GroupMember[]>([]);
   223→
   224→  // Bottom panel is ALWAYS pinned when a task is selected at the top.
   225→  // For track tasks → renders StagesList; for regular tasks → renders FlatSubtasksList.
   226→  const task = boardTasks.find(t => t.id === selectedTaskId) || null;
   227→  const taskId = task?.id || null;
   228→  const shouldShow = !!task;
   229→
   230→  const selectedBoard = boards.find(b => b.id === selectedBoardId);
   231→  const boardColor = selectedBoard?.color || '#00d9ff';
   232→
   233→  // Fetch group members for assignee dropdowns
   234→  useEffect(() => {
   235→    const gid = useAuthStore.getState().currentGroupId;
   236→    if (!gid) return;
   237→    let cancelled = false;
   238→    fetch(`/api/groups/${gid}/members`)
   239→      .then(r => r.json())
   240→      .then(d => {
   241→        if (!cancelled && Array.isArray(d)) setMembers(d);
   242→      })
   243→      .catch(() => {});
   244→    return () => { cancelled = true; };
   245→  }, []);
   246→
   247→  // Reset collapse state when switching tasks (prev-tracker pattern avoids useEffect-setState)
   248→  const [prevTaskId, setPrevTaskId] = useState<string | null>(taskId);
   249→  if (taskId !== prevTaskId) {
   250→    setPrevTaskId(taskId);
   251→    setIsCollapsed(false);
   252→  }
   253→
   254→  const reloadTasks = useCallback(async () => {
   255→    if (!selectedBoardId) return;
   256→    // Always fetch deep — non-track boards may still contain track tasks with stages
   257→    const url = `/api/tasks?boardId=${selectedBoardId}&deep=true`;
   258→    const res = await fetch(url);

   261→  }, [selectedBoardId, setBoardTasks]);
   262→
   263→  const c = useMemo<ColorSet>(() => ({
   264→    raw: boardColor,

   276→    a7: hexToRgba(boardColor, 0.7),
   277→  }), [boardColor]);
   278→
   279→  if (!shouldShow || !task) return null;
   280→
   281→  const stages = task.children || [];
   282→  const stagesCount = stages.length;
   283→  const allSubtasks = stages.flatMap(s => s.children || []);
   284→  const subtasksCount = allSubtasks.length;
   285→  // Progress is based on STAGE completion, not subtask completion
   286→  const doneStages = stages.filter(s => s.status === 'done').length;
   287→  const progress = stagesCount > 0 ? Math.round((doneStages / stagesCount) * 100) : 0;
   288→  const progressColor = progress === 100 ? '#34d399' : '#FCEE0A';
   289→
   290→  return (
   291→    <div
   292→      className="flex-shrink-0 flex flex-col cp-panel"

   294→        height: isCollapsed ? '42px' : '360px',

   304→      {/* Header */}
   305→      <div className="cp-header" style={{ borderBottom: isCollapsed ? 'none' : `1px solid ${c.a3}` }}>
   306→        <button
   307→          onClick={() => setIsCollapsed(v => !v)}
   308→          className="flex items-center gap-2 min-w-0 flex-1 text-left group"
   309→          title={isCollapsed ? 'Развернуть' : 'Свернуть'}

   738→/* ── Stages List (for track tasks) ──────────────────────── */
   739→
   740→function StagesList({
   741→  task, boardColor, c, members, selectedStageId, onSelectStage, reloadTasks,
   742→}: {
   743→  task: Task;
   744→  boardColor: string;
   745→  c: ColorSet;
   746→  members: GroupMember[];
   747→  selectedStageId: string | null;
   748→  onSelectStage: (stageId: string) => void;
   749→  reloadTasks: () => Promise<void>;
   750→}) {
   751→  const stages = task.children || [];
   752→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(() => {
   753→    // Auto-expand first stage or the selected one
   754→    const initial: Record<string, boolean> = {};
   755→    if (stages.length > 0) {
   756→      initial[stages[0].id] = true;
   757→    }
   758→    if (selectedStageId && stages.find(s => s.id === selectedStageId)) {
   759→      initial[selectedStageId] = true;
   760→    }
   761→    return initial;
   762→  });
   763→  const [addingStage, setAddingStage] = useState(false);
   764→  const [newStageTitle, setNewStageTitle] = useState('');
   765→
   766→  // If selectedStageId changes, ensure that stage is expanded (prev-tracker pattern avoids useEffect-setState)
   767→  const [prevSelectedStageId, setPrevSelectedStageId] = useState<string | null>(selectedStageId);
   768→  if (selectedStageId !== prevSelectedStageId) {
   769→    setPrevSelectedStageId(selectedStageId);
   770→    if (selectedStageId && stages.find(s => s.id === selectedStageId) && !expandedStages[selectedStageId]) {
   771→      setExpandedStages(prev => ({ ...prev, [selectedStageId]: true }));
   772→    }
   773→  }
   774→
   775→  const toggleExpand = (stageId: string) => {
   776→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   777→  };
   778→
   779→  const addStage = async () => {
   780→    if (!newStageTitle.trim()) return;
   781→    await fetch('/api/tasks', {
   782→      method: 'POST',
   783→      headers: { 'Content-Type': 'application/json' },
   784→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: task.boardId }),
   785→    });
   786→    setNewStageTitle('');
   787→    setAddingStage(false);
   788→    await reloadTasks();
   789→  };
   790→
   791→  const moveStage = async (index: number, direction: -1 | 1) => {
   792→    const target = index + direction;
   793→    if (target < 0 || target >= stages.length) return;
   794→    const a = stages[index];
   795→    const b = stages[target];
   796→    // Swap hexR values
   797→    await Promise.all([
   798→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: a.id, hexR: b.hexR, hexQ: b.hexQ }) }),
   799→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: b.id, hexR: a.hexR, hexQ: a.hexQ }) }),
   800→    ]);
   801→    await reloadTasks();
   802→  };
   803→
   804→  const deleteStage = async (stageId: string) => {
   805→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   806→    await reloadTasks();
   807→  };
   808→
   809→  const updateStage = async (stageId: string, patch: Record<string, unknown>) => {
   810→    await fetch('/api/tasks', {
   811→      method: 'PUT',
   812→      headers: { 'Content-Type': 'application/json' },
   813→      body: JSON.stringify({ id: stageId, ...patch }),
   814→    });
   815→    await reloadTasks();
   816→  };
   817→
   818→  return (
   819→    <div className="py-2 space-y-1.5">
   820→      {stages.length === 0 && !addingStage && (
   821→        <div className="text-center py-6 px-4">
   822→          <ListChecks className="w-6 h-6 mx-auto mb-2 text-slate-700" />
   823→          <p className="text-[11px] text-slate-500 mb-1">Этапов пока нет</p>
   824→          <p className="text-[10px] text-slate-600 mb-3">Создайте первый этап производства трека</p>
   825→        </div>
   826→      )}
   827→
   828→      {stages.map((stage, idx) => (
   829→        <StageCard
   830→          key={stage.id}
   831→          stage={stage}
   832→          index={idx}
   833→          total={stages.length}
   834→          boardColor={boardColor}
   835→          c={c}
   836→          members={members}
   837→          isExpanded={!!expandedStages[stage.id]}
   838→          isSelected={selectedStageId === stage.id}
   839→          onToggleExpand={() => toggleExpand(stage.id)}
   840→          onSelect={() => onSelectStage(stage.id)}
   841→          onMoveUp={() => moveStage(idx, -1)}
   842→          onMoveDown={() => moveStage(idx, 1)}
   843→          onDelete={() => deleteStage(stage.id)}
   844→          onUpdate={(patch) => updateStage(stage.id, patch)}
   845→          reloadTasks={reloadTasks}
   846→        />
   847→      ))}
   848→
   849→      {/* Add stage form */}
   850→      {addingStage ? (
   851→        <div
   852→          className="mx-2 rounded-lg p-2.5 space-y-1.5"
   853→          style={{ border: `1px dashed ${c.a3}`, backgroundColor: c.a04 }}
   854→        >
   855→          <Input
   856→            value={newStageTitle}
   857→            onChange={(e) => setNewStageTitle(e.target.value)}
   858→            onKeyDown={(e) => {
   859→              if (e.key === 'Enter') void addStage();
   860→              if (e.key === 'Escape') { setAddingStage(false); setNewStageTitle(''); }
   861→            }}
   862→            placeholder="Название этапа..."
   863→            className="bg-[rgba(8,8,16,0.92)] text-[11px] text-slate-200 placeholder:text-slate-600 h-7 rounded-md border focus:outline-none focus-visible:ring-0 transition-colors px-2"
   864→            style={{ borderColor: c.a3 }}
   865→            onFocus={(e) => { e.currentTarget.style.borderColor = '#FCEE0A'; }}
   866→            onBlur={(e) => { e.currentTarget.style.borderColor = c.a3; }}
   867→            autoFocus
   868→          />
   869→          <div className="flex items-center gap-1.5">
   870→            <button
   871→              onClick={() => void addStage()}
   872→              disabled={!newStageTitle.trim()}
   873→              className="text-[10px] px-2 py-1 rounded font-medium transition-all disabled:opacity-40"
   874→              style={{ color: '#fff', backgroundColor: c.a5 }}
   875→            >
   876→              + Добавить этап
   877→            </button>
   878→            <button
   879→              onClick={() => { setAddingStage(false); setNewStageTitle(''); }}
   880→              className="text-[10px] text-slate-500 hover:text-slate-300 px-2 py-1"
   881→            >
   882→              Отмена
   883→            </button>
   884→          </div>
   885→        </div>
   886→      ) : (
   887→        <button
   888→          onClick={() => setAddingStage(true)}
   889→          className="cp-add-btn mx-2 w-[calc(100%-1rem)] justify-center"
   890→        >
   891→          <Plus className="w-3 h-3" /> Добавить этап
   892→        </button>
   893→      )}
   894→    </div>
   895→  );
   896→}

   898→/* ── Stage Card ─────────────────────────────────────────── */
   899→
   900→function StageCard({
   901→  stage, index, total, boardColor, c, members,
   902→  isExpanded, isSelected,
   903→  onToggleExpand, onSelect, onMoveUp, onMoveDown, onDelete, onUpdate, reloadTasks,
   904→}: {
   905→  stage: TaskChild;
   906→  index: number;
   907→  total: number;
   908→  boardColor: string;
   909→  c: ColorSet;
   910→  members: GroupMember[];
   911→  isExpanded: boolean;
   912→  isSelected: boolean;
   913→  onToggleExpand: () => void;
   914→  onSelect: () => void;
   915→  onMoveUp: () => void;
   916→  onMoveDown: () => void;
   917→  onDelete: () => void;
   918→  onUpdate: (patch: Record<string, unknown>) => Promise<void>;
   919→  reloadTasks: () => Promise<void>;
   920→}) {
   921→  const [isEditingTitle, setIsEditingTitle] = useState(false);
   922→  const [titleDraft, setTitleDraft] = useState(stage.title);
   923→  const [descDraft, setDescDraft] = useState(stage.description || '');
   924→  const [isEditingDesc, setIsEditingDesc] = useState(false);
   925→
   926→  const subtasks = stage.children || [];
   927→  const stageProgress = getProgress(subtasks);
   928→  const StatusIcon = STATUS_ICON[stage.status] || Circle;
   929→  const statusHex = STATUS_HEX[stage.status] || '#64748b';
   930→  const priorityHex = PRIORITY_HEX[stage.priority] || '#64748b';
   931→
   932→  // Sync local draft state with prop changes when not actively editing (prev-tracker pattern)
   933→  const [prevStageTitle, setPrevStageTitle] = useState(stage.title);
   934→  if (stage.title !== prevStageTitle) {
   935→    setPrevStageTitle(stage.title);
   936→    if (!isEditingTitle) setTitleDraft(stage.title);
   937→  }
   938→  const [prevStageDesc, setPrevStageDesc] = useState(stage.description);
   939→  if (stage.description !== prevStageDesc) {
   940→    setPrevStageDesc(stage.description);
   941→    if (!isEditingDesc) setDescDraft(stage.description || '');
   942→  }
   943→
   944→  const saveTitle = async () => {
   945→    if (titleDraft.trim() && titleDraft !== stage.title) {

   948→    setIsEditingTitle(false);
   949→  };

   951→  const saveDesc = async () => {
   952→    if (descDraft.trim() !== (stage.description || '').trim()) {
   953→      await onUpdate({ description: descDraft.trim() || null });
   954→    }
   955→    setIsEditingDesc(false);
   956→  };
   957→

   959→    const order: TaskStatus[] = ['todo', 'in-progress', 'done'];
   960→    const idx = order.indexOf(stage.status as TaskStatus);
   961→    const next = order[(idx + 1) % order.length];
   962→    await onUpdate({ status: next });
   963→  };
   964→
   965→  const deadlineInfo = getDeadlineInfo(stage.deadline || null);
   966→

   968→
   969→  return (

   982→          onClick={(e) => { e.stopPropagation(); void cycleStatus(); }}
   983→          className="flex-shrink-0 p-1 transition-all hover:opacity-70"
   984→          title="Сменить статус"
   985→          style={{ filter: `drop-shadow(0 0 4px ${statusHex}60)` }}
   986→        >
   987→          <StatusIcon className="w-4 h-4" style={{ color: statusHex }} />
   988→        </button>
   989→
   990→        {/* Priority selector (interactive bars) */}
   991→        <PrioritySelector
   992→          priority={stage.priority}
   993→          onChange={(v) => void onUpdate({ priority: v })}
   994→          size="sm"
   995→          boardColor={boardColor}
   996→        />
   997→
   998→        {/* Title (inline editable) */}
   999→        {isEditingTitle ? (
  1000→          <input
  1001→            value={titleDraft}
  1002→            onChange={(e) => setTitleDraft(e.target.value)}
  1003→            onKeyDown={(e) => {
  1004→              if (e.key === 'Enter') { e.preventDefault(); void saveTitle(); }
  1005→              if (e.key === 'Escape') { setIsEditingTitle(false); setTitleDraft(stage.title); }
  1006→            }}
  1007→            onBlur={() => void saveTitle()}
  1008→            className="flex-1 min-w-0 bg-[rgba(8,8,16,0.92)] rounded-md px-2 py-1 text-[12px] text-slate-100 font-medium focus:outline-none focus-visible:ring-0 transition-colors"
  1009→            style={{ border: `1.5px solid ${c.a5}` }}
  1010→            onFocus={(e) => { e.currentTarget.style.borderColor = '#FCEE0A'; }}
  1011→            autoFocus
  1012→          />
  1013→        ) : (
  1014→          <button
  1015→            onDoubleClick={() => setIsEditingTitle(true)}
  1016→            onClick={() => { onSelect(); onToggleExpand(); }}
  1017→            className={cn(
  1018→              'cp-stage-title flex-1 min-w-0 text-left truncate transition-colors',
  1019→              stage.status === 'done' ? 'text-slate-600 line-through' : 'text-[#FCEE0A]'
  1020→            )}
  1021→            title="Двойной клик — переименовать"
  1022→          >
  1023→            {stage.title}
  1024→          </button>
  1025→        )}
  1026→
  1027→        {/* Subtask count badge — fixed width so percentages align across all stages */}
  1028→        {subtasks.length > 0 && (
  1029→          <span
  1030→            className="text-[10px] px-2 py-0.5 rounded-full flex-shrink-0 tabular-nums font-bold text-right"

  1033→          >

  1035→          </span>

  1193→function SubtasksList({
  1194→  stage, boardColor, c, members, onUpdate, onDelete, onMove, reloadTasks,
  1195→}: {
  1196→  stage: TaskChild;
  1197→  boardColor: string;
  1198→  c: ColorSet;
  1199→  members: GroupMember[];
  1200→  onUpdate: (subId: string, patch: Record<string, unknown>) => Promise<void>;
  1201→  onDelete: (subId: string) => Promise<void>;
  1202→  onMove: (idx: number, dir: -1 | 1) => Promise<void>;
  1203→  reloadTasks: () => Promise<void>;
  1204→}) {
  1205→  const subtasks = stage.children || [];
  1206→  const [expandedSubs, setExpandedSubs] = useState<Record<string, boolean>>({});
  1207→  const [addingSub, setAddingSub] = useState(false);
  1208→  const [newSubTitle, setNewSubTitle] = useState('');
  1209→
  1210→  const toggleExpand = (id: string) => setExpandedSubs(prev => ({ ...prev, [id]: !prev[id] }));
  1211→
  1212→  const addSub = async () => {
  1213→    if (!newSubTitle.trim()) return;
  1214→    await fetch('/api/tasks', {
  1215→      method: 'POST',
  1216→      headers: { 'Content-Type': 'application/json' },
  1217→      body: JSON.stringify({ title: newSubTitle.trim(), parentId: stage.id }),
  1218→    });
  1219→    setNewSubTitle('');
  1220→    setAddingSub(false);
  1221→    await reloadTasks();
  1222→  };
  1223→
  1224→  if (subtasks.length === 0 && !addingSub) {
  1225→    return (
  1226→      <div className="ml-3 pl-2 py-1">
  1227→        <button
  1228→          onClick={() => setAddingSub(true)}
  1229→          className="cp-add-btn"
  1230→        >
  1231→          <Plus className="w-2.5 h-2.5" /> Добавить подзадачу
  1232→        </button>
  1233→      </div>
  1234→    );
  1235→  }
  1236→
  1237→  return (
  1238→    <div className="ml-3 pl-2 space-y-0.5">

  1241→          key={sub.id}

  1244→          total={subtasks.length}
  1245→          boardColor={boardColor}
  1246→          c={c}
  1247→          members={members}
  1248→          isExpanded={!!expandedSubs[sub.id]}
  1249→          onToggleExpand={() => toggleExpand(sub.id)}
  1250→          onMoveUp={() => onMove(idx, -1)}
  1251→          onMoveDown={() => onMove(idx, 1)}
  1252→          onDelete={() => onDelete(sub.id)}
  1253→          onUpdate={(patch) => onUpdate(sub.id, patch)}
  1254→        />
  1255→      ))}
  1256→
  1257→      {addingSub ? (
  1258→        <div className="pt-1">
  1259→          <Input
  1260→            value={newSubTitle}
  1261→            onChange={(e) => setNewSubTitle(e.target.value)}
  1262→            onKeyDown={(e) => {
  1263→              if (e.key === 'Enter') void addSub();
  1264→              if (e.key === 'Escape') { setAddingSub(false); setNewSubTitle(''); }
  1265→            }}
  1266→            placeholder="Название подзадачи..."
  1267→            className="bg-[rgba(8,8,16,0.92)] text-[10px] text-slate-300 placeholder:text-slate-600 h-6 rounded-md border focus:outline-none focus-visible:ring-0 transition-colors px-2"
  1268→            style={{ borderColor: c.a25 }}
  1269→            onFocus={(e) => { e.currentTarget.style.borderColor = '#FCEE0A'; }}
  1270→            onBlur={(e) => { e.currentTarget.style.borderColor = c.a25; }}
  1271→            autoFocus
  1272→          />
  1273→          <div className="flex items-center gap-1.5 mt-1">
  1274→            <button
  1275→              onClick={() => void addSub()}
  1276→              disabled={!newSubTitle.trim()}
  1277→              className="text-[9px] px-1.5 py-0.5 rounded font-medium transition-all disabled:opacity-40"
  1278→              style={{ color: '#fff', backgroundColor: c.a5 }}
  1279→            >
  1280→              + Добавить
  1281→            </button>
  1282→            <button
  1283→              onClick={() => { setAddingSub(false); setNewSubTitle(''); }}
  1284→              className="text-[9px] text-slate-500 hover:text-slate-300"
  1285→            >
  1286→              Отмена
  1287→            </button>
  1288→          </div>
  1289→        </div>
  1290→      ) : (
  1291→        <button
  1292→          onClick={() => setAddingSub(true)}
  1293→          className="cp-add-btn"
  1294→        >
  1295→          <Plus className="w-2.5 h-2.5" /> Добавить подзадачу
  1296→        </button>

  1307→}: {
  1308→  subtask: TaskGrandchild;
  1309→  index: number;
  1310→  total: number;
  1311→  boardColor: string;
  1312→  c: ColorSet;
  1313→  members: GroupMember[];
  1314→  isExpanded: boolean;
  1315→  onToggleExpand: () => void;
  1316→  onMoveUp: () => void;

  1319→  onUpdate: (patch: Record<string, unknown>) => Promise<void>;

  1324→  const [isEditingDesc, setIsEditingDesc] = useState(false);
  1325→
  1326→  const StatusIcon = STATUS_ICON[subtask.status] || Circle;
  1327→  const statusHex = STATUS_HEX[subtask.status] || '#64748b';
  1328→  const priorityHex = PRIORITY_HEX[subtask.priority] || '#64748b';
  1329→  const subDone = subtask.status === 'done';
  1330→
  1331→  // Sync local draft state with prop changes when not actively editing (prev-tracker pattern)
  1332→  const [prevSubTitle, setPrevSubTitle] = useState(subtask.title);
  1333→  if (subtask.title !== prevSubTitle) {
  1334→    setPrevSubTitle(subtask.title);
  1335→    if (!isEditingTitle) setTitleDraft(subtask.title);
  1336→  }
  1337→  const [prevSubDesc, setPrevSubDesc] = useState(subtask.description);
  1338→  if (subtask.description !== prevSubDesc) {
  1339→    setPrevSubDesc(subtask.description);
  1340→    if (!isEditingDesc) setDescDraft(subtask.description || '');
  1341→  }
  1342→
  1343→  const saveTitle = async () => {
  1344→    if (titleDraft.trim() && titleDraft !== subtask.title) {
  1345→      await onUpdate({ title: titleDraft.trim() });
  1346→    }
  1347→    setIsEditingTitle(false);
  1348→  };
  1349→
  1350→  const saveDesc = async () => {
  1351→    if (descDraft.trim() !== (subtask.description || '').trim()) {
  1352→      await onUpdate({ description: descDraft.trim() || null });

  1354→    setIsEditingDesc(false);
  1355→  };
  1356→
  1357→  const cycleStatus = async () => {
  1358→    const order: TaskStatus[] = ['todo', 'in-progress', 'done'];
  1359→    const idx = order.indexOf(subtask.status as TaskStatus);
  1360→    const next = order[(idx + 1) % order.length];
  1361→    await onUpdate({ status: next });
  1362→  };
  1363→
  1364→  const assigneeMember = null; // unused — AssigneePicker handles display
  1365→
  1366→  return (
  1367→    <div>
  1368→      <div className={cn('cp-subtask-row group/sub', subDone && 'cp-subtask-row-done')}>
  1369→        {/* Status cycle */}
  1370→        <button
  1371→          onClick={(e) => { e.stopPropagation(); void cycleStatus(); }}
  1372→          className="flex-shrink-0 p-0.5 rounded transition-opacity hover:opacity-70"
  1373→          title="Сменить статус"
  1374→        >
  1375→          <StatusIcon className="w-3.5 h-3.5" style={{ color: statusHex }} />
  1376→        </button>
  1377→
  1378→        {/* Priority selector (interactive bars) */}
  1379→        <PrioritySelector
  1380→          priority={subtask.priority}
  1381→          onChange={(v) => void onUpdate({ priority: v })}
  1382→          size="xs"
  1383→          boardColor={boardColor}
  1384→        />
  1385→
  1386→        {/* Title (inline editable) */}
  1387→        {isEditingTitle ? (
  1388→          <input
  1389→            value={titleDraft}
  1390→            onChange={(e) => setTitleDraft(e.target.value)}
  1391→            onKeyDown={(e) => {
  1392→              if (e.key === 'Enter') { e.preventDefault(); void saveTitle(); }
  1393→              if (e.key === 'Escape') { setIsEditingTitle(false); setTitleDraft(subtask.title); }
  1394→            }}
  1395→            onBlur={() => void saveTitle()}
  1396→            className="flex-1 min-w-0 bg-[rgba(8,8,16,0.92)] rounded px-2 py-0.5 text-[11px] text-slate-100 focus:outline-none focus-visible:ring-0 transition-colors"
  1397→            style={{ border: `1.5px solid ${c.a4}` }}
  1398→            onFocus={(e) => { e.currentTarget.style.borderColor = '#FCEE0A'; }}
  1399→            autoFocus
  1400→          />
  1401→        ) : (
  1402→          <button
  1403→            onDoubleClick={() => setIsEditingTitle(true)}
  1404→            onClick={onToggleExpand}
  1405→            className={cn(
  1406→              'flex-1 min-w-0 text-left text-[11px] font-medium truncate transition-colors',
  1407→              subDone ? 'text-slate-600 line-through' : 'text-[#FCEE0A]'
  1408→            )}
  1409→            title="Двойной клик — переименовать"
  1410→          >

  1658→function parseAssignees(raw: string | null): string[] {
  1659→  if (!raw) return [];

  1661→    const parsed = JSON.parse(raw);
  1662→    if (Array.isArray(parsed)) return parsed.filter(Boolean);
  1663→    return [raw];
  1664→  } catch {
  1665→    return [raw];
  1666→  }

  1670→  if (ids.length === 0) return null;
