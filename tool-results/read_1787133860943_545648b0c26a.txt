    54→/* ────────────────────────────────────────────────────────────────────────── */
    55→/*  Group switcher — small < name > control. Only renders when the user      */
    56→/*  has more than one group.                                                 */
    57→/* ────────────────────────────────────────────────────────────────────────── */
    58→function GroupSwitcher({
    59→  groups,
    60→  currentIndex,
    61→  onSelect,
    62→}: {
    63→  groups: Group[];
    64→  currentIndex: number;
    65→  onSelect: (index: number) => void;
    66→}) {
    67→  if (groups.length <= 1) return null;
    68→  const atStart = currentIndex <= 0;
    69→  const atEnd = currentIndex >= groups.length - 1;
    70→  const total = groups.length;
    71→
    72→  return (
    73→    <div className="flex items-center justify-center gap-2 mt-2">
    74→      <button
    75→        onClick={() => !atStart && onSelect(currentIndex - 1)}
    76→        disabled={atStart}
    77→        aria-label="Previous group"
    78→        className={cn(
    79→          'flex h-6 w-6 items-center justify-center transition-all',
    80→          atStart ? 'opacity-30 cursor-not-allowed' : 'hover:opacity-100 opacity-70'
    81→        )}
    82→        style={{
    83→          clipPath: BTN_CLIP,
    84→          background: hexToRgba(CYAN, 0.08),
    85→          boxShadow: `inset 0 0 0 1px ${hexToRgba(CYAN, 0.3)}`,
    86→        }}
    87→      >
    88→        <ChevronLeft className="h-3.5 w-3.5" style={{ color: CYAN }} />
    89→      </button>
    90→      <span
    91→        className="text-[10px] uppercase tracking-[0.18em] font-bold"
    92→        style={{ color: hexToRgba('#e2e8f0', 0.7) }}
    93→      >
    94→        {currentIndex + 1} / {total}
    95→      </span>
    96→      <button
    97→        onClick={() => !atEnd && onSelect(currentIndex + 1)}
    98→        disabled={atEnd}
    99→        aria-label="Next group"
   100→        className={cn(
   101→          'flex h-6 w-6 items-center justify-center transition-all',
   102→          atEnd ? 'opacity-30 cursor-not-allowed' : 'hover:opacity-100 opacity-70'
   103→        )}
   104→        style={{
   105→          clipPath: BTN_CLIP,
   106→          background: hexToRgba(CYAN, 0.08),
   107→          boxShadow: `inset 0 0 0 1px ${hexToRgba(CYAN, 0.3)}`,
   108→        }}
   109→      >
   110→        <ChevronRight className="h-3.5 w-3.5" style={{ color: CYAN }} />
   111→      </button>
   112→    </div>
   113→  );
   114→}

   116→/* ────────────────────────────────────────────────────────────────────────── */
   117→/*  Editable group description — click to edit, PATCH /api/groups/[id]       */
   118→/*  on blur or Cmd/Ctrl+Enter.                                               */
   119→/* ────────────────────────────────────────────────────────────────────────── */
   120→function EditableDescription({
   121→  group,
   122→  onSaved,
   123→}: {
   124→  group: Group | null;
   125→  onSaved?: (next: Group) => void;
   126→}) {
   127→  const [editing, setEditing] = useState(false);
   128→  const [draft, setDraft] = useState(group?.description || '');
   129→  const [saving, setSaving] = useState(false);
   130→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   131→
   132→  // Keep draft in sync with the latest group description when not editing.
   133→  useEffect(() => {
   134→    if (!editing) setDraft(group?.description || '');
   135→  }, [group?.description, editing, group?.id]);
   136→
   137→  // Focus + select-all when entering edit mode.
   138→  useEffect(() => {
   139→    if (editing && textareaRef.current) {
   140→      textareaRef.current.focus();
   141→      textareaRef.current.select();
   142→    }
   143→  }, [editing]);
   144→
   145→  const save = useCallback(async () => {
   146→    if (!group) return;
   147→    const trimmed = draft.trim();
   148→    if (trimmed === (group.description || '').trim()) {
   149→      setEditing(false);
   150→      return;
   151→    }
   152→    setSaving(true);
   153→    try {
   154→      const res = await fetch(`/api/groups/${group.id}`, {
   155→        method: 'PATCH',
   156→        headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify({ description: trimmed }),
   158→      });
   159→      if (res.ok) {
   160→        const updated = await res.json();
   161→        onSaved?.(updated);
   162→      }
   163→    } catch {
   164→      /* swallow */
   165→    } finally {
   166→      setSaving(false);
   167→      setEditing(false);
   168→    }
   169→  }, [draft, group, onSaved]);
   170→
   171→  if (editing) {
   172→    return (
   173→      <div className="mt-2">
   174→        <textarea
   175→          ref={textareaRef}
   176→          value={draft}
   177→          onChange={(e) => setDraft(e.target.value)}
   178→          onBlur={save}
   179→          onKeyDown={(e) => {
   180→            if ((e.metaKey || e.ctrlKey) && e.key === 'Enter') {
   181→              e.preventDefault();
   182→              save();
   183→            }
   184→            if (e.key === 'Escape') {
   185→              setDraft(group?.description || '');
   186→              setEditing(false);
   187→            }
   188→          }}
   189→          rows={3}
   190→          maxLength={500}
   191→          placeholder="Введите описание группы…"
   192→          className="w-full resize-none rounded-md bg-[#070b14] px-2 py-1.5 text-[11px] leading-relaxed text-slate-200 placeholder:text-slate-600 focus:outline-none"
   193→          style={{
   194→            clipPath: BTN_CLIP,
   195→            boxShadow: `inset 0 0 0 1px ${hexToRgba(CYAN, 0.5)}`,
   196→          }}
   197→        />
   198→        <div className="mt-1 flex items-center justify-between">
   199→          <span className="text-[9px] uppercase tracking-wider text-slate-600">
   200→            {saving ? 'Сохранение…' : '⌘/Ctrl+Enter — сохранить · Esc — отмена'}
   201→          </span>
   202→          <span className="text-[9px] tabular-nums text-slate-600">{draft.length}/500</span>
   203→        </div>
   204→      </div>
   205→    );
   206→  }
   207→
   208→  return (
   209→    <button
   210→      onClick={() => setEditing(true)}
   211→      className="group mt-2 block w-full text-left"
   212→      title="Нажмите, чтобы редактировать описание"
   213→    >
   214→      <div
   215→        className="relative px-2 py-1.5 text-[11px] leading-relaxed text-slate-300 transition-all"
   216→        style={{
   217→          clipPath: BTN_CLIP,
   218→          background: hexToRgba(CYAN, 0.04),
   219→          boxShadow: `inset 0 0 0 1px ${hexToRgba(CYAN, 0.18)}`,
   220→        }}
   221→      >
   222→        {group?.description?.trim() ? (
   223→          <p className="whitespace-pre-wrap break-words pr-5">{group.description}</p>
   224→        ) : (
   225→          <p className="italic text-slate-500 pr-5">Нет описания. Нажмите, чтобы добавить…</p>
   226→        )}
   227→        <Pencil
   228→          className="absolute right-1.5 top-1.5 h-3 w-3 text-slate-600 opacity-0 transition-opacity group-hover:opacity-100"
   229→        />
   230→      </div>
   231→    </button>
   232→  );
   233→}

   235→/* ────────────────────────────────────────────────────────────────────────── */
   236→/*  Artist profile card — group avatar, name, genre, editable description,   */
   237→/*  performance stats, group switcher, linked projects list.                  */
   238→/* ────────────────────────────────────────────────────────────────────────── */
   239→function ArtistProfileCard() {
   240→  const { user, currentGroupId, setCurrentGroupId, setCurrentGroupName, setCurrentGroupInviteCode } = useAuthStore();
   241→  const currentGroup = useDataStore((s) => s.currentGroup);
   242→  const setCurrentGroup = useDataStore((s) => s.setCurrentGroup);
   243→  const projects = useDataStore((s) => s.projects);
   244→  const tracks = useDataStore((s) => s.tracks);
   245→  const ideas = useDataStore((s) => s.ideas);
   246→  const navigate = useNavigationStore((s) => s.navigate);
   247→
   248→  // Local copy of the user's group memberships — fetched on mount via
   249→  // /api/groups?userId=… so the < > switcher can work even though the
   250→  // global `groups` array isn't populated elsewhere.
   251→  const [userGroups, setUserGroups] = useState<Group[]>([]);
   252→  useEffect(() => {
   253→    if (!user?.id) return;
   254→    let cancelled = false;
   255→    fetch(`/api/groups?userId=${user.id}`)
   256→      .then((r) => (r.ok ? r.json() : []))
   257→      .then((list: Group[]) => {
   258→        if (!cancelled) setUserGroups(list);
   259→      })
   260→      .catch(() => {});
   261→    return () => {
   262→      cancelled = true;
   263→    };
   264→  }, [user?.id]);
   265→
   266→  const currentIndex = useMemo(() => {
   267→    if (!currentGroupId) return 0;
   268→    const idx = userGroups.findIndex((g) => g.id === currentGroupId);
   269→    return idx === -1 ? 0 : idx;
   270→  }, [userGroups, currentGroupId]);
   271→
   272→  // Group switcher handler — set currentGroupId + pull fresh group info.
   273→  const handleSwitchGroup = useCallback(
   274→    (nextIndex: number) => {
   275→      const next = userGroups[nextIndex];
   276→      if (!next) return;
   277→      setCurrentGroupId(next.id);
   278→      setCurrentGroupName(next.name);
   279→      setCurrentGroupInviteCode(next.inviteCode);
   280→      // Refetch the full group record so currentGroup has all fields.
   281→      fetch(`/api/groups/${next.id}`)
   282→        .then((r) => (r.ok ? r.json() : null))
   283→        .then((g) => {
   284→          if (g) {
   285→            setCurrentGroup(g);
   286→            setCurrentGroupName(g.name);
   287→            setCurrentGroupInviteCode(g.inviteCode);
   288→          }
   289→        })
   290→        .catch(() => {});
   291→    },
   292→    [userGroups, setCurrentGroupId, setCurrentGroupName, setCurrentGroupInviteCode, setCurrentGroup]
   293→  );
   294→
   295→  // Projects linked to this group — filter by groupId.
   296→  const groupProjects = useMemo<Project[]>(() => {
   297→    if (!currentGroup) return [];
   298→    return projects.filter((p) => p.groupId === currentGroup.id);
   299→  }, [projects, currentGroup]);
   300→
   301→  // Tracks count for this group.
   302→  const groupTrackCount = useMemo(() => {
   303→    if (!currentGroup) return 0;
   304→    const ids = new Set(groupProjects.map((p) => p.id));
   305→    return tracks.filter((t) => ids.has(t.projectId)).length;
   306→  }, [tracks, groupProjects, currentGroup]);
   307→
   308→  // Ideas count for this group.
   309→  const groupIdeaCount = useMemo(() => {
   310→    if (!currentGroup) return 0;
   311→    return ideas.filter((i) => i.groupId === currentGroup.id).length;
   312→  }, [ideas, currentGroup]);
   313→
   314→  // Member count — fetched alongside group info.
   315→  const [memberCount, setMemberCount] = useState(0);
   316→  useEffect(() => {
   317→    if (!currentGroup?.id) return;
   318→    fetch(`/api/groups/${currentGroup.id}/members`)
   319→      .then((r) => (r.ok ? r.json() : []))
   320→      .then((m: unknown) => setMemberCount(Array.isArray(m) ? m.length : 0))
   321→      .catch(() => {});
   322→  }, [currentGroup?.id]);
   323→
   324→  // Created date (formatted).
   325→  const createdLabel = useMemo(() => {
   326→    if (!currentGroup) return '—';
   327→    const raw = (userGroups.find((g) => g.id === currentGroup.id) as unknown as { createdAt?: string } | undefined)?.createdAt;
   328→    if (!raw) return '—';
   329→    try {
   330→      return new Date(raw).toLocaleDateString('ru-RU', { year: 'numeric', month: 'short', day: 'numeric' });
   331→    } catch {
   332→      return '—';
   333→    }
   334→  }, [currentGroup, userGroups]);
   335→
   336→  if (!currentGroup) {
   337→    return (
   338→      <div className="px-3 py-3">
   339→        <div
   340→          className="p-3 text-center text-[11px] text-slate-500"
   341→          style={{
   342→            clipPath: CARD_CLIP,
   343→            background: 'rgba(8,12,22,0.9)',
   344→            boxShadow: `inset 0 0 0 1px ${hexToRgba(CYAN, 0.2)}`,
   345→          }}
   346→        >
   347→          Группа не выбрана
   348→        </div>
   349→      </div>
   350→    );
   351→  }
   352→
   353→  const stats: { icon: typeof Users; label: string; value: number; color: string }[] = [
   354→    { icon: Users, label: 'Участн.', value: memberCount, color: CYAN },
   355→    { icon: FolderOpen, label: 'Проекты', value: groupProjects.length, color: YELLOW },
   356→    { icon: Music2, label: 'Треки', value: groupTrackCount, color: AMBER },
   357→    { icon: Lightbulb, label: 'Идеи', value: groupIdeaCount, color: GREEN },
   358→  ];
   359→
   360→  return (
   361→    <div className="relative px-3 py-3">
   362→      <div
   363→        className="relative p-3"
   364→        style={{
   365→          clipPath: CARD_CLIP,
   366→          background:
   367→            'linear-gradient(160deg, rgba(10,14,22,0.95) 0%, rgba(8,12,20,0.85) 100%)',
   368→          boxShadow: `inset 0 0 0 1px ${hexToRgba(CYAN, 0.3)}, 0 0 18px ${hexToRgba(CYAN, 0.06)}`,
   369→        }}
   370→      >
   371→        {/* Top accent stripe */}
   372→        <div
   373→          className="absolute left-0 right-0 top-0 h-px"
   374→          style={{
   375→            background: `linear-gradient(90deg, transparent, ${hexToRgba(YELLOW, 0.6)}, transparent)`,
   376→          }}
   377→        />
   378→
   379→        {/* Avatar + switcher row */}
   380→        <div className="flex flex-col items-center pt-1">
   381→          <div
   382→            className="relative flex h-16 w-16 items-center justify-center"
   383→            style={{
   384→              clipPath: AVATAR_CLIP,
   385→              background: hexToRgba(YELLOW, 0.1),
   386→              boxShadow: `inset 0 0 0 1.5px ${hexToRgba(YELLOW, 0.5)}, 0 0 18px ${hexToRgba(YELLOW, 0.15)}`,
   387→            }}
   388→          >
   389→            <Avatar className="h-14 w-14" style={{ borderRadius: 0 }}>
   390→              <AvatarImage src={currentGroup.avatarUrl} alt={currentGroup.name} />
   391→              <AvatarFallback
   392→                className="bg-transparent text-lg font-bold"
   393→                style={{
   394→                  color: YELLOW,
   395→                  textShadow: `0 0 8px ${hexToRgba(YELLOW, 0.6)}`,
   396→                }}
   397→              >
   398→                {currentGroup.name?.charAt(0)?.toUpperCase() || 'G'}

   401→          </div>
   402→
   403→          {/* Group switcher — only when multiple groups */}
   404→          <GroupSwitcher
   405→            groups={userGroups}
   406→            currentIndex={currentIndex}
   407→            onSelect={handleSwitchGroup}
   408→          />
   409→        </div>
   410→
   411→        {/* Group name + genre */}
   412→        <div className="mt-2 text-center">
   413→          <p
   414→            className="truncate text-sm font-bold uppercase tracking-[0.14em]"
   415→            style={{
   416→              color: YELLOW,
   417→              textShadow: `0 0 8px ${hexToRgba(YELLOW, 0.5)}`,
   418→            }}
   419→            title={currentGroup.name}
   420→          >
   421→            {currentGroup.name}
   422→          </p>
   423→          {currentGroup.genre && (
   424→            <p
   425→              className="mt-0.5 text-[10px] uppercase tracking-[0.18em] font-medium"
   426→              style={{ color: CYAN }}

   434→        {/* Invite code (small, with copy) */}

   451→                clipPath: BTN_CLIP,
