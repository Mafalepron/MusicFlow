   107→   107→        ? selectedInstruments.map(i => `Запись: ${i}`)
   108→   108→        : STAGE_SUBTASKS[key] || [];

   110→   110→        id: uid(),
   111→   111→        emoji: tpl.emoji,

   137→   137→            id: uid(),
   138→   138→            emoji: tpl.emoji,
   139→   139→            label: tpl.label,
   140→   140→            description: '',
   141→   141→            subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
   142→   142→            sourceKey: tpl.key,
   143→   143→          });

   165→   165→    }));

   192→   192→  };

   195→   195→    setEditableStages(prev => prev.filter(s => s.id !== stageId));
   196→   196→  };

   198→   198→  const addNewStage = () => {

   201→   201→      id: uid(),
   202→   202→      emoji: DEFAULT_STAGE_EMOJIS[editableStages.length % DEFAULT_STAGE_EMOJIS.length],

   215→   215→    setEditableStages(prev => prev.map(s => {
   216→   216→      if (s.id !== stageId) return s;
   217→   217→      return { ...s, subtasks: s.subtasks.map(st => st.id === subtaskId ? { ...st, ...updates } : st) };

   222→   222→    setEditableStages(prev => prev.map(s => {
   223→   223→      if (s.id !== stageId) return s;
   224→   224→      return { ...s, subtasks: s.subtasks.filter(st => st.id !== subtaskId) };
   225→   225→    }));

   348→   348→              />
   349→   349→            </div>

   368→   368→              <div className="flex items-center gap-2 mb-1">
