   153→// Build a nested tree from flat comment list
   154→interface CommentNode extends Comment {
   155→  replies: CommentNode[];
   156→}
   157→
   158→function buildCommentTree(comments: Comment[]): CommentNode[] {
   159→  const map = new Map<string, CommentNode>();
   160→  const roots: CommentNode[] = [];
   161→  
   162→  for (const c of comments) {
   163→    map.set(c.id, { ...c, replies: [] });
   164→  }
   165→  for (const c of comments) {
   166→    const node = map.get(c.id)!;
   167→    if (c.parentId && map.has(c.parentId)) {
   168→      map.get(c.parentId)!.replies.push(node);
   169→    } else {
   170→      roots.push(node);
   171→    }
   172→  }
   173→  roots.sort((a, b) => a.timestampMs - b.timestampMs);
   174→  for (const r of roots) {
   175→    r.replies.sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   176→  }
   177→  return roots;
   178→}

   199→function formatTimestamp(ms: number): string {
   200→  const totalSeconds = ms / 1000;
   201→  const minutes = Math.floor(totalSeconds / 60);
   202→  const seconds = Math.floor(totalSeconds % 60);
   203→  const milliseconds = Math.floor((totalSeconds % 1) * 10);
   204→  return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}.${milliseconds}`;
   205→}
   206→
   207→function formatDuration(seconds: number): string {
   208→  const mins = Math.floor(seconds / 60);
   209→  const secs = Math.floor(seconds % 60);
   210→  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   211→}

   213→function getInitials(name?: string | null): string {
   214→  if (!name) return '?';
   215→  return name
   216→    .trim()
   217→    .split(' ')
   218→    .map((n) => n[0])
   219→    .join('')
   220→    .toUpperCase()
   221→    .slice(0, 2);
   222→}
