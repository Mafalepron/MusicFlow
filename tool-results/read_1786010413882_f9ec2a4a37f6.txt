    94→/* ── Priority Bars (visual indicator) ──────────────────── */
    95→
    96→function PriorityBars({ priority, size = 'sm' }: { priority: string; size?: 'sm' | 'xs' }) {
    97→  const level = PRIORITY_LEVELS[priority] || 2;
    98→  const hex = PRIORITY_HEX[priority] || '#64748b';
    99→  const barW = size === 'xs' ? 'w-[2.5px]' : 'w-[3px]';
   100→  const gap = size === 'xs' ? 'gap-[1.5px]' : 'gap-[2px]';
   101→
   102→  return (
   103→    <div className={cn('flex items-end', gap)} style={{ height: size === 'xs' ? '10px' : '12px' }}>
   104→      {[1, 2, 3].map(i => (
   105→        <div
   106→          key={i}
   107→          className={cn(barW, 'rounded-sm transition-all duration-200')}
   108→          style={{
   109→            height: i === 1 ? '30%' : i === 2 ? '65%' : '100%',
   110→            backgroundColor: i <= level ? hex : hexToRgba(hex, 0.15),
   111→            boxShadow: i <= level ? `0 0 3px ${hex}60` : 'none',
   112→          }}
   113→        />
   114→      ))}
   115→    </div>
   116→  );
   117→}

   119→/* ── Priority Selector (popover) ──────────────────────── */
   120→
   121→function PrioritySelector({
   122→  priority,
   123→  onChange,
   124→  size = 'sm',
   125→}: {
   126→  priority: string;
   127→  onChange: (v: string) => void;
   128→  size?: 'sm' | 'xs';
   129→}) {
   130→  const [open, setOpen] = useState(false);
   131→  const level = PRIORITY_LEVELS[priority] || 2;
   132→
   133→  return (
   134→    <Popover open={open} onOpenChange={setOpen}>
   135→      <PopoverTrigger asChild>
   136→        <button
   137→          type="button"
   138→          onClick={(e) => e.stopPropagation()}
   139→          className={cn(
   140→            'flex items-center gap-1.5 rounded-md px-1.5 py-0.5 transition-all cursor-pointer border border-transparent hover:bg-slate-800/60',
   141→            size === 'xs' ? 'h-5' : 'h-6',
   142→          )}
   143→          title={`Приоритет: ${PRIORITY_LABELS[priority] || priority}`}
   144→        >
   145→          <PriorityBars priority={priority} size={size} />
   146→          <span
   147→            className={cn('font-medium', size === 'xs' ? 'text-[9px]' : 'text-[10px]')}
   148→            style={{ color: PRIORITY_HEX[priority] || '#64748b' }}
   149→          >
   150→            {level === 1 ? 'P1' : level === 2 ? 'P2' : 'P3'}
   151→          </span>
   152→        </button>
   153→      </PopoverTrigger>
   154→      <PopoverContent
   155→        className="w-44 p-2 bg-slate-900 border-slate-700/80 shadow-2xl shadow-black/40 z-[70]"
   156→        align="start"
   157→        sideOffset={6}
   158→        onClick={(e) => e.stopPropagation()}
   159→      >
   160→        <p className="text-[9px] uppercase tracking-wider text-slate-600 font-semibold mb-1.5 px-1">Приоритет</p>
   161→        <div className="space-y-1">
   162→          {PRIORITIES.map(p => {
   163→            const isActive = priority === p.value;
   164→            const pLevel = PRIORITY_LEVELS[p.value];
   165→            return (
   166→              <button
   167→                key={p.value}
   168→                onClick={(e) => {
   169→                  e.stopPropagation();
   170→                  onChange(p.value);
   171→                  setOpen(false);
   172→                }}
   173→                className={cn(
   174→                  'w-full flex items-center gap-2.5 px-2 py-1.5 rounded-md transition-all',
   175→                  isActive ? 'bg-slate-800' : 'hover:bg-slate-800/60',
   176→                )}
   177→                style={isActive ? { boxShadow: `inset 2px 0 0 ${p.hex}` } : undefined}
   178→              >
   179→                <PriorityBars priority={p.value} size="sm" />
   180→                <div className="flex-1 text-left">
   181→                  <span className={cn('text-[11px] font-medium', isActive ? 'text-slate-100' : 'text-slate-300')}>
   182→                    {p.label}
   183→                  </span>
   184→                  <span className="text-[9px] text-slate-600 ml-1">P{pLevel}</span>
   185→                </div>
   186→                {isActive && <Check className="w-3 h-3" style={{ color: p.hex }} />}
   187→              </button>
   188→            );
   189→          })}
   190→        </div>
   191→      </PopoverContent>
   192→    </Popover>
   193→  );
   194→}

   698→/* ── Stages List (for track tasks) ──────────────────────── */
   699→
   700→function StagesList({
   701→  task, boardColor, c, members, selectedStageId, onSelectStage, reloadTasks,
   702→}: {
   703→  task: Task;
   704→  boardColor: string;
   705→  c: ColorSet;
   706→  members: GroupMember[];
   707→  selectedStageId: string | null;
   708→  onSelectStage: (stageId: string) => void;
   709→  reloadTasks: () => Promise<void>;
   710→}) {
   711→  const stages = task.children || [];
   712→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(() => {
   713→    // Auto-expand first stage or the selected one
   714→    const initial: Record<string, boolean> = {};
   715→    if (stages.length > 0) {
   716→      initial[stages[0].id] = true;
   717→    }
   718→    if (selectedStageId && stages.find(s => s.id === selectedStageId)) {
   719→      initial[selectedStageId] = true;
   720→    }
   721→    return initial;
   722→  });
   723→  const [addingStage, setAddingStage] = useState(false);
   724→  const [newStageTitle, setNewStageTitle] = useState('');
   725→
   726→  // If selectedStageId changes, ensure that stage is expanded (prev-tracker pattern avoids useEffect-setState)
   727→  const [prevSelectedStageId, setPrevSelectedStageId] = useState<string | null>(selectedStageId);
   728→  if (selectedStageId !== prevSelectedStageId) {
   729→    setPrevSelectedStageId(selectedStageId);
   730→    if (selectedStageId && stages.find(s => s.id === selectedStageId) && !expandedStages[selectedStageId]) {
   731→      setExpandedStages(prev => ({ ...prev, [selectedStageId]: true }));
   732→    }
   733→  }
   734→
   735→  const toggleExpand = (stageId: string) => {
   736→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   737→  };
   738→
   739→  const addStage = async () => {
   740→    if (!newStageTitle.trim()) return;
   741→    await fetch('/api/tasks', {
   742→      method: 'POST',
   743→      headers: { 'Content-Type': 'application/json' },
   744→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: task.boardId }),
   745→    });
   746→    setNewStageTitle('');
   747→    setAddingStage(false);
   748→    await reloadTasks();
   749→  };
   750→
   751→  const moveStage = async (index: number, direction: -1 | 1) => {
   752→    const target = index + direction;
   753→    if (target < 0 || target >= stages.length) return;
   754→    const a = stages[index];
   755→    const b = stages[target];
   756→    // Swap hexR values
   757→    await Promise.all([
   758→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: a.id, hexR: b.hexR, hexQ: b.hexQ }) }),
   759→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: b.id, hexR: a.hexR, hexQ: a.hexQ }) }),
   760→    ]);
   761→    await reloadTasks();
   762→  };
   763→
   764→  const deleteStage = async (stageId: string) => {
   765→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   766→    await reloadTasks();
   767→  };
   768→
   769→  const updateStage = async (stageId: string, patch: Record<string, unknown>) => {
   770→    await fetch('/api/tasks', {
   771→      method: 'PUT',
   772→      headers: { 'Content-Type': 'application/json' },
   773→      body: JSON.stringify({ id: stageId, ...patch }),
   774→    });
   775→    await reloadTasks();
   776→  };
   777→
   778→  return (
   779→    <div className="py-2 space-y-1.5">
   780→      {stages.length === 0 && !addingStage && (
   781→        <div className="text-center py-6 px-4">
   782→          <ListChecks className="w-6 h-6 mx-auto mb-2 text-slate-700" />
   783→          <p className="text-[11px] text-slate-500 mb-1">Этапов пока нет</p>
   784→          <p className="text-[10px] text-slate-600 mb-3">Создайте первый этап производства трека</p>
   785→        </div>
   786→      )}
   787→
   788→      {stages.map((stage, idx) => (
   789→        <StageCard
   790→          key={stage.id}
   791→          stage={stage}
   792→          index={idx}
   793→          total={stages.length}
   794→          boardColor={boardColor}
   795→          c={c}
   796→          members={members}
   797→          isExpanded={!!expandedStages[stage.id]}
   798→          isSelected={selectedStageId === stage.id}
   799→          onToggleExpand={() => toggleExpand(stage.id)}
   800→          onSelect={() => onSelectStage(stage.id)}
   801→          onMoveUp={() => moveStage(idx, -1)}
   802→          onMoveDown={() => moveStage(idx, 1)}
   803→          onDelete={() => deleteStage(stage.id)}
   804→          onUpdate={(patch) => updateStage(stage.id, patch)}
   805→          reloadTasks={reloadTasks}
   806→        />
   807→      ))}
   808→
   809→      {/* Add stage form */}
   810→      {addingStage ? (
   811→        <div
   812→          className="mx-2 rounded-lg p-2.5 space-y-1.5"
   813→          style={{ border: `1px dashed ${c.a3}`, backgroundColor: c.a04 }}
   814→        >
   815→          <Input
   816→            value={newStageTitle}
   817→            onChange={(e) => setNewStageTitle(e.target.value)}
   818→            onKeyDown={(e) => {
   819→              if (e.key === 'Enter') void addStage();
   820→              if (e.key === 'Escape') { setAddingStage(false); setNewStageTitle(''); }
   821→            }}
   822→            placeholder="Название этапа..."
   823→            className="bg-slate-900/80 border-slate-700/50 text-[11px] text-slate-200 placeholder:text-slate-600 h-7"
   824→            style={{ borderColor: c.a3 }}
   825→            autoFocus
   826→          />
   827→          <div className="flex items-center gap-1.5">
   828→            <button
   829→              onClick={() => void addStage()}
   830→              disabled={!newStageTitle.trim()}
   831→              className="text-[10px] px-2 py-1 rounded font-medium transition-all disabled:opacity-40"
   832→              style={{ color: '#fff', backgroundColor: c.a5 }}
   833→            >
   834→              + Добавить этап
   835→            </button>
   836→            <button
   837→              onClick={() => { setAddingStage(false); setNewStageTitle(''); }}
   838→              className="text-[10px] text-slate-500 hover:text-slate-300 px-2 py-1"
   839→            >
   840→              Отмена
   841→            </button>
   842→          </div>
   843→        </div>
   844→      ) : (
   845→        <button
   846→          onClick={() => setAddingStage(true)}
   847→          className="cp-add-btn mx-2 w-[calc(100%-1rem)] justify-center"
   848→        >
   849→          <Plus className="w-3 h-3" /> Добавить этап
   850→        </button>
   851→      )}
   852→    </div>
   853→  );
   854→}

   858→function StageCard({

   862→}: {

   864→  index: number;
   865→  total: number;
   866→  boardColor: string;
   867→  c: ColorSet;
   868→  members: GroupMember[];
   869→  isExpanded: boolean;
   870→  isSelected: boolean;

   872→  onSelect: () => void;
   873→  onMoveUp: () => void;
   874→  onMoveDown: () => void;
   875→  onDelete: () => void;
   876→  onUpdate: (patch: Record<string, unknown>) => Promise<void>;

  1335→          <input
  1336→            value={titleDraft}
  1337→            onChange={(e) => setTitleDraft(e.target.value)}
  1338→            onKeyDown={(e) => {
  1339→              if (e.key === 'Enter') { e.preventDefault(); void saveTitle(); }
  1340→              if (e.key === 'Escape') { setIsEditingTitle(false); setTitleDraft(subtask.title); }
  1341→            }}

  1343→            className="flex-1 min-w-0 bg-slate-900/90 rounded px-2 py-0.5 text-[11px] text-slate-100 focus:outline-none"

  1502→    ]);
  1503→    await reloadTasks();
  1504→  };
  1505→

  1507→    await fetch(`/api/tasks?id=${subId}`, { method: 'DELETE' });

  1537→          index={idx}
  1538→          total={subtasks.length}

  1598→function parseAssignees(raw: string | null): string[] {
  1599→  if (!raw) return [];

  1601→    const parsed = JSON.parse(raw);
  1602→    if (Array.isArray(parsed)) return parsed.filter(Boolean);
  1603→    return [raw];
  1604→  } catch {
  1605→    return [raw];
  1606→  }
  1607→}

  1609→function serializeAssignees(ids: string[]): string | null {
  1610→  if (ids.length === 0) return null;
  1611→  return JSON.stringify(ids);
  1612→}

  1778→async function updateSubtask(subId: string, patch: Record<string, unknown>, reloadTasks: () => Promise<void>) {
  1779→  await fetch('/api/tasks', {
  1780→    method: 'PUT',

  1787→async function moveSubtask(stage: TaskChild, idx: number, dir: -1 | 1, reloadTasks: () => Promise<void>) {
  1788→  const subtasks = stage.children || [];
  1789→  const target = idx + dir;
  1790→  if (target < 0 || target >= subtasks.length) return;
  1791→  const a = subtasks[idx];
  1792→  const b = subtasks[target];
  1793→  await Promise.all([
  1794→    fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: a.id, hexR: b.hexR, hexQ: b.hexQ }) }),
  1795→    fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: b.id, hexR: a.hexR, hexQ: a.hexQ }) }),
  1796→  ]);
  1797→  await reloadTasks();
  1798→}

  1800→function getProgress(children: { status: string }[]): number {
  1801→  if (children.length === 0) return 0;
  1802→  const done = children.filter(c => c.status === 'done').length;
  1803→  return Math.round((done / children.length) * 100);
