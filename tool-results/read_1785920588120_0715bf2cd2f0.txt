    96→/* ── Track Detail View (3-level hierarchy) ──────────── */
    97→
    98→const EMOJI_RE = /^(\p{Emoji_Presentation}|\p{Extended_Pictographic})/u;
    99→const DESC_LIMIT = 300;
   100→
   101→function TrackDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   102→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   103→  const [stages, setStages] = useState<TaskChild[]>(task.children || []);
   104→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(
   105→    Object.fromEntries((task.children || []).map(c => [c.id, true]))
   106→  );
   107→  const [expandedDescIds, setExpandedDescIds] = useState<Record<string, boolean>>({});
   108→  const [editingDescId, setEditingDescId] = useState<string | null>(null);
   109→  const [descDraft, setDescDraft] = useState('');
   110→  const [addingGrandchild, setAddingGrandchild] = useState<string | null>(null);
   111→  const [newGcTitle, setNewGcTitle] = useState('');
   112→  const [addingStage, setAddingStage] = useState(false);
   113→  const [newStageTitle, setNewStageTitle] = useState('');
   114→
   115→  const config = task.trackConfig ? JSON.parse(task.trackConfig) : null;
   116→  const allSubtasks = stages.flatMap(s => s.children || []);
   117→  const progress = getProgress(allSubtasks);
   118→  const boardColor = board?.color || '#00d9ff';
   119→  const bc = boardColorStyles(boardColor);
   120→
   121→  const getProgressColor = () => {
   122→    if (progress === 100) return '#10b981';
   123→    if (progress > 50) return boardColor;
   124→    if (progress > 0) return '#f59e0b';
   125→    return '#334155';
   126→  };
   127→  const getProgressTextColor = () => {
   128→    if (progress === 100) return '#34d399';
   129→    if (progress > 50) return boardColor;
   130→    if (progress > 0) return '#f59e0b';
   131→    return '#475569';
   132→  };
   133→
   134→  const reloadTasks = useCallback(async () => {
   135→    if (!selectedBoardId) return;
   136→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   137→    const data = await res.json();
   138→    setBoardTasks(data.tasks);
   139→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   140→    if (updated) setStages(updated.children || []);
   141→  }, [selectedBoardId, setBoardTasks, task.id]);
   142→
   143→  const toggleGrandchildStatus = async (e: React.MouseEvent, id: string) => {
   144→    e.stopPropagation();
   145→    const order: TaskStatus[] = ['todo', 'in-progress', 'review', 'done'];
   146→    const all = stages.flatMap(s => s.children || []);
   147→    const gc = all.find(c => c.id === id);
   148→    if (!gc) return;
   149→    const idx = order.indexOf(gc.status as TaskStatus);
   150→    const next = order[(idx + 1) % order.length];
   151→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id, status: next }) });
   152→    await reloadTasks();
   153→  };
   154→
   155→  const toggleStageExpanded = (stageId: string) => {
   156→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   157→  };
   158→
   159→  const toggleDescExpanded = (id: string) => {
   160→    setExpandedDescIds(prev => ({ ...prev, [id]: !prev[id] }));
   161→  };
   162→
   163→  const startEditDesc = (id: string, current: string) => {
   164→    setEditingDescId(id);
   165→    setDescDraft(current || '');
   166→  };
   167→
   168→  const saveDesc = async (id: string) => {
   169→    await fetch('/api/tasks', {
   170→      method: 'PUT',
   171→      headers: { 'Content-Type': 'application/json' },
   172→      body: JSON.stringify({ id, description: descDraft.trim() || null }),
   173→    });
   174→    setEditingDescId(null);
   175→    setDescDraft('');
   176→    await reloadTasks();
   177→  };
   178→
   179→  const cancelEditDesc = () => {
   180→    setEditingDescId(null);
   181→    setDescDraft('');
   182→  };
   183→
   184→  const deleteGrandchild = async (e: React.MouseEvent, id: string) => {
   185→    e.stopPropagation();
   186→    await fetch(`/api/tasks?id=${id}`, { method: 'DELETE' });
   187→    await reloadTasks();
   188→  };
   189→
   190→  const addGrandchild = async (stageId: string) => {
   191→    if (!newGcTitle.trim()) return;
   192→    await fetch('/api/tasks', {
   193→      method: 'POST',
   194→      headers: { 'Content-Type': 'application/json' },
   195→      body: JSON.stringify({ title: newGcTitle.trim(), parentId: stageId }),
   196→    });
   197→    setNewGcTitle('');
   198→    setAddingGrandchild(null);
   199→    await reloadTasks();
   200→  };
   201→
   202→  const addStage = async () => {
   203→    if (!newStageTitle.trim()) return;
   204→    await fetch('/api/tasks', {
   205→      method: 'POST',
   206→      headers: { 'Content-Type': 'application/json' },
   207→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: selectedBoardId }),
   208→    });
   209→    setNewStageTitle('');
   210→    setAddingStage(false);
   211→    await reloadTasks();
   212→  };
   213→
   214→  const deleteStage = async (e: React.MouseEvent, stageId: string) => {
   215→    e.stopPropagation();
   216→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   217→    await reloadTasks();
   218→  };
   219→
   220→  const handleDelete = async () => {
   221→    await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   222→    setSelectedTaskId(null);
   223→    if (selectedBoardId) {
   224→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   225→      const data = await res.json();
   226→      setBoardTasks(data.tasks);
   227→    }
   228→  };
   229→
   230→  return (
   231→    <div className="flex-1 overflow-y-auto">
   232→      {/* Header */}
   233→      <div className="border-b border-slate-800/50 p-4">
   234→        <div className="flex items-center gap-1.5 mb-2">
   235→          <span className="text-[9px] px-1.5 py-0.5 rounded font-medium uppercase tracking-wider"
   236→            style={{ color: boardColor, backgroundColor: boardColor + '15' }}>
   237→            Трек
   238→          </span>
   239→          <div className="flex-1" />
   240→          <button onClick={() => setEditingTask(task)} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors">
   241→            <Pencil className="w-3.5 h-3.5" />
   242→          </button>
   243→          <button onClick={handleDelete} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors">
   244→            <Trash2 className="w-3.5 h-3.5" />
   245→          </button>
   246→        </div>
   247→        <h3 className="text-sm font-semibold text-white leading-tight mb-1.5">{task.title}</h3>
   248→        {task.description && (
   249→          <p className="text-[11px] text-slate-500 mb-1.5 line-clamp-2">{task.description}</p>
   250→        )}
   251→        {config && (
   252→          <div className="flex flex-wrap gap-1">
   253→            {config.instruments?.map((inst: string) => (
   254→              <span key={inst} className="text-[9px] bg-slate-800/80 text-slate-400 px-1.5 py-0.5 rounded">{inst}</span>
   255→            ))}
   256→          </div>
   257→        )}
   258→      </div>
   259→
   260→      {/* Progress */}
   261→      <div className="border-b border-slate-800/30 px-4 py-3">
   262→        <div className="flex items-center justify-between mb-1.5">
   263→          <span className="text-[11px] text-slate-400 font-medium">Общий прогресс</span>
   264→          <span className="text-[11px] font-bold" style={{ color: getProgressTextColor() }}>{progress}%</span>
   265→        </div>
   266→        <div className="relative h-2 bg-slate-800/80 rounded-full overflow-hidden">
   267→          <div className="h-full rounded-full transition-all duration-700 ease-out" style={{ width: `${progress}%`, backgroundColor: getProgressColor() }} />
   268→          {progress > 0 && progress < 100 && (
   269→            <div
   270→              className="absolute inset-0 rounded-full animate-pulse opacity-20"
   271→              style={{
   272→                background: `linear-gradient(90deg, transparent, ${getProgressColor()}, transparent)`,
   273→              }}
   274→            />
   275→          )}
   276→        </div>
   277→        <div className="flex items-center justify-between mt-1">
   278→          <span className="text-[9px] text-slate-600">{allSubtasks.filter(s => s.status === 'done').length} из {allSubtasks.length} задач</span>
   279→          {progress === 100 && <span className="text-[9px] text-emerald-400 font-medium">✓ Завершено</span>}
   280→        </div>
   281→      </div>
   282→
   283→      {/* Stages as collapsible groups */}
   284→      <div className="px-3 py-2 space-y-1">
   285→        {stages.map((stage) => {
   286→          const grandchildren = stage.children || [];
   287→          const stageProgress = getProgress(grandchildren);
   288→          const stageDone = stage.status === 'done';
   289→          const isExpanded = expandedStages[stage.id] !== false;
   290→          const isCustom = !EMOJI_RE.test(stage.title);
   291→          const StatusIcon = SUBTASK_STATUS_ICON[stage.status] || Circle;
   292→          const hasDesc = !!stage.description;
   293→          const descExpanded = expandedDescIds[stage.id] === true;
   294→          const isEditingDesc = editingDescId === stage.id;
   295→
   296→          return (
   297→            <div key={stage.id} className="rounded-lg border border-slate-800/30 overflow-hidden">
   298→              {/* Stage header — full panel click to expand */}
   299→              <button
   300→                onClick={() => {
   301→                  useKanbanStore.getState().setSelectedStageForPanel({ taskId: task.id, stageId: stage.id });
   302→                  toggleStageExpanded(stage.id);
   303→                }}
   304→                className="w-full flex items-center gap-2 px-2.5 py-2 hover:bg-slate-900/60 transition-colors text-left group/stage"
   305→              >
   306→                <StatusIcon className={cn('w-3.5 h-3.5 flex-shrink-0', SUBTASK_STATUS_COLOR[stage.status])} />
   307→                {isCustom && <Cat className="w-3 h-3 flex-shrink-0 text-amber-400/60" />}
   308→                <span className={cn('flex-1 text-[11px] font-medium min-w-0',
   309→                  stageDone ? 'text-slate-500 line-through' : 'text-slate-300'
   310→                )} onDoubleClick={(e) => { e.stopPropagation(); useKanbanStore.getState().setDescriptionEditorItem({ id: stage.id, title: stage.title, description: stage.description, parentId: task.id, parentTitle: task.title }); }}>
   311→                  {stage.title}
   312→                </span>
   313→                {/* Description icon */}
   314→                <button
   315→                  onClick={(e) => { e.stopPropagation(); if (hasDesc) toggleDescExpanded(stage.id); else startEditDesc(stage.id, stage.description || ''); }}
   316→                  className={cn('flex-shrink-0 transition-opacity',
   317→                    hasDesc ? '' : ''
   318→                  )}
   319→                  style={{ color: hasDesc ? '#64748b' : 'transparent' }}
   320→                  onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.color = boardColor; (e.currentTarget as HTMLElement).style.opacity = '1'; }}
   321→                  onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.color = hasDesc ? '#64748b' : 'transparent'; (e.currentTarget as HTMLElement).style.opacity = hasDesc ? '1' : '0'; }}
   322→                >
   323→                  <MessageSquare className="w-3 h-3" />
   324→                </button>
   325→                {grandchildren.length > 0 && (
   326→                  <span className="text-[9px] text-slate-600 mr-1">{stageDone ? '✓' : `${stageProgress}%`}</span>
   327→                )}
   328→                <button
   329→                  onClick={(e) => { e.stopPropagation(); deleteStage(e, stage.id); }}
   330→                  className="flex-shrink-0 p-0.5 rounded opacity-0 group-hover/stage:opacity-100 hover:bg-rose-500/10 text-slate-600 hover:text-rose-400 transition-all"
   331→                >
   332→                  <Trash2 className="w-3 h-3" />
   333→                </button>
   334→                {isExpanded
   335→                  ? <ChevronUp className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   336→                  : <ChevronDown className="w-3 h-3 text-slate-600 group-hover/stage:text-slate-400" />
   337→                }
   338→              </button>
   339→
   340→              {/* Expanded content */}
   341→              {isExpanded && (
   342→                <div>
   343→                  {/* Description card */}
   344→                  {hasDesc && !isEditingDesc && (
   345→                    <div className="mx-2.5 mb-1.5">
   346→                      <div className="bg-slate-900/60 border border-slate-800/40 rounded-md p-2">
   347→                        {descExpanded ? (
   348→                          <>
   349→                            <p className="text-[10px] text-slate-400 whitespace-pre-wrap leading-relaxed">
   350→                              {stage.description && stage.description.length > DESC_LIMIT
   351→                                ? stage.description.slice(0, DESC_LIMIT) + '...'
   352→                                : stage.description}
   353→                            </p>
   354→                            <div className="flex items-center gap-2 mt-1.5">
   355→                              <button
   356→                                onClick={() => startEditDesc(stage.id, stage.description || '')}
   357→                                className="text-[9px] transition-colors"
   358→                                style={{ color: boardColor + '80' }}
   359→                                onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.color = boardColor; }}
   360→                                onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.color = boardColor + '80'; }}
   361→                              >Редактировать</button>
   362→                              <button
   363→                                onClick={() => toggleDescExpanded(stage.id)}
   364→                                className="text-[9px] text-slate-600 hover:text-slate-400 transition-colors"
   365→                              >Свернуть</button>
   366→                            </div>
   367→                          </>
   368→                        ) : (
   369→                          <p className="text-[10px] text-slate-500 line-clamp-2">
   370→                            {stage.description}
   371→                          </p>
   372→                        )}
   373→                      </div>
   374→                    </div>
   375→                  )}
   376→                  {/* Editing description */}
   377→                  {isEditingDesc && (
   378→                    <div className="mx-2.5 mb-1.5">
   379→                      <div className="bg-slate-900/60 rounded-md p-2 space-y-1.5" style={{ border: `1px solid ${boardColor}30` }}>
   380→                        <Textarea
   381→                          value={descDraft}
   382→                          onChange={(e) => setDescDraft(e.target.value)}
   383→                          placeholder="Описание этапа..."
   384→                          maxLength={DESC_LIMIT}
   385→                          className="bg-slate-800/80 border-slate-700/50 text-[10px] text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none"
   386→                          autoFocus
   387→                        />
   388→                        <div className="flex items-center justify-between">
   389→                          <span className="text-[8px] text-slate-600">{descDraft.length}/{DESC_LIMIT}</span>
   390→                          <div className="flex items-center gap-1">
   391→                            <button onClick={cancelEditDesc} className="text-[9px] text-slate-500 hover:text-slate-300 px-1.5 py-0.5 rounded hover:bg-slate-800 transition-colors">Отмена</button>
   392→                            <button
   393→                              onClick={() => saveDesc(stage.id)}
   394→                              className="text-[9px] px-1.5 py-0.5 rounded transition-colors"
   395→                              style={{ color: boardColor, backgroundColor: boardColor + '10' }}
   396→                              onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.backgroundColor = boardColor + '20'; }}
   397→                              onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.backgroundColor = boardColor + '10'; }}
   398→                            >Сохранить</button>
   399→                          </div>
   400→                        </div>
   401→                      </div>
   402→                    </div>
   403→                  )}
   404→
   405→                  {/* Progress bar */}
   406→                  {grandchildren.length > 0 && (
   407→                    <div className="px-2.5 pb-1.5">
   408→                      <div className="h-1 bg-slate-800/60 rounded-full overflow-hidden">
   409→                        <div
   410→                          className="h-full rounded-full transition-all duration-500"
   411→                          style={{ width: `${stageProgress}%`, backgroundColor: stageProgress === 100 ? '#10b981' : stageProgress > 50 ? boardColor : stageProgress > 0 ? '#f59e0b' : '#334155' }}
   412→                        />
   413→                      </div>
   414→                    </div>
   415→                  )}
   416→
   417→                  {/* Grandchildren list */}
   418→                  <div className="space-y-0.5 px-2 pb-2">
   419→                    {grandchildren.map((gc) => {
   420→                      const GcIcon = SUBTASK_STATUS_ICON[gc.status] || Circle;
   421→                      const gcHasDesc = !!gc.description;
   422→                      const gcDescExpanded = expandedDescIds[gc.id] === true;
   423→                      const gcIsEditingDesc = editingDescId === gc.id;
   424→                      return (
   425→                        <div key={gc.id} className="rounded-md border border-transparent hover:border-slate-800/30 overflow-hidden">
   426→                          <div
   427→                            className="flex items-center gap-2 px-2 py-1 rounded-md hover:bg-slate-800/40 transition-colors group/gc cursor-pointer"
   428→                            onClick={() => {
   429→                              useKanbanStore.getState().setSelectedStageForPanel({
   430→                                taskId: task.id, stageId: stage.id,
   431→                              });
   432→                            }}
   433→                          >
   434→                            <button
   435→                              onClick={(e) => toggleGrandchildStatus(e, gc.id)}
   436→                              className={cn('flex-shrink-0 hover:opacity-70 transition-opacity', SUBTASK_STATUS_COLOR[gc.status] || 'text-slate-500')}
   437→                            >
   438→                              <GcIcon className="w-3 h-3" />
   439→                            </button>
   440→                            <span className={cn('flex-1 text-[10px] min-w-0',
   441→                              gc.status === 'done' ? 'text-slate-600 line-through' : 'text-slate-400'
   442→                            )}>
   443→                              {gc.title}
   444→                            </span>
   445→                            {/* GC description icon */}
   446→                            <button
   447→                              onClick={(e) => { e.stopPropagation(); if (gcHasDesc) toggleDescExpanded(gc.id); else startEditDesc(gc.id, gc.description || ''); }}
   448→                              className={cn('flex-shrink-0 transition-opacity',
   449→                                gcHasDesc ? '' : ''

   771→function TaskForm({ mode, task, boardColor }: { mode: 'create' | 'edit'; task?: Task; boardColor?: string }) {

   779→  const [category, setCategory] = useState<TaskCategory>('general');
   780→  const [deadline, setDeadline] = useState<string | null>(null);

   782→
   783→  useEffect(() => {
   784→    if (mode === 'edit' && task) {
   785→      setTitle(task.title); setDescription(task.description || ''); setStatus(task.status);
   786→      setPriority(task.priority); setAssignee(task.assignee || ''); setCategory(task.category);
   787→      setDeadline(task.deadline || null);
   788→    }
   789→  }, [mode, task]);
   790→
   791→  const reloadTasks = useCallback(async () => {
   792→    if (selectedBoardId) {
   793→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   794→      const data = await res.json();
   795→      setBoardTasks(data.tasks);
   796→    }
   797→    if (selectedProjectId) {
   798→      const bRes = await fetch(`/api/boards?projectId=${selectedProjectId}`);
   799→      const bData = await bRes.json();
   800→      useKanbanStore.getState().setBoards(bData.boards);
   801→    }

   803→
   804→  const handleSave = async () => {
   805→    if (!title.trim()) return;
   806→    setSaving(true);
   807→    try {
   808→      if (mode === 'edit' && task) {
   809→        await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: task.id, title, description: description || null, status, priority, assignee: assignee || null, category, boardId: selectedBoardId, deadline }) });
   810→      } else {
   811→        await fetch('/api/tasks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title, description: description || null, status, priority, assignee: assignee || null, category, parentId: null, boardId: selectedBoardId, deadline }) });
   812→      }
   813→      await reloadTasks(); setEditingTask(null); setIsCreating(false);
   814→      if (mode === 'edit' && task) setSelectedTaskId(task.id);
   815→    } finally { setSaving(false); }
   816→  };
   817→
   818→  const handleDelete = async () => {
   819→    if (!task) return;
   820→    setSaving(true);
   821→    try {
   822→      await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   823→      await reloadTasks(); setEditingTask(null); setSelectedTaskId(null);
   824→    } finally { setSaving(false); }
   825→  };
   826→
   827→  return (
   828→    <div className="flex-1 overflow-y-auto">
   829→      <div className="p-4 space-y-3">
   830→        <div className="flex items-center justify-between">
   831→          <h3 className="text-xs font-semibold uppercase tracking-wider" style={{ color }}>{mode === 'edit' ? 'Редактировать' : 'Новая задача'}</h3>
   832→          <button onClick={() => { setEditingTask(null); setIsCreating(false); }} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300"><X className="h-3.5 w-3.5" /></button>
   833→        </div>
   834→        <div className="space-y-2.5">
   835→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Название</Label><Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Задача..." className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-8" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   836→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Описание</Label><Textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Описание..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   837→          <div className="grid grid-cols-2 gap-2">
   838→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Статус</Label><Select value={status} onValueChange={(v) => setStatus(v as TaskStatus)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{STATUSES.map((s) => (<SelectItem key={s.value} value={s.value}><span className={s.color}>{s.label}</span></SelectItem>))}</SelectContent></Select></div>
   839→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Приоритет</Label><Select value={priority} onValueChange={(v) => setPriority(v as TaskPriority)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{PRIORITIES.map((p) => (<SelectItem key={p.value} value={p.value}><div className="flex items-center gap-1.5"><div className={cn('w-1.5 h-1.5 rounded-full', p.dot)} />{p.label}</div></SelectItem>))}</SelectContent></Select></div>
   840→          </div>
   841→          <div className="grid grid-cols-2 gap-2">
   842→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Категория</Label><Select value={category} onValueChange={(v) => setCategory(v as TaskCategory)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{CATEGORIES.map((c) => (<SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>))}</SelectContent></Select></div>
   843→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Ответственный</Label><Input value={assignee} onChange={(e) => setAssignee(e.target.value)} placeholder="Имя..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 placeholder:text-slate-600 h-8" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   844→          </div>
   845→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Дедлайн</Label><DeadlinePicker value={deadline} onChange={setDeadline} size="md" /></div>
   846→        </div>
   847→        <div className="flex gap-2 pt-1">
   848→          <Button onClick={handleSave} disabled={!title.trim() || saving} className="flex-1 text-white h-8 text-xs" style={{ backgroundColor: color, boxShadow: `0 4px 14px ${color}30` }}><Save className="w-3 h-3 mr-1" />{saving ? '...' : mode === 'edit' ? 'Сохранить' : 'Создать'}</Button>
   849→          {mode === 'edit' && <Button onClick={handleDelete} variant="destructive" disabled={saving} className="h-8 text-xs">Удалить</Button>}
   850→        </div>
   851→      </div>
   852→    </div>
   853→  );
   854→}
   855→
   856→function getProgress(children: { status: string }[]): number {
   857→  if (children.length === 0) return 0;
   858→  return Math.round((children.filter(c => c.status === 'done').length / children.length) * 100);
   859→}
