   245→// Build a nested tree from flat comment list
   246→interface CommentNode extends Comment {
   247→  replies: CommentNode[];
   248→}
   249→
   250→function buildCommentTree(
   251→  comments: Comment[],
   252→  rootComparator?: (a: CommentNode, b: CommentNode) => number
   253→): CommentNode[] {
   254→  const map = new Map<string, CommentNode>();
   255→  const roots: CommentNode[] = [];
   256→  
   257→  for (const c of comments) {
   258→    map.set(c.id, { ...c, replies: [] });
   259→  }
   260→  for (const c of comments) {
   261→    const node = map.get(c.id)!;
   262→    if (c.parentId && map.has(c.parentId)) {
   263→      map.get(c.parentId)!.replies.push(node);
   264→    } else {
   265→      roots.push(node);
   266→    }
   267→  }
   268→  // Sort root comments — caller may pass a custom comparator (e.g. by date/author/status).
   269→  // Default: ascending by timestampMs (oldest first) — preserves the original behavior.
   270→  if (rootComparator) {
   271→    roots.sort(rootComparator);
   272→  } else {
   273→    roots.sort((a, b) => a.timestampMs - b.timestampMs);
   274→  }
   275→  for (const r of roots) {
   276→    r.replies.sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   277→  }
   278→  return roots;
   279→}

   331→function formatTimestamp(ms: number): string {
   332→  const totalSeconds = ms / 1000;
   333→  const minutes = Math.floor(totalSeconds / 60);
   334→  const seconds = Math.floor(totalSeconds % 60);
   335→  const milliseconds = Math.floor((totalSeconds % 1) * 10);
   336→  return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}.${milliseconds}`;
   337→}
   338→
   339→function formatDuration(seconds: number): string {
   340→  const mins = Math.floor(seconds / 60);
   341→  const secs = Math.floor(seconds % 60);
   342→  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   343→}

   345→function getInitials(name?: string | null): string {
   346→  if (!name) return '?';
   347→  return name
   348→    .trim()
   349→    .split(' ')
   350→    .map((n) => n[0])
   351→    .join('')
   352→    .toUpperCase()
   353→    .slice(0, 2);
   354→}

   356→// Recursively flatten ALL descendants of a task tree.
   357→// The backend returns tasks with up to 3 levels of `children` (task → child → grandchild),
   358→// but this helper is generic so it works for any depth. Used by the track + project
   359→// progress panels to count every subtask regardless of how deep it nests.
   360→function countAllDescendants(tasks: { children?: unknown[] }[]): any[] {
   361→  const result: any[] = [];
   362→  const walk = (nodes: unknown[]) => {
   363→    for (const n of nodes as any[]) {
   364→      result.push(n);
   365→      if (n && Array.isArray(n.children) && n.children.length > 0) {
   366→        walk(n.children);
   367→      }
   368→    }
   369→  };
   370→  walk(tasks as unknown[]);
   371→  return result;
   372→}
