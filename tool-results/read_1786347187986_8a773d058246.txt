    53→/* ── Helpers ──────────────────────────────────────────── */
    54→
    55→function getProgress(children: { status: string }[] | undefined): number {
    56→  if (!children || children.length === 0) return 0;
    57→  const done = children.filter(c => c.status === 'done').length;
    58→  return Math.round((done / children.length) * 100);
    59→}
    60→
    61→interface CountableTask {
    62→  status: string;
    63→  children?: CountableTask[] | unknown;
    64→}
    65→
    66→function countAll(tasks: CountableTask[]): { total: number; done: number } {
    67→  let total = 0;
    68→  let done = 0;
    69→  for (const t of tasks) {
    70→    total++;
    71→    if (t.status === 'done') done++;
    72→    if (Array.isArray(t.children) && t.children.length > 0) {
    73→      const sub = countAll(t.children as CountableTask[]);
    74→      total += sub.total;
    75→      done += sub.done;
    76→    }
    77→  }
    78→  return { total, done };
    79→}
    80→
    81→function formatDeadline(value: string): string {
    82→  const d = new Date(value);
    83→  if (isNaN(d.getTime())) return value;
    84→  const day = String(d.getDate()).padStart(2, '0');
    85→  const months = ['янв', 'фев', 'мар', 'апр', 'май', 'июн', 'июл', 'авг', 'сен', 'окт', 'ноя', 'дек'];
    86→  const month = months[d.getMonth()] || '';
    87→  const year = d.getFullYear();
    88→  return `${day} ${month} ${year}`;
    89→}

   178→    doneTasks += c.done;
