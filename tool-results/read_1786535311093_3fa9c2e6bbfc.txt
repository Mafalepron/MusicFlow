    71→/* ─── Waveform Progress Bar — reusable animated audio waveform ─── */
    72→/* progress: 0-100 (completion %). accentColor: hex color.              */
    73→/* On hover: playhead sweeps 0%→progress%, filled bars equalize-bounce.  */
    74→/* 0% edge case: waveform stays static + dim, no animation triggers.     */
    75→function WaveformProgressBar({ progress, accentColor, height = 40, bars: barCount = 32 }: {
    76→  progress: number;
    77→  accentColor: string;
    78→  height?: number;
    79→  bars?: number;
    80→}) {
    81→  const [hovered, setHovered] = useState(false);
    82→  const [h, setH] = useState(false);
    83→  // Clamp + normalize
    84→  const pct = Math.max(0, Math.min(100, Math.round(progress)));
    85→  const hasProgress = pct > 0;
    86→  // Deterministic waveform shape (pseudo-random from accentColor + bars count for stability)
    87→  const waveBars = useMemo(() => {
    88→    const seed = accentColor.split('').reduce((a, c) => a + c.charCodeAt(0), 0) + barCount * 17;
    89→    return Array.from({ length: barCount }, (_, i) => {
    90→      const base = 0.45 + 0.35 * Math.sin((i / barCount) * Math.PI * 4 + (seed % 7));
    91→      const harm = 0.15 * Math.sin((i / barCount) * Math.PI * 11 + (seed % 13));
    92→      const noise = ((seed * (i + 3) * 7) % 23) / 100 - 0.1;
    93→      return Math.max(0.15, Math.min(0.95, base + harm + noise));
    94→    });
    95→  }, [accentColor, barCount]);
    96→
    97→  return (
    98→    <div
    99→      className="relative overflow-hidden"
   100→      style={{
   101→        height: `${height}px`,
   102→        background: 'rgba(0,0,0,0.4)',
   103→        borderRadius: '2px',
   104→        border: `0.5px solid ${hexToRgba(accentColor, 0.2)}`,
   105→      }}
   106→      onMouseEnter={() => { setHovered(true); setH(true); }}
   107→      onMouseLeave={() => { setHovered(false); setH(false); }}
   108→    >
   109→      {/* Center axis line */}
   110→      <div className="absolute left-0 right-0 top-1/2 h-px pointer-events-none" style={{ background: hexToRgba(accentColor, 0.15) }} />
   111→
   112→      {/* Waveform bars */}
   113→      <div className="absolute inset-0 flex items-center justify-between px-1 pointer-events-none">
   114→        {waveBars.map((v, i) => {
   115→          // Bar is "filled" if it's within the progress region (left→right)
   116→          const barProgressPct = ((i + 1) / barCount) * 100;
   117→          const isFilled = hasProgress && barProgressPct <= pct;
   118→          // On hover with progress, bars up to pct animate; bars beyond pct stay muted
   119→          const shouldAnimate = hovered && hasProgress && isFilled;
   120→          return (
   121→            <div
   122→              key={i}
   123→              style={{
   124→                width: '2px',
   125→                height: `${Math.round(v * 100)}%`,
   126→                background: isFilled ? accentColor : hexToRgba(accentColor, 0.18),
   127→                opacity: isFilled ? (h ? 1 : 0.75) : 0.5,
   128→                boxShadow: isFilled ? `0 0 3px ${hexToRgba(accentColor, 0.7)}` : 'none',
   129→                transformOrigin: 'center',
   130→                borderRadius: '0.5px',
   131→                // CSS variable for equalizer base height (so keyframe can scale around it)
   132→                ['--kb5-base' as string]: v,
   133→                animation: shouldAnimate ? `kb5-eq-bounce ${0.9 + (i % 5) * 0.18}s ease-in-out ${(i * 0.05).toFixed(2)}s infinite` : 'none',
   134→                transition: 'opacity 280ms ease, background 280ms ease, box-shadow 280ms ease',
   135→              }}
   136→            />
   137→          );
   138→        })}
   139→      </div>
   140→
   141→      {/* Progress divider marker (vertical line at progress boundary) — only if hasProgress */}
   142→      {hasProgress && (
   143→        <div
   144→          className="absolute inset-y-0 pointer-events-none"
   145→          style={{
   146→            left: `${pct}%`,
   147→            width: '1px',
   148→            background: accentColor,
   149→            boxShadow: `0 0 6px ${accentColor}, 0 0 3px ${accentColor}`,
   150→            opacity: h ? 1 : 0.6,
   151→            transition: 'opacity 220ms ease',
   152→          }}
   153→        />
   154→      )}
   155→
   156→      {/* Playhead sweep — only animates on hover AND when there's progress */}
   157→      {hovered && hasProgress && (
   158→        <div
   159→          className="absolute inset-y-0 pointer-events-none"
   160→          style={{
   161→            width: '24px',
   162→            background: `linear-gradient(90deg, transparent, ${hexToRgba(accentColor, 0.3)} 40%, ${hexToRgba('#ffffff', 0.4)} 50%, ${hexToRgba(accentColor, 0.3)} 60%, transparent)`,
   163→            ['--kb5-progress' as string]: `${pct}%`,
   164→            animation: 'kb5-playhead-sweep 1.6s ease-out',
   165→            boxShadow: `0 0 10px ${hexToRgba(accentColor, 0.5)}`,
   166→          }}
   167→        >
   168→          {/* Playhead vertical line */}
   169→          <div className="absolute inset-y-0 left-1/2 w-px" style={{ background: '#ffffff', boxShadow: `0 0 6px ${accentColor}` }} />
   170→        </div>
   171→      )}
   172→
   173→      {/* Progress percentage label (top-right) */}
   174→      <div
   175→        className="absolute top-0.5 right-1.5 text-[9px] font-extrabold tabular-nums font-mono pointer-events-none"
   176→        style={{
   177→          color: hasProgress ? accentColor : hexToRgba(accentColor, 0.4),
   178→          textShadow: hasProgress ? `0 0 4px ${hexToRgba(accentColor, 0.5)}` : 'none',
   179→          opacity: h ? 1 : 0.7,
   180→          transition: 'opacity 220ms ease',
   181→        }}
   182→      >
   183→        {pct}%
   184→      </div>
   185→    </div>
   186→  );
   187→}

   926→function Carousel({ children }: { children: React.ReactNode }) {
   927→  const scrollRef = useRef<HTMLDivElement>(null);
   928→  const [canLeft, setCanLeft] = useState(false);
   929→  const [canRight, setCanRight] = useState(false);
   930→
   931→  const updateArrows = useCallback(() => {
   932→    const el = scrollRef.current;
   933→    if (!el) return;
   934→    setCanLeft(el.scrollLeft > 8);
   935→    setCanRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 8);
   936→  }, []);
   937→
   938→  useEffect(() => {
   939→    updateArrows();
   940→    const el = scrollRef.current;
   941→    if (!el) return;
   942→    el.addEventListener('scroll', updateArrows, { passive: true });
   943→    window.addEventListener('resize', updateArrows);
   944→    return () => {
   945→      el.removeEventListener('scroll', updateArrows);
   946→      window.removeEventListener('resize', updateArrows);
   947→    };
   948→  }, [updateArrows]);
   949→
   950→  const scroll = (dir: 'left' | 'right') => {
   951→    const el = scrollRef.current;
   952→    if (!el) return;
   953→    const amount = el.clientWidth * 0.7;
   954→    el.scrollBy({ left: dir === 'left' ? -amount : amount, behavior: 'smooth' });
   955→  };
   956→
   957→  return (
   958→    <div className="relative group/carousel" style={{ padding: '0 44px' }}>
   959→      {/* Left arrow — custom chamfered frame, always active */}
   960→      <button
   961→        onClick={() => scroll('left')}

   970→          opacity: 0.9,
   971→        }}
   972→        onMouseEnter={(e) => { e.currentTarget.style.opacity = '1'; e.currentTarget.style.transform = 'translateY(-50%)'; }}
   973→        onMouseLeave={(e) => { e.currentTarget.style.opacity = '0.9'; e.currentTarget.style.transform = 'translateY(-50%)'; }}
   974→        aria-label="Прокрутить влево"
   975→      >
   976→        <ChevronLeft className="w-4 h-4" />
   977→      </button>
   978→
   979→      {/* Scrollable container — hidden scrollbar */}
   980→      <div
   981→        ref={scrollRef}
   982→        className="flex gap-3 overflow-x-auto py-1"
   983→        style={{
   984→          scrollbarWidth: 'none',
   985→          msOverflowStyle: 'none',
   986→        }}
   987→      >
   988→        {children}
   989→      </div>
   990→
   991→      {/* Right arrow — custom chamfered frame, always active */}
   992→      <button
   993→        onClick={() => scroll('right')}
   994→        className="absolute right-0 top-1/2 -translate-y-1/2 z-10 flex h-9 w-9 items-center justify-center transition-all duration-200"
   995→        style={{
   996→          background: '#161a24',
   997→          border: '1px solid #00a8c6',
   998→          borderRadius: '4px',
   999→          boxShadow: '0 0 8px rgba(0,168,198,0.25)',
  1000→          color: '#00a8c6',
  1001→          cursor: 'pointer',
  1002→          opacity: 0.9,
  1003→        }}

  1005→        onMouseLeave={(e) => { e.currentTarget.style.opacity = '0.9'; e.currentTarget.style.transform = 'translateY(-50%)'; }}
