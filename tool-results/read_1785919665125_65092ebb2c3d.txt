   116→// Build a nested tree from flat comment list
   117→interface CommentNode extends Comment {
   118→  replies: CommentNode[];
   119→}
   120→
   121→function buildCommentTree(comments: Comment[]): CommentNode[] {
   122→  const map = new Map<string, CommentNode>();
   123→  const roots: CommentNode[] = [];
   124→  
   125→  for (const c of comments) {
   126→    map.set(c.id, { ...c, replies: [] });
   127→  }
   128→  for (const c of comments) {
   129→    const node = map.get(c.id)!;
   130→    if (c.parentId && map.has(c.parentId)) {
   131→      map.get(c.parentId)!.replies.push(node);
   132→    } else {
   133→      roots.push(node);
   134→    }
   135→  }
   136→  roots.sort((a, b) => a.timestampMs - b.timestampMs);
   137→  for (const r of roots) {
   138→    r.replies.sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   139→  }
   140→  return roots;
   141→}

   162→function formatTimestamp(ms: number): string {
   163→  const totalSeconds = ms / 1000;
   164→  const minutes = Math.floor(totalSeconds / 60);
   165→  const seconds = Math.floor(totalSeconds % 60);
   166→  const milliseconds = Math.floor((totalSeconds % 1) * 10);
   167→  return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}.${milliseconds}`;
   168→}
   169→
   170→function formatDuration(seconds: number): string {
   171→  const mins = Math.floor(seconds / 60);
   172→  const secs = Math.floor(seconds % 60);
   173→  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   174→}

   176→function getInitials(name?: string | null): string {
   177→  if (!name) return '?';
   178→  return name
   179→    .trim()
   180→    .split(' ')
   181→    .map((n) => n[0])
   182→    .join('')
   183→    .toUpperCase()
   184→    .slice(0, 2);
   185→}

   910→  // Parse comment references: [Comment #N @ MM:SS.s] → clickable badge that highlights comment
   911→  function renderChatText(text: string) {
   912→    // Split on both patterns: comment links (with optional version) and timestamp links
   913→    const parts = text.split(/(\[Comment #\d+ @ \d{2}:\d{2}\.\d(?: \| v\d+)?\]|\[\d{2}:\d{2}\.\d\])/g);
   914→    return parts.map((part, i) => {
   915→      // Check for comment link pattern: [Comment #N @ MM:SS.s] or [Comment #N @ MM:SS.s | vN]
   916→      const commentMatch = part.match(/^\[Comment #(\d+) @ (\d{2}):(\d{2})\.(\d)(?: \| (v\d+))?\]$/);
   917→      if (commentMatch) {
   918→        const commentNum = parseInt(commentMatch[1], 10);
   919→        const mins = parseInt(commentMatch[2], 10);
   920→        const secs = parseInt(commentMatch[3], 10);
   921→        const tenths = parseInt(commentMatch[4], 10);
   922→        const ms = (mins * 60 + secs) * 1000 + tenths * 100;
   923→        const versionLabel = commentMatch[5] || null;
   924→        return (
   925→          <button
   926→            key={i}
   927→            className="inline-flex items-center gap-1 rounded bg-[#FB923C]/15 px-2 py-0.5 text-xs font-semibold text-[#FB923C] hover:bg-[#FB923C]/25 transition-colors"
   928→            onClick={() => {
   929→              // Find the comment by number across ALL versions (roots sorted by timestamp)
   930→              const allRoots = comments.filter((c) => !c.parentId);
   931→              const sortedRoots = [...allRoots].sort((a, b) => a.timestampMs - b.timestampMs || a.createdAt.localeCompare(b.createdAt));
   932→              // Build per-version number maps
   933→              const target = sortedRoots.find((c) => {
   934→                const vComments = comments.filter((cc) => cc.versionId === c.versionId);
   935→                const vRoots = vComments.filter((cc) => !cc.parentId);
   936→                const idx = vRoots.indexOf(c);
   937→                return idx + 1 === commentNum;
   938→              });
   939→              if (!target) return;
   940→              const targetVersion = versions.find((v) => v.id === target.versionId);
   941→              // If comment is on a different version, show switch dialog
   942→              if (targetVersion && activeVersion && target.versionId !== activeVersion.id) {
   943→                setPendingVersionSwitch({
   944→                  commentId: target.id,
   945→                  versionId: targetVersion.id,
   946→                  versionLabel: `v${targetVersion.version}` + (targetVersion.label ? ` (${targetVersion.label})` : ''),
   947→                  timestampMs: target.timestampMs,
   948→                });
   949→                return;
   950→              }
   951→              // Same version or no version mismatch — seek and focus
   952→              seekTo(ms / 1000);
   953→              setFocusedCommentId(target.id);
   954→              setTimeout(() => setFocusedCommentId(null), 4000);
   955→            }}
   956→          >
   957→            <Send className="h-3 w-3 rotate-[-30deg]" />
   958→            #{commentNum}{versionLabel && <span className="ml-0.5 opacity-60">{versionLabel}</span>}
   959→          </button>
   960→        );
   961→      }
   962→      // Check for timestamp pattern: [MM:SS.s]
   963→      const match = part.match(/^\[(\d{2}):(\d{2})\.(\d)\]$/);
   964→      if (match) {
   965→        const mins = parseInt(match[1], 10);
   966→        const secs = parseInt(match[2], 10);
   967→        const tenths = parseInt(match[3], 10);
   968→        const ms = (mins * 60 + secs) * 1000 + tenths * 100;
   969→        return (
   970→          <button
   971→            key={i}
   972→            className="inline-flex items-center gap-0.5 rounded bg-[#00E5FF]/15 px-1.5 py-0.5 text-[10px] font-medium text-[#00E5FF] hover:bg-[#00E5FF]/25 transition-colors"
   973→            onClick={() => seekTo(ms / 1000)}
   974→          >
   975→            <Clock className="h-2.5 w-2.5" />
   976→            {part}
   977→          </button>
   978→        );
   979→      }
   980→      return <span key={i}>{part}</span>;
   981→    });
