   577→                      </button>
   578→                    )}
   579→                  </div>
   580→                </div>
   581→              )}
   582→            </div>
   583→          );
   584→        })}
   585→
   586→        {/* Add custom stage */}
   587→        {addingStage ? (
   588→          <div className="rounded-lg border border-dashed border-slate-700/50 p-2 space-y-1.5">
   589→            <div className="flex items-center gap-2">
   590→              <Cat className="w-3 h-3 text-amber-400/60 flex-shrink-0" />

   592→                value={newStageTitle}

   613→            <Plus className="w-3 h-3" /> Добавить этап
   614→          </button>
   615→        )}
   616→      </div>
   617→    </div>
   618→  );
   619→}

   621→/* ── Regular Task Detail View ──────────── */
   622→
   623→function TaskDetailView({ task, board }: { task: Task; board?: { title: string; color: string } }) {
   624→  const { setBoardTasks, selectedBoardId, setEditingTask, setSelectedTaskId } = useKanbanStore();
   625→  const [subtasks, setSubtasks] = useState<TaskChild[]>(task.children || []);
   626→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
   627→  const [newSubtaskDeadline, setNewSubtaskDeadline] = useState<string | null>(null);
   628→  const [addingSubtask, setAddingSubtask] = useState(false);
   629→  const [expandedSubtasks, setExpandedSubtasks] = useState(true);
   630→
   631→  const boardColor = board?.color || '#00d9ff';
   632→  const progress = getProgress(subtasks);
   633→
   634→  const getProgressColor = () => {
   635→    if (progress === 100) return '#10b981';
   636→    if (progress > 50) return boardColor;
   637→    if (progress > 0) return '#f59e0b';
   638→    return '#334155';
   639→  };
   640→  const getProgressTextColor = () => {
   641→    if (progress === 100) return '#34d399';
   642→    if (progress > 50) return boardColor;
   643→    if (progress > 0) return '#f59e0b';
   644→    return '#475569';
   645→  };
   646→
   647→  const reloadTasks = useCallback(async () => {
   648→    if (!selectedBoardId) return;
   649→    const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   650→    const data = await res.json();
   651→    setBoardTasks(data.tasks);
   652→    const updated = data.tasks.find((t: Task) => t.id === task.id);
   653→    if (updated) setSubtasks(updated.children || []);
   654→  }, [selectedBoardId, setBoardTasks, task.id]);
   655→
   656→  const cycleSubtaskStatus = async (e: React.MouseEvent, subtaskId: string) => {
   657→    e.stopPropagation();
   658→    const order: TaskStatus[] = ['todo', 'in-progress', 'review', 'done'];
   659→    const sub = subtasks.find(s => s.id === subtaskId);
   660→    if (!sub) return;
   661→    const idx = order.indexOf(sub.status as TaskStatus);
   662→    const next = order[(idx + 1) % order.length];
   663→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: subtaskId, status: next }) });
   664→    await reloadTasks();
   665→  };
   666→
   667→  const deleteSubtask = async (e: React.MouseEvent, subtaskId: string) => {
   668→    e.stopPropagation();
   669→    await fetch(`/api/tasks?id=${subtaskId}`, { method: 'DELETE' });
   670→    await reloadTasks();
   671→  };
   672→
   673→  const addSubtask = async () => {
   674→    if (!newSubtaskTitle.trim()) return;
   675→    await fetch('/api/tasks', {
   676→      method: 'POST', headers: { 'Content-Type': 'application/json' },
   677→      body: JSON.stringify({ title: newSubtaskTitle.trim(), parentId: task.id, status: 'todo', ...(newSubtaskDeadline ? { deadline: newSubtaskDeadline } : {}) }),
   678→    });
   679→    setNewSubtaskTitle(''); setNewSubtaskDeadline(null); setAddingSubtask(false);
   680→    await reloadTasks();
   681→  };
   682→
   683→  const updateTaskDeadline = async (deadline: string | null) => {
   684→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: task.id, deadline }) });
   685→    await reloadTasks();
   686→  };
   687→
   688→  const updateSubtaskDeadline = async (subtaskId: string, deadline: string | null) => {
   689→    await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: subtaskId, deadline }) });
   690→    await reloadTasks();
   691→  };
   692→
   693→  const handleDeleteTask = async () => {
   694→    await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   695→    setSelectedTaskId(null);
   696→    if (selectedBoardId) {
   697→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   698→      const data = await res.json();
   699→      setBoardTasks(data.tasks);
   700→    }
   701→  };
   702→
   703→  return (
   704→    <div className="flex-1 overflow-y-auto">
   705→      <div className="border-b border-slate-800/50 p-4">
   706→        <div className="flex items-center gap-1.5 mb-2.5">
   707→          {board && (
   708→            <span className="text-[9px] uppercase tracking-wider px-1.5 py-0.5 rounded font-medium"
   709→              style={{ color: board.color, backgroundColor: board.color + '15' }}>{board.title}</span>
   710→          )}
   711→          <span className={cn('text-[10px] font-medium px-1.5 py-0.5 rounded',
   712→            task.status === 'todo' ? '' :
   713→            task.status === 'in-progress' ? 'text-orange-400 bg-orange-500/10' :
   714→            task.status === 'review' ? 'text-rose-400 bg-rose-500/10' :
   715→            'text-emerald-400 bg-emerald-500/10'
   716→          )}
   717→          style={task.status === 'todo' ? { color: boardColor, backgroundColor: boardColor + '10' } : undefined}
   718→          >
   719→            {STATUSES.find(s => s.value === task.status)?.label || task.status}
   720→          </span>
   721→          <div className="flex-1" />
   722→          <button onClick={() => setEditingTask(task)} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors"><Pencil className="w-3.5 h-3.5" /></button>
   723→          <button onClick={handleDeleteTask} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-rose-400 transition-colors"><Trash2 className="w-3.5 h-3.5" /></button>
   724→        </div>
   725→        <h3 className={cn('text-sm font-semibold leading-tight mb-1', task.status === 'done' ? 'text-slate-500 line-through' : 'text-slate-100')}>{task.title}</h3>
   726→        {task.description && <p className="text-[11px] text-slate-400 leading-relaxed mb-2">{task.description}</p>}
   727→        <div className="flex items-center gap-3 text-[10px] text-slate-500">
   728→          {task.assignee && <span>👤 {task.assignee}</span>}
   729→          <span className={cn('flex items-center gap-1',
   730→            task.priority === 'high' ? 'text-rose-400' : task.priority === 'medium' ? 'text-amber-400' : 'text-slate-500'
   731→          )}>
   732→            {task.priority === 'high' ? '🔴' : task.priority === 'medium' ? '🟡' : '⚪'}
   733→            {PRIORITIES.find(p => p.value === task.priority)?.label || task.priority}
   734→          </span>
   735→        </div>
   736→      </div>
   737→      <div className="border-b border-slate-800/30 px-4 py-3">
   738→        <DeadlinePicker value={task.deadline || null} onChange={updateTaskDeadline} isDone={task.status === 'done'} size="md" />
   739→      </div>
   740→      {subtasks.length > 0 && (
   741→        <div className="border-b border-slate-800/30 px-4 py-3">
   742→          <div className="flex items-center justify-between mb-1.5">
   743→            <span className="text-[11px] text-slate-400 font-medium">Прогресс</span>
   744→            <span className="text-[11px] font-bold" style={{ color: getProgressTextColor() }}>{progress}%</span>
   745→          </div>
   746→          <div className="relative h-2 bg-slate-800/80 rounded-full overflow-hidden">
   747→            <div className="h-full rounded-full transition-all duration-700 ease-out" style={{ width: `${progress}%`, backgroundColor: getProgressColor() }} />
   748→            {progress > 0 && progress < 100 && (
   749→              <div
   750→                className="absolute inset-0 rounded-full animate-pulse opacity-20"
   751→                style={{
   752→                  background: `linear-gradient(90deg, transparent, ${getProgressColor()}, transparent)`,
   753→                }}
   754→              />
   755→            )}
   756→          </div>
   757→          <div className="flex items-center justify-between mt-1">
   758→            <span className="text-[9px] text-slate-600">{subtasks.filter(s => s.status === 'done').length} из {subtasks.length} подзадач</span>
   759→            {progress === 100 && <span className="text-[9px] text-emerald-400 font-medium">✓ Завершено</span>}
   760→          </div>
   761→        </div>
   762→      )}
   763→      <div className="px-4 py-3">
   764→        <button onClick={() => setExpandedSubtasks(!expandedSubtasks)} className="flex items-center gap-1.5 mb-2 group w-full text-left">
   765→          {expandedSubtasks
   766→            ? <ChevronUp className="w-3.5 h-3.5 text-slate-600 group-hover:text-slate-400 transition-colors" />
   767→            : <ChevronDown className="w-3.5 h-3.5 text-slate-600 group-hover:text-slate-400 transition-colors" />}
   768→          <span className="text-[11px] font-semibold text-slate-400 uppercase tracking-wider">Подзадачи</span>
   769→          {subtasks.length > 0 && <span className="text-[9px] text-slate-600 bg-slate-800/50 px-1.5 py-0.5 rounded ml-1">{subtasks.length}</span>}
   770→        </button>
   771→        {expandedSubtasks && (
   772→          <div className="space-y-1.5">
   773→            {subtasks.length === 0 && !addingSubtask && <p className="text-[10px] text-slate-600 py-2 text-center">Нет подзадач</p>}

   775→              const Icon = SUBTASK_STATUS_ICON[sub.status] || Circle;

   778→                  key={sub.id}
   779→                  className="group/sub flex items-center gap-2 px-2 py-1.5 rounded-lg hover:bg-slate-900/80 transition-colors cursor-pointer"
   780→                  onClick={() => {
   781→                    useKanbanStore.getState().setDescriptionEditorItem({

   783→                      parentId: task.id, parentTitle: task.title,

   788→                  <span className={cn('flex-1 text-[11px] leading-tight min-w-0', sub.status === 'done' ? 'text-slate-500 line-through' : 'text-slate-300')}>{sub.title}</span>
   789→                  <div className="relative flex-shrink-0" onClick={(e) => e.stopPropagation()}>
   790→                    <DeadlinePicker value={sub.deadline || null} onChange={(d) => updateSubtaskDeadline(sub.id, d)} isDone={sub.status === 'done'} size="sm" inline />

   792→                  <button onClick={(e) => deleteSubtask(e, sub.id)} className="p-0.5 rounded opacity-0 group-hover/sub:opacity-100 hover:bg-slate-800 text-slate-600 hover:text-rose-400 transition-all flex-shrink-0"><X className="w-3 h-3" /></button>

   796→            {addingSubtask ? (

   798→                <Input value={newSubtaskTitle} onChange={(e) => setNewSubtaskTitle(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && addSubtask()} placeholder="Название подзадачи..." autoFocus className="bg-slate-800/80 border-slate-700/50 text-[11px] text-slate-200 placeholder:text-slate-600 h-7" onFocus={(e) => { e.target.style.borderColor = boardColor + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} />

   802→                </div>
   803→                <div className="flex gap-1.5 pt-0.5">
   804→                  <button onClick={addSubtask} disabled={!newSubtaskTitle.trim()} className="flex-1 text-[10px] text-white rounded px-2 py-1.5 transition-colors font-medium disabled:opacity-40" style={{ backgroundColor: boardColor }} onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.filter = 'brightness(1.1)'; }} onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.filter = ''; }}>Добавить</button>
   805→                  <button onClick={() => { setAddingSubtask(false); setNewSubtaskTitle(''); setNewSubtaskDeadline(null); }} className="text-[10px] text-slate-500 hover:text-slate-300 px-2 py-1.5 transition-colors">Отмена</button>
   806→                </div>
   807→              </div>
   808→            ) : (
   809→              <button onClick={() => setAddingSubtask(true)} className="flex items-center gap-1.5 px-2 py-1 text-[10px] text-slate-600 transition-colors rounded-lg hover:bg-slate-900/50 w-full" onMouseEnter={(e) => { (e.currentTarget as HTMLElement).style.color = boardColor; }} onMouseLeave={(e) => { (e.currentTarget as HTMLElement).style.color = ''; }}><Plus className="w-3 h-3" /> Добавить подзадачу</button>
   810→            )}
   811→          </div>
   812→        )}
   813→      </div>
   814→    </div>
   815→  );
   816→}

   818→/* ── Task Form (create / edit) ──────────── */
   819→
   820→function TaskForm({ mode, task, boardColor }: { mode: 'create' | 'edit'; task?: Task; boardColor?: string }) {
   821→  const { selectedBoardId, setBoardTasks, selectedProjectId, setEditingTask, setIsCreating, setSelectedTaskId } = useKanbanStore();
   822→  const color = boardColor || '#00d9ff';
   823→  const [title, setTitle] = useState('');
   824→  const [description, setDescription] = useState('');
   825→  const [status, setStatus] = useState<TaskStatus>('todo');
   826→  const [priority, setPriority] = useState<TaskPriority>('medium');
   827→  const [assignee, setAssignee] = useState('');
   828→  const [category, setCategory] = useState<TaskCategory>('general');
   829→  const [deadline, setDeadline] = useState<string | null>(null);
   830→  const [saving, setSaving] = useState(false);
   831→
   832→  useEffect(() => {
   833→    if (mode === 'edit' && task) {
   834→      setTitle(task.title); setDescription(task.description || ''); setStatus(task.status);
   835→      setPriority(task.priority); setAssignee(task.assignee || ''); setCategory(task.category);
   836→      setDeadline(task.deadline || null);
   837→    }
   838→  }, [mode, task]);
   839→
   840→  const reloadTasks = useCallback(async () => {
   841→    if (selectedBoardId) {
   842→      const res = await fetch(`/api/tasks?boardId=${selectedBoardId}`);
   843→      const data = await res.json();
   844→      setBoardTasks(data.tasks);
   845→    }
   846→    if (selectedProjectId) {
   847→      const bRes = await fetch(`/api/boards?projectId=${selectedProjectId}`);
   848→      const bData = await bRes.json();
   849→      useKanbanStore.getState().setBoards(bData.boards);
   850→    }
   851→  }, [selectedBoardId, selectedProjectId, setBoardTasks]);
   852→
   853→  const handleSave = async () => {
   854→    if (!title.trim()) return;
   855→    setSaving(true);
   856→    try {
   857→      if (mode === 'edit' && task) {
   858→        await fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: task.id, title, description: description || null, status, priority, assignee: assignee || null, category, boardId: selectedBoardId, deadline }) });
   859→      } else {
   860→        await fetch('/api/tasks', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ title, description: description || null, status, priority, assignee: assignee || null, category, parentId: null, boardId: selectedBoardId, deadline }) });
   861→      }
   862→      await reloadTasks(); setEditingTask(null); setIsCreating(false);
   863→      if (mode === 'edit' && task) setSelectedTaskId(task.id);
   864→    } finally { setSaving(false); }
   865→  };
   866→
   867→  const handleDelete = async () => {
   868→    if (!task) return;
   869→    setSaving(true);
   870→    try {
   871→      await fetch(`/api/tasks?id=${task.id}`, { method: 'DELETE' });
   872→      await reloadTasks(); setEditingTask(null); setSelectedTaskId(null);
   873→    } finally { setSaving(false); }
   874→  };
   875→
   876→  return (
   877→    <div className="flex-1 overflow-y-auto">
   878→      <div className="p-4 space-y-3">
   879→        <div className="flex items-center justify-between">
   880→          <h3 className="text-xs font-semibold uppercase tracking-wider" style={{ color }}>{mode === 'edit' ? 'Редактировать' : 'Новая задача'}</h3>
   881→          <button onClick={() => { setEditingTask(null); setIsCreating(false); }} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300"><X className="h-3.5 w-3.5" /></button>
   882→        </div>
   883→        <div className="space-y-2.5">
   884→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Название</Label><Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Задача..." className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-8" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   885→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Описание</Label><Textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Описание..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-300 placeholder:text-slate-600 min-h-[50px] resize-none" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   886→          <div className="grid grid-cols-2 gap-2">
   887→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Статус</Label><Select value={status} onValueChange={(v) => setStatus(v as TaskStatus)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{STATUSES.map((s) => (<SelectItem key={s.value} value={s.value}><span className={s.color}>{s.label}</span></SelectItem>))}</SelectContent></Select></div>
   888→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Приоритет</Label><Select value={priority} onValueChange={(v) => setPriority(v as TaskPriority)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{PRIORITIES.map((p) => (<SelectItem key={p.value} value={p.value}><div className="flex items-center gap-1.5"><div className={cn('w-1.5 h-1.5 rounded-full', p.dot)} />{p.label}</div></SelectItem>))}</SelectContent></Select></div>
   889→          </div>
   890→          <div className="grid grid-cols-2 gap-2">
   891→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Категория</Label><Select value={category} onValueChange={(v) => setCategory(v as TaskCategory)}><SelectTrigger className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 h-8"><SelectValue /></SelectTrigger><SelectContent className="bg-slate-900 border-slate-700">{CATEGORIES.map((c) => (<SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>))}</SelectContent></Select></div>
   892→            <div className="space-y-1"><Label className="text-[11px] text-slate-500">Ответственный</Label><Input value={assignee} onChange={(e) => setAssignee(e.target.value)} placeholder="Имя..." className="bg-slate-900/80 border-slate-700/50 text-xs text-slate-200 placeholder:text-slate-600 h-8" onFocus={(e) => { e.target.style.borderColor = color + '80'; }} onBlur={(e) => { e.target.style.borderColor = ''; }} /></div>
   893→          </div>
   894→          <div className="space-y-1"><Label className="text-[11px] text-slate-500">Дедлайн</Label><DeadlinePicker value={deadline} onChange={setDeadline} size="md" /></div>
   895→        </div>
   896→        <div className="flex gap-2 pt-1">
   897→          <Button onClick={handleSave} disabled={!title.trim() || saving} className="flex-1 text-white h-8 text-xs" style={{ backgroundColor: color, boxShadow: `0 4px 14px ${color}30` }}><Save className="w-3 h-3 mr-1" />{saving ? '...' : mode === 'edit' ? 'Сохранить' : 'Создать'}</Button>
   898→          {mode === 'edit' && <Button onClick={handleDelete} variant="destructive" disabled={saving} className="h-8 text-xs">Удалить</Button>}
   899→        </div>
   900→      </div>
   901→    </div>
   902→  );
   903→}
   904→
   905→function getProgress(children: { status: string }[]): number {
   906→  if (children.length === 0) return 0;
   907→  return Math.round((children.filter(c => c.status === 'done').length / children.length) * 100);
   908→}
