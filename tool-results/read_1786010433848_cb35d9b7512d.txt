    95→    95→
    96→    96→function PriorityBars({ priority, size = 'sm' }: { priority: string; size?: 'sm' | 'xs' }) {
    97→    97→  const level = PRIORITY_LEVELS[priority] || 2;
    98→    98→  const hex = PRIORITY_HEX[priority] || '#64748b';
    99→    99→  const barW = size === 'xs' ? 'w-[2.5px]' : 'w-[3px]';
   100→   100→  const gap = size === 'xs' ? 'gap-[1.5px]' : 'gap-[2px]';
   101→   101→
   102→   102→  return (
   103→   103→    <div className={cn('flex items-end', gap)} style={{ height: size === 'xs' ? '10px' : '12px' }}>
   104→   104→      {[1, 2, 3].map(i => (
   105→   105→        <div
   106→   106→          key={i}
   107→   107→          className={cn(barW, 'rounded-sm transition-all duration-200')}
   108→   108→          style={{
   109→   109→            height: i === 1 ? '30%' : i === 2 ? '65%' : '100%',
   110→   110→            backgroundColor: i <= level ? hex : hexToRgba(hex, 0.15),
   111→   111→            boxShadow: i <= level ? `0 0 3px ${hex}60` : 'none',
   112→   112→          }}
   113→   113→        />
   114→   114→      ))}
   115→   115→    </div>
   116→   116→  );
   117→   117→}

   698→   698→/* ── Stages List (for track tasks) ──────────────────────── */
   699→   699→
   700→   700→function StagesList({
   701→   701→  task, boardColor, c, members, selectedStageId, onSelectStage, reloadTasks,
   702→   702→}: {
   703→   703→  task: Task;
   704→   704→  boardColor: string;
   705→   705→  c: ColorSet;
   706→   706→  members: GroupMember[];
   707→   707→  selectedStageId: string | null;
   708→   708→  onSelectStage: (stageId: string) => void;
   709→   709→  reloadTasks: () => Promise<void>;
   710→   710→}) {
   711→   711→  const stages = task.children || [];
   712→   712→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(() => {
   713→   713→    // Auto-expand first stage or the selected one
   714→   714→    const initial: Record<string, boolean> = {};
   715→   715→    if (stages.length > 0) {
   716→   716→      initial[stages[0].id] = true;
   717→   717→    }
   718→   718→    if (selectedStageId && stages.find(s => s.id === selectedStageId)) {
   719→   719→      initial[selectedStageId] = true;
   720→   720→    }
   721→   721→    return initial;
   722→   722→  });
   723→   723→  const [addingStage, setAddingStage] = useState(false);
   724→   724→  const [newStageTitle, setNewStageTitle] = useState('');
   725→   725→
   726→   726→  // If selectedStageId changes, ensure that stage is expanded (prev-tracker pattern avoids useEffect-setState)
   727→   727→  const [prevSelectedStageId, setPrevSelectedStageId] = useState<string | null>(selectedStageId);
   728→   728→  if (selectedStageId !== prevSelectedStageId) {
   729→   729→    setPrevSelectedStageId(selectedStageId);
   730→   730→    if (selectedStageId && stages.find(s => s.id === selectedStageId) && !expandedStages[selectedStageId]) {
   731→   731→      setExpandedStages(prev => ({ ...prev, [selectedStageId]: true }));
   732→   732→    }
   733→   733→  }
   734→   734→
   735→   735→  const toggleExpand = (stageId: string) => {
   736→   736→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   737→   737→  };
   738→   738→
   739→   739→  const addStage = async () => {
   740→   740→    if (!newStageTitle.trim()) return;
   741→   741→    await fetch('/api/tasks', {
   742→   742→      method: 'POST',
   743→   743→      headers: { 'Content-Type': 'application/json' },
   744→   744→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: task.boardId }),
   745→   745→    });
   746→   746→    setNewStageTitle('');
   747→   747→    setAddingStage(false);
   748→   748→    await reloadTasks();
   749→   749→  };
   750→   750→
   751→   751→  const moveStage = async (index: number, direction: -1 | 1) => {
   752→   752→    const target = index + direction;
   753→   753→    if (target < 0 || target >= stages.length) return;
   754→   754→    const a = stages[index];
   755→   755→    const b = stages[target];
   756→   756→    // Swap hexR values
   757→   757→    await Promise.all([
   758→   758→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: a.id, hexR: b.hexR, hexQ: b.hexQ }) }),
   759→   759→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: b.id, hexR: a.hexR, hexQ: a.hexQ }) }),
   760→   760→    ]);
   761→   761→    await reloadTasks();
   762→   762→  };
   763→   763→
   764→   764→  const deleteStage = async (stageId: string) => {
   765→   765→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   766→   766→    await reloadTasks();
   767→   767→  };
   768→   768→
   769→   769→  const updateStage = async (stageId: string, patch: Record<string, unknown>) => {
   770→   770→    await fetch('/api/tasks', {
   771→   771→      method: 'PUT',
   772→   772→      headers: { 'Content-Type': 'application/json' },
   773→   773→      body: JSON.stringify({ id: stageId, ...patch }),
   774→   774→    });
   775→   775→    await reloadTasks();
   776→   776→  };
   777→   777→
   778→   778→  return (
   779→   779→    <div className="py-2 space-y-1.5">
   780→   780→      {stages.length === 0 && !addingStage && (
   781→   781→        <div className="text-center py-6 px-4">
   782→   782→          <ListChecks className="w-6 h-6 mx-auto mb-2 text-slate-700" />
   783→   783→          <p className="text-[11px] text-slate-500 mb-1">Этапов пока нет</p>
   784→   784→          <p className="text-[10px] text-slate-600 mb-3">Создайте первый этап производства трека</p>
   785→   785→        </div>
   786→   786→      )}
   787→   787→
   788→   788→      {stages.map((stage, idx) => (
   789→   789→        <StageCard
   790→   790→          key={stage.id}
   791→   791→          stage={stage}
   792→   792→          index={idx}
   793→   793→          total={stages.length}
   794→   794→          boardColor={boardColor}
   795→   795→          c={c}
   796→   796→          members={members}
   797→   797→          isExpanded={!!expandedStages[stage.id]}
   798→   798→          isSelected={selectedStageId === stage.id}
   799→   799→          onToggleExpand={() => toggleExpand(stage.id)}
   800→   800→          onSelect={() => onSelectStage(stage.id)}
   801→   801→          onMoveUp={() => moveStage(idx, -1)}
   802→   802→          onMoveDown={() => moveStage(idx, 1)}
   803→   803→          onDelete={() => deleteStage(stage.id)}
   804→   804→          onUpdate={(patch) => updateStage(stage.id, patch)}
   805→   805→          reloadTasks={reloadTasks}
   806→   806→        />
   807→   807→      ))}
   808→   808→
   809→   809→      {/* Add stage form */}
   810→   810→      {addingStage ? (
   811→   811→        <div
   812→   812→          className="mx-2 rounded-lg p-2.5 space-y-1.5"
   813→   813→          style={{ border: `1px dashed ${c.a3}`, backgroundColor: c.a04 }}
   814→   814→        >
   815→   815→          <Input
   816→   816→            value={newStageTitle}
   817→   817→            onChange={(e) => setNewStageTitle(e.target.value)}
   818→   818→            onKeyDown={(e) => {
   819→   819→              if (e.key === 'Enter') void addStage();
   820→   820→              if (e.key === 'Escape') { setAddingStage(false); setNewStageTitle(''); }
   821→   821→            }}
   822→   822→            placeholder="Название этапа..."
   823→   823→            className="bg-slate-900/80 border-slate-700/50 text-[11px] text-slate-200 placeholder:text-slate-600 h-7"
   824→   824→            style={{ borderColor: c.a3 }}
   825→   825→            autoFocus
   826→   826→          />
   827→   827→          <div className="flex items-center gap-1.5">
   828→   828→            <button
   829→   829→              onClick={() => void addStage()}
   830→   830→              disabled={!newStageTitle.trim()}
   831→   831→              className="text-[10px] px-2 py-1 rounded font-medium transition-all disabled:opacity-40"
   832→   832→              style={{ color: '#fff', backgroundColor: c.a5 }}
   833→   833→            >
   834→   834→              + Добавить этап
   835→   835→            </button>
   836→   836→            <button
   837→   837→              onClick={() => { setAddingStage(false); setNewStageTitle(''); }}
   838→   838→              className="text-[10px] text-slate-500 hover:text-slate-300 px-2 py-1"
   839→   839→            >
   840→   840→              Отмена
   841→   841→            </button>
   842→   842→          </div>
   843→   843→        </div>
   844→   844→      ) : (
   845→   845→        <button
   846→   846→          onClick={() => setAddingStage(true)}
   847→   847→          className="cp-add-btn mx-2 w-[calc(100%-1rem)] justify-center"
   848→   848→        >
   849→   849→          <Plus className="w-3 h-3" /> Добавить этап
   850→   850→        </button>
   851→   851→      )}
   852→   852→    </div>
   853→   853→  );
   854→   854→}

   857→   857→
   858→   858→function StageCard({
   859→   859→  stage, index, total, boardColor, c, members,
   860→   860→  isExpanded, isSelected,
   861→   861→  onToggleExpand, onSelect, onMoveUp, onMoveDown, onDelete, onUpdate, reloadTasks,
   862→   862→}: {
   863→   863→  stage: TaskChild;
   864→   864→  index: number;
   865→   865→  total: number;
   866→   866→  boardColor: string;
   867→   867→  c: ColorSet;
   868→   868→  members: GroupMember[];
   869→   869→  isExpanded: boolean;
