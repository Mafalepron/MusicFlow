   138→/* ─── Kanban Card ─── */
   139→function KanbanCard({ task, onClick }: { task: Task; onClick: () => void }) {
   140→  const [h, setH] = useState(false);
   141→  const isAuto = !!task.soundflowProjectId;
   142→  const color = isAuto ? Y : C;
   143→  const children = task.children || [];
   144→  const done = children.filter(c => c.status === 'done').length;
   145→  const pct = children.length > 0 ? Math.round((done / children.length) * 100) : 0;
   146→
   147→  return (
   148→    <div
   149→      onClick={onClick}
   150→      onMouseEnter={() => setH(true)}
   151→      onMouseLeave={() => setH(false)}
   152→      className="cursor-pointer p-4"
   153→      style={{
   154→        borderRadius: '10px',
   155→        background: h ? `linear-gradient(135deg, ${hexToRgba(color, 0.15)}, rgba(16,20,30,0.95))` : `linear-gradient(135deg, ${hexToRgba(color, 0.08)}, rgba(14,18,28,0.85))`,
   156→        border: `1px solid ${h ? hexToRgba(color, 0.55) : hexToRgba(color, 0.25)}`,
   157→        boxShadow: h ? `0 0 0 1px ${hexToRgba(color, 0.2)}, 0 6px 24px ${hexToRgba(color, 0.15)}, 0 4px 12px rgba(0,0,0,0.35)` : `0 0 0 1px ${hexToRgba(color, 0.05)}, 0 4px 10px rgba(0,0,0,0.25)`,
   158→        transition: 'all 220ms cubic-bezier(0.4,0,0.2,1)',
   159→        transform: h ? 'translateY(-3px) scale(1.01)' : 'none',
   160→      }}
   161→    >
   162→      <div className="mb-2 flex items-center justify-between">
   163→        <span
   164→          className="rounded px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wider"
   165→          style={{ background: hexToRgba(color, 0.12), color, border: `1px solid ${hexToRgba(color, 0.25)}` }}
   166→        >
   167→          {isAuto ? 'AUTO' : 'KANBAN'}
   168→        </span>
   169→        <span className="text-[10px] text-slate-600">{task.projectType || 'general'}</span>
   170→      </div>
   171→      <h3 className="mb-3 text-sm font-medium text-slate-200 line-clamp-2" style={{ minHeight: '2.5em' }}>
   172→        {task.title}
   173→      </h3>
   174→      <div className="flex items-center gap-2">
   175→        <div className="flex-1 h-1 rounded-full bg-slate-800 overflow-hidden">
   176→          <div
   177→            className="h-full rounded-full transition-all"
   178→            style={{ width: `${pct}%`, background: color, boxShadow: pct > 0 ? `0 0 4px ${hexToRgba(color, 0.5)}` : 'none' }}
   179→          />
   180→        </div>
   181→        <span className="text-[10px] font-semibold tabular-nums" style={{ color: pct === 100 ? G : color }}>
   182→          {pct}%
   183→        </span>
   184→      </div>
   185→    </div>
   186→  );
   187→}

   445→function Carousel({ children }: { children: React.ReactNode }) {
   446→  const scrollRef = useRef<HTMLDivElement>(null);
   447→  const [canLeft, setCanLeft] = useState(false);
   448→  const [canRight, setCanRight] = useState(false);
   449→
   450→  const updateArrows = useCallback(() => {
   451→    const el = scrollRef.current;
   452→    if (!el) return;
   453→    setCanLeft(el.scrollLeft > 8);
   454→    setCanRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 8);

   457→  useEffect(() => {
   458→    updateArrows();
   459→    const el = scrollRef.current;
   460→    if (!el) return;
   461→    el.addEventListener('scroll', updateArrows, { passive: true });
   462→    window.addEventListener('resize', updateArrows);
   463→    return () => {
   464→      el.removeEventListener('scroll', updateArrows);
   465→      window.removeEventListener('resize', updateArrows);
   466→    };
   467→  }, [updateArrows]);
