    48→// ── Unified card for both auto projects and kanban projects ──
    49→// `kind` distinguishes the source so we render the right badge + open the right view.
    50→type UnifiedCard =
    51→  | { kind: 'auto'; project: Project; trackCount: number }
    52→  | { kind: 'kanban'; task: Task; boardCount: number };

    54→// Helper: get the underlying SoundFlow project ID for a card (used when PATCHing folderId)
    55→function getCardProjectId(card: UnifiedCard): string | null {
    56→  if (card.kind === 'auto') return card.project.id;
    57→  return null; // kanban-only projects don't have a SoundFlow Project record
    58→}
    59→
    60→// Helper: get the current folderId for a card
    61→function getCardFolderId(card: UnifiedCard): string | null {
    62→  if (card.kind === 'auto') return card.project.folderId ?? null;
    63→  return null;
    64→}

   358→function FolderSection({

   360→  cards,
   361→  folders,

   363→  isFavorite,

   367→  onMoveProject,
   368→  onRename,
   369→  onDelete,
   370→  defaultExpanded,
   371→}: {
   372→  folder: Folder;
   373→  cards: UnifiedCard[];
   374→  folders: Folder[];
   375→  tracks: ReturnType<typeof useDataStore.getState>['tracks'];
   376→  isFavorite: (id: string) => boolean;
   377→  toggleFavorite: (id: string) => void;
   378→  onOpenProject: (project: Project) => void;
   379→  onOpenKanban: (kanbanTaskId: string) => void;
   380→  onMoveProject: (projectId: string, folderId: string | null) => void;
   381→  onRename: (id: string, newTitle: string) => void;
   382→  onDelete: (id: string) => void;
   383→  defaultExpanded: boolean;
   384→}) {
   385→  const [expanded, setExpanded] = useState(defaultExpanded);
   386→  const [renaming, setRenaming] = useState(false);
   387→  const [renameValue, setRenameValue] = useState(folder.title);
   388→  const [confirmingDelete, setConfirmingDelete] = useState(false);
   389→
   390→  const getTrackCount = (projectId: string) =>
   391→    tracks.filter((t) => t.projectId === projectId).length;
   392→
   393→  return (
   394→    <div
   395→      className="overflow-hidden"
   396→      style={{
   397→        borderRadius: '10px',
   398→        background: 'linear-gradient(135deg, rgba(252,238,10,0.04) 0%, rgba(14,18,28,0.55) 60%)',
   399→        border: `1px solid ${expanded ? 'rgba(252,238,10,0.4)' : 'rgba(252,238,10,0.22)'}`,
   400→        boxShadow: expanded
   401→          ? '0 0 0 1px rgba(252,238,10,0.08), 0 6px 20px rgba(0,0,0,0.35), inset 0 1px 0 rgba(255,255,255,0.04)'
   402→          : '0 0 0 1px rgba(252,238,10,0.05), 0 4px 12px rgba(0,0,0,0.25)',
   403→        transition: 'all 220ms cubic-bezier(0.4,0,0.2,1)',
   404→      }}
   405→    >
   406→      {/* Header (clickable to toggle expand) */}
   407→      <button
   408→        onClick={() => setExpanded((v) => !v)}
   409→        className="flex w-full items-center gap-3 px-4 py-3 transition-colors hover:bg-white/[0.02]"
   410→        style={{ cursor: 'pointer', background: 'transparent', border: 'none' }}
   411→      >
   412→        <div
   413→          className="flex h-7 w-7 shrink-0 items-center justify-center"
   414→          style={{
   415→            clipPath: 'polygon(0 0, calc(100% - 3px) 0, 100% 3px, 100% 100%, 3px 100%, 0 calc(100% - 3px))',
   416→            background: 'linear-gradient(135deg, rgba(252,238,10,0.18), rgba(241,241,0,0.1))',
   417→            border: '1px solid rgba(252,238,10,0.45)',
   418→            boxShadow: '0 0 8px rgba(252,238,10,0.2)',
   419→          }}
   420→        >
   421→          <FolderOpen className="w-3.5 h-3.5" style={{ color: '#FCEE0A' }} />
   422→        </div>
   423→
   424→        {renaming ? (
   425→          <input
   426→            autoFocus
   427→            value={renameValue}
   428→            onChange={(e) => setRenameValue(e.target.value)}
   429→            onClick={(e) => e.stopPropagation()}
   430→            onKeyDown={(e) => {
   431→              if (e.key === 'Enter' && renameValue.trim()) {
   432→                onRename(folder.id, renameValue.trim());
   433→                setRenaming(false);
   434→              } else if (e.key === 'Escape') {
   435→                setRenameValue(folder.title);
   436→                setRenaming(false);
   437→              }
   438→            }}
   439→            className="flex-1 bg-transparent text-sm font-semibold text-slate-100 outline-none border-b border-[#FCEE0A]/50"
   440→            style={{ fontFamily: 'inherit' }}
   441→          />
   442→        ) : (
   443→          <h3 className="flex-1 text-left text-sm font-semibold text-slate-100 truncate">
   444→            {folder.title}
   445→          </h3>
   446→        )}
   447→
   448→        {/* Count badge */}
   449→        <span
   450→          className="rounded px-1.5 py-0.5 text-[9px] font-bold uppercase tracking-wider tabular-nums"
   451→          style={{
   452→            background: 'rgba(252,238,10,0.08)',
   453→            color: '#FCEE0A',
   454→            border: '1px solid rgba(252,238,10,0.25)',
   455→          }}
   456→        >
   457→          {cards.length} {cards.length === 1 ? 'проект' : cards.length > 4 ? 'проектов' : 'проекта'}
   458→        </span>
   459→
   460→        {/* Rename / delete actions (stop propagation so they don't toggle expand) */}
   461→        {renaming ? (
   462→          <>
   463→            <button
   464→              onClick={(e) => {
   465→                e.stopPropagation();
   466→                if (renameValue.trim()) {
   467→                  onRename(folder.id, renameValue.trim());
   468→                  setRenaming(false);
   469→                }
   470→              }}
   471→              className="flex h-6 w-6 items-center justify-center text-[#FCEE0A] hover:bg-white/[0.06] rounded transition-colors"

   481→                setRenaming(false);
   482→              }}
   483→              className="flex h-6 w-6 items-center justify-center text-slate-400 hover:bg-white/[0.06] rounded transition-colors"

   485→              style={{ background: 'transparent', border: 'none', cursor: 'pointer' }}

   487→              <X className="w-3.5 h-3.5" />
   488→            </button>
   489→          </>
   490→        ) : confirmingDelete ? (
   491→          <>
   492→            <span className="text-[10px] text-slate-400 mr-1">Удалить?</span>
   493→            <button
   494→              onClick={(e) => {

   497→                setConfirmingDelete(false);

   501→                background: 'linear-gradient(135deg, #ef4444, #b91c1c)',
   502→                color: '#fff',
   503→                border: '1px solid rgba(239,68,68,0.6)',
   504→                cursor: 'pointer',
   505→              }}

   507→              Да
   508→            </button>
   509→            <button
   510→              onClick={(e) => {
   511→                e.stopPropagation();
   512→                setConfirmingDelete(false);
   513→              }}
   514→              className="flex h-6 px-2 items-center justify-center rounded text-[10px] font-bold uppercase tracking-wider transition-colors hover:bg-white/[0.06]"
   515→              style={{
   516→                background: 'rgba(30, 35, 50, 0.6)',
   517→                color: '#94a3b8',

   520→              }}
   521→            >
   522→              Нет
   523→            </button>
   524→          </>
   525→        ) : (
   526→          <>
   527→            <button
   528→              onClick={(e) => {
   529→                e.stopPropagation();
   530→                setRenameValue(folder.title);
   531→                setRenaming(true);

   533→              className="flex h-6 w-6 items-center justify-center text-slate-400 hover:text-[#FCEE0A] hover:bg-white/[0.06] rounded transition-colors"

   539→            </button>
   540→            <button
   541→              onClick={(e) => {
   542→                e.stopPropagation();
   543→                setConfirmingDelete(true);
   544→              }}
   545→              className="flex h-6 w-6 items-center justify-center text-slate-400 hover:text-red-400 hover:bg-white/[0.06] rounded transition-colors"
   546→              aria-label="Удалить папку"
   547→              title="Удалить папку"
   548→              style={{ background: 'transparent', border: 'none', cursor: 'pointer' }}
   549→            >
   550→              <Trash2 className="w-3.5 h-3.5" />
   551→            </button>
   552→          </>
   553→        )}
   554→
   555→        {/* Expand chevron */}
   556→        <ChevronDown
   557→          className="w-4 h-4 text-slate-400 transition-transform duration-200"
   558→          style={{ transform: expanded ? 'rotate(180deg)' : 'rotate(0deg)' }}
   559→        />
   560→      </button>
   561→
   562→      {/* Body (cards grid) */}
   563→      {expanded && (

   566→            <motion.div
   567→              key={`folder-${folder.id}`}
   568→              variants={containerVariants}
   569→              initial="hidden"
   570→              animate="show"
   571→              className="grid grid-cols-1 gap-4 md:grid-cols-2 lg:grid-cols-3"
   572→            >
   573→              {cards.map((card) => {
   574→                const cardId = card.kind === 'auto'
   575→                  ? (card.project.kanbanTaskId || card.project.id)
   576→                  : card.task.id;
   577→                const trackCount = card.kind === 'auto'
   578→                  ? getTrackCount(card.project.id)
   579→                  : card.boardCount;
   580→                return (
   581→                  <ProjectCardUnified
   582→                    key={card.kind === 'auto' ? `auto-${card.project.id}` : `kanban-${card.task.id}`}
   583→                    data={card.kind === 'auto'

   586→                    onClick={() => {
   587→                      if (card.kind === 'auto') {
   588→                        onOpenProject(card.project);
   589→                      } else {
   590→                        onOpenKanban(card.task.id);
   591→                      }
   592→                    }}
   593→                    onOpenKanban={() => {
   594→                      if (card.kind === 'auto' && card.project.kanbanTaskId) {
   595→                        onOpenKanban(card.project.kanbanTaskId);
   596→                      } else if (card.kind === 'kanban') {
   597→                        onOpenKanban(card.task.id);
   598→                      }
   599→                    }}
   600→                    isFavorite={isFavorite(cardId)}
   601→                    onToggleFavorite={() => toggleFavorite(cardId)}
   602→                    folders={folders}
   603→                    onMoveToFolder={(folderId) => {
   604→                      const pid = getCardProjectId(card);
   605→                      if (pid) onMoveProject(pid, folderId);
   606→                    }}
   607→                  />
   608→                );
   609→              })}
   610→            </motion.div>
   611→          ) : (
   612→            <div className="flex flex-col items-center justify-center py-6 text-center">
   613→              <p className="text-xs text-slate-500">
   614→                В папке нет проектов. Переместите сюда проект кнопкой{' '}
   615→                <FolderInput className="inline-block w-3 h-3 -mt-0.5" style={{ color: 'rgba(252,238,10,0.6)' }} />{' '}
   616→                на карточке.
   617→              </p>
   618→            </div>
   619→          )}
   620→        </div>
   621→      )}
   622→    </div>
   623→  );
   624→}
