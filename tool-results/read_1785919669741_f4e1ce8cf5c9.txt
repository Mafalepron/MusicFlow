   116→   116→// Build a nested tree from flat comment list
   117→   117→interface CommentNode extends Comment {
   118→   118→  replies: CommentNode[];
   119→   119→}
   120→   120→
   121→   121→function buildCommentTree(comments: Comment[]): CommentNode[] {
   122→   122→  const map = new Map<string, CommentNode>();
   123→   123→  const roots: CommentNode[] = [];
   124→   124→  
   125→   125→  for (const c of comments) {
   126→   126→    map.set(c.id, { ...c, replies: [] });
   127→   127→  }
   128→   128→  for (const c of comments) {
   129→   129→    const node = map.get(c.id)!;
   130→   130→    if (c.parentId && map.has(c.parentId)) {
   131→   131→      map.get(c.parentId)!.replies.push(node);
   132→   132→    } else {
   133→   133→      roots.push(node);
   134→   134→    }
   135→   135→  }
   136→   136→  roots.sort((a, b) => a.timestampMs - b.timestampMs);
   137→   137→  for (const r of roots) {
   138→   138→    r.replies.sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   139→   139→  }
   140→   140→  return roots;
   141→   141→}

   162→   162→function formatTimestamp(ms: number): string {
   163→   163→  const totalSeconds = ms / 1000;
   164→   164→  const minutes = Math.floor(totalSeconds / 60);
   165→   165→  const seconds = Math.floor(totalSeconds % 60);
   166→   166→  const milliseconds = Math.floor((totalSeconds % 1) * 10);
   167→   167→  return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}.${milliseconds}`;
   168→   168→}
   169→   169→
   170→   170→function formatDuration(seconds: number): string {
   171→   171→  const mins = Math.floor(seconds / 60);
   172→   172→  const secs = Math.floor(seconds % 60);
   173→   173→  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   174→   174→}

   176→   176→function getInitials(name?: string | null): string {
   177→   177→  if (!name) return '?';
   178→   178→  return name
   179→   179→    .trim()
   180→   180→    .split(' ')
   181→   181→    .map((n) => n[0])
   182→   182→    .join('')
   183→   183→    .toUpperCase()
   184→   184→    .slice(0, 2);
   185→   185→}
