   245→// Build a nested tree from flat comment list
   246→interface CommentNode extends Comment {
   247→  replies: CommentNode[];
   248→}
   249→
   250→function buildCommentTree(comments: Comment[]): CommentNode[] {
   251→  const map = new Map<string, CommentNode>();
   252→  const roots: CommentNode[] = [];
   253→  
   254→  for (const c of comments) {
   255→    map.set(c.id, { ...c, replies: [] });
   256→  }
   257→  for (const c of comments) {
   258→    const node = map.get(c.id)!;
   259→    if (c.parentId && map.has(c.parentId)) {
   260→      map.get(c.parentId)!.replies.push(node);
   261→    } else {
   262→      roots.push(node);
   263→    }
   264→  }
   265→  roots.sort((a, b) => a.timestampMs - b.timestampMs);
   266→  for (const r of roots) {
   267→    r.replies.sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   268→  }
   269→  return roots;
   270→}

   291→function formatTimestamp(ms: number): string {
   292→  const totalSeconds = ms / 1000;
   293→  const minutes = Math.floor(totalSeconds / 60);
   294→  const seconds = Math.floor(totalSeconds % 60);
   295→  const milliseconds = Math.floor((totalSeconds % 1) * 10);
   296→  return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}.${milliseconds}`;
   297→}
   298→
   299→function formatDuration(seconds: number): string {
   300→  const mins = Math.floor(seconds / 60);
   301→  const secs = Math.floor(seconds % 60);
   302→  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   303→}

   305→function getInitials(name?: string | null): string {
   306→  if (!name) return '?';
   307→  return name
   308→    .trim()
   309→    .split(' ')
   310→    .map((n) => n[0])
   311→    .join('')
   312→    .toUpperCase()
   313→    .slice(0, 2);
   314→}

  1043→        ctx.fill();
  1044→      }
