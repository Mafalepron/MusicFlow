   111→// Build a nested tree from flat comment list
   112→interface CommentNode extends Comment {
   113→  replies: CommentNode[];
   114→}
   115→
   116→function buildCommentTree(comments: Comment[]): CommentNode[] {
   117→  const map = new Map<string, CommentNode>();
   118→  const roots: CommentNode[] = [];
   119→  
   120→  for (const c of comments) {
   121→    map.set(c.id, { ...c, replies: [] });
   122→  }
   123→  for (const c of comments) {
   124→    const node = map.get(c.id)!;
   125→    if (c.parentId && map.has(c.parentId)) {
   126→      map.get(c.parentId)!.replies.push(node);
   127→    } else {
   128→      roots.push(node);
   129→    }
   130→  }
   131→  roots.sort((a, b) => a.timestampMs - b.timestampMs);
   132→  for (const r of roots) {
   133→    r.replies.sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   134→  }
   135→  return roots;
   136→}

   157→function formatTimestamp(ms: number): string {
   158→  const totalSeconds = ms / 1000;
   159→  const minutes = Math.floor(totalSeconds / 60);
   160→  const seconds = Math.floor(totalSeconds % 60);
   161→  const milliseconds = Math.floor((totalSeconds % 1) * 10);
   162→  return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}.${milliseconds}`;
   163→}
   164→
   165→function formatDuration(seconds: number): string {
   166→  const mins = Math.floor(seconds / 60);
   167→  const secs = Math.floor(seconds % 60);
   168→  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   169→}

   171→function getInitials(name?: string | null): string {
   172→  if (!name) return '?';
   173→  return name
   174→    .trim()
   175→    .split(' ')
   176→    .map((n) => n[0])
   177→    .join('')
   178→    .toUpperCase()
   179→    .slice(0, 2);
   180→}
