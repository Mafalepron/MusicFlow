    94→/* ── Priority Bars (visual indicator) ──────────────────── */
    95→
    96→function PriorityBars({ priority, size = 'sm' }: { priority: string; size?: 'sm' | 'xs' }) {
    97→  const level = PRIORITY_LEVELS[priority] || 2;
    98→  const hex = PRIORITY_HEX[priority] || '#64748b';
    99→  const barW = size === 'xs' ? 'w-[2.5px]' : 'w-[3px]';
   100→  const gap = size === 'xs' ? 'gap-[1.5px]' : 'gap-[2px]';
   101→
   102→  return (
   103→    <div className={cn('flex items-end', gap)} style={{ height: size === 'xs' ? '10px' : '12px' }}>
   104→      {[1, 2, 3].map(i => (
   105→        <div
   106→          key={i}
   107→          className={cn(barW, 'rounded-sm transition-all duration-200')}
   108→          style={{
   109→            height: i === 1 ? '30%' : i === 2 ? '65%' : '100%',
   110→            backgroundColor: i <= level ? hex : hexToRgba(hex, 0.15),
   111→            boxShadow: i <= level ? `0 0 3px ${hex}60` : 'none',
   112→          }}
   113→        />
   114→      ))}
   115→    </div>
   116→  );
   117→}

   119→/* ── Priority Selector (popover) ──────────────────────── */

   123→  onChange,
   124→  size = 'sm',

   179→              <button

   198→                    {p.label}

   200→                  <span className="text-[9px] text-slate-600 ml-1">P{pLevel}</span>

   206→        </div>

   214→export default function DescriptionBottomPanel() {
   215→  const {
   216→    boardTasks, selectedTaskId, selectedBoardId,
   217→    selectedStageForPanel, setSelectedStageForPanel,
   218→    setBoardTasks, boards,
   219→  } = useKanbanStore();
   220→
   221→  const [isCollapsed, setIsCollapsed] = useState(false);
   222→  const [members, setMembers] = useState<GroupMember[]>([]);
   223→
   224→  // Bottom panel is ALWAYS pinned when a task is selected at the top.
   225→  // For track tasks → renders StagesList; for regular tasks → renders FlatSubtasksList.
   226→  const task = boardTasks.find(t => t.id === selectedTaskId) || null;
   227→  const taskId = task?.id || null;
   228→  const shouldShow = !!task;
   229→
   230→  const selectedBoard = boards.find(b => b.id === selectedBoardId);
   231→  const boardColor = selectedBoard?.color || '#00d9ff';
   232→
   233→  // Fetch group members for assignee dropdowns
   234→  useEffect(() => {
   235→    const gid = useAuthStore.getState().currentGroupId;
   236→    if (!gid) return;
   237→    let cancelled = false;
   238→    fetch(`/api/groups/${gid}/members`)
   239→      .then(r => r.json())
   240→      .then(d => {
   241→        if (!cancelled && Array.isArray(d)) setMembers(d);
   242→      })
   243→      .catch(() => {});
   244→    return () => { cancelled = true; };
   245→  }, []);
   246→
   247→  // Reset collapse state when switching tasks (prev-tracker pattern avoids useEffect-setState)
   248→  const [prevTaskId, setPrevTaskId] = useState<string | null>(taskId);
   249→  if (taskId !== prevTaskId) {
   250→    setPrevTaskId(taskId);
   251→    setIsCollapsed(false);
   252→  }
   253→
   254→  const reloadTasks = useCallback(async () => {
   255→    if (!selectedBoardId) return;
   256→    const isTracks = useKanbanStore.getState().boards.find(b => b.id === selectedBoardId)?.boardType === 'tracks';
   257→    const url = `/api/tasks?boardId=${selectedBoardId}${isTracks ? '&deep=true' : ''}`;
   258→    const res = await fetch(url);
   259→    const data = await res.json();
   260→    setBoardTasks(data.tasks);
   261→  }, [selectedBoardId, setBoardTasks]);
   262→
   263→  const c = useMemo<ColorSet>(() => ({
   264→    raw: boardColor,
   265→    a04: hexToRgba(boardColor, 0.04),
   266→    a08: hexToRgba(boardColor, 0.08),
   267→    a1: hexToRgba(boardColor, 0.1),

   275→    a6: hexToRgba(boardColor, 0.6),
   276→    a7: hexToRgba(boardColor, 0.7),
   277→  }), [boardColor]);
   278→
   279→  if (!shouldShow || !task) return null;
   280→
   281→  const stages = task.children || [];
   282→  const stagesCount = stages.length;
   283→  const allSubtasks = stages.flatMap(s => s.children || []);
   284→  const subtasksCount = allSubtasks.length;
   285→  // Progress is based on STAGE completion, not subtask completion
   286→  const doneStages = stages.filter(s => s.status === 'done').length;
   287→  const progress = stagesCount > 0 ? Math.round((doneStages / stagesCount) * 100) : 0;
   288→  const progressColor = progress === 100 ? '#34d399' : '#FCEE0A';
   289→
   290→  return (
   291→    <div
   292→      className="flex-shrink-0 flex flex-col cp-panel"
   293→      style={{

   300→
   301→      {/* Top neon border with glitch */}
   302→      <div className="cp-neon-top" style={{ background: `linear-gradient(90deg, transparent, ${c.raw} 20%, #FCEE0A 50%, ${c.raw} 80%, transparent)` }} />
   303→
   304→      {/* Header */}
   305→      <div className="cp-header" style={{ borderBottom: isCollapsed ? 'none' : `1px solid ${c.a3}` }}>
   306→        <button
   307→          onClick={() => setIsCollapsed(v => !v)}
   308→          className="flex items-center gap-2 min-w-0 flex-1 text-left group"
   309→          title={isCollapsed ? 'Развернуть' : 'Свернуть'}

   318→            {'// '}{task.title}

   732→/* ── Stages List (for track tasks) ──────────────────────── */
   733→
   734→function StagesList({
   735→  task, boardColor, c, members, selectedStageId, onSelectStage, reloadTasks,
   736→}: {
   737→  task: Task;
   738→  boardColor: string;
   739→  c: ColorSet;
   740→  members: GroupMember[];
   741→  selectedStageId: string | null;
   742→  onSelectStage: (stageId: string) => void;
   743→  reloadTasks: () => Promise<void>;
   744→}) {
   745→  const stages = task.children || [];
   746→  const [expandedStages, setExpandedStages] = useState<Record<string, boolean>>(() => {
   747→    // Auto-expand first stage or the selected one
   748→    const initial: Record<string, boolean> = {};
   749→    if (stages.length > 0) {
   750→      initial[stages[0].id] = true;
   751→    }
   752→    if (selectedStageId && stages.find(s => s.id === selectedStageId)) {
   753→      initial[selectedStageId] = true;
   754→    }
   755→    return initial;
   756→  });
   757→  const [addingStage, setAddingStage] = useState(false);
   758→  const [newStageTitle, setNewStageTitle] = useState('');
   759→
   760→  // If selectedStageId changes, ensure that stage is expanded (prev-tracker pattern avoids useEffect-setState)
   761→  const [prevSelectedStageId, setPrevSelectedStageId] = useState<string | null>(selectedStageId);
   762→  if (selectedStageId !== prevSelectedStageId) {
   763→    setPrevSelectedStageId(selectedStageId);
   764→    if (selectedStageId && stages.find(s => s.id === selectedStageId) && !expandedStages[selectedStageId]) {
   765→      setExpandedStages(prev => ({ ...prev, [selectedStageId]: true }));
   766→    }
   767→  }
   768→
   769→  const toggleExpand = (stageId: string) => {
   770→    setExpandedStages(prev => ({ ...prev, [stageId]: !prev[stageId] }));
   771→  };
   772→
   773→  const addStage = async () => {
   774→    if (!newStageTitle.trim()) return;
   775→    await fetch('/api/tasks', {
   776→      method: 'POST',
   777→      headers: { 'Content-Type': 'application/json' },
   778→      body: JSON.stringify({ title: newStageTitle.trim(), parentId: task.id, boardId: task.boardId }),
   779→    });
   780→    setNewStageTitle('');
   781→    setAddingStage(false);
   782→    await reloadTasks();
   783→  };
   784→
   785→  const moveStage = async (index: number, direction: -1 | 1) => {
   786→    const target = index + direction;
   787→    if (target < 0 || target >= stages.length) return;
   788→    const a = stages[index];
   789→    const b = stages[target];
   790→    // Swap hexR values
   791→    await Promise.all([
   792→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: a.id, hexR: b.hexR, hexQ: b.hexQ }) }),
   793→      fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: b.id, hexR: a.hexR, hexQ: a.hexQ }) }),
   794→    ]);
   795→    await reloadTasks();
   796→  };
   797→
   798→  const deleteStage = async (stageId: string) => {
   799→    await fetch(`/api/tasks?id=${stageId}`, { method: 'DELETE' });
   800→    await reloadTasks();
   801→  };
   802→
   803→  const updateStage = async (stageId: string, patch: Record<string, unknown>) => {
   804→    await fetch('/api/tasks', {
   805→      method: 'PUT',
   806→      headers: { 'Content-Type': 'application/json' },
   807→      body: JSON.stringify({ id: stageId, ...patch }),
   808→    });
   809→    await reloadTasks();
   810→  };
   811→
   812→  return (
   813→    <div className="py-2 space-y-1.5">
   814→      {stages.length === 0 && !addingStage && (
   815→        <div className="text-center py-6 px-4">
   816→          <ListChecks className="w-6 h-6 mx-auto mb-2 text-slate-700" />
   817→          <p className="text-[11px] text-slate-500 mb-1">Этапов пока нет</p>
   818→          <p className="text-[10px] text-slate-600 mb-3">Создайте первый этап производства трека</p>
   819→        </div>
   820→      )}
   821→
   822→      {stages.map((stage, idx) => (
   823→        <StageCard
   824→          key={stage.id}
   825→          stage={stage}
   826→          index={idx}
   827→          total={stages.length}
   828→          boardColor={boardColor}
   829→          c={c}
   830→          members={members}
   831→          isExpanded={!!expandedStages[stage.id]}
   832→          isSelected={selectedStageId === stage.id}
   833→          onToggleExpand={() => toggleExpand(stage.id)}
   834→          onSelect={() => onSelectStage(stage.id)}
   835→          onMoveUp={() => moveStage(idx, -1)}
   836→          onMoveDown={() => moveStage(idx, 1)}
   837→          onDelete={() => deleteStage(stage.id)}
   838→          onUpdate={(patch) => updateStage(stage.id, patch)}
   839→          reloadTasks={reloadTasks}
   840→        />
   841→      ))}
   842→
   843→      {/* Add stage form */}
   844→      {addingStage ? (
   845→        <div
   846→          className="mx-2 rounded-lg p-2.5 space-y-1.5"
   847→          style={{ border: `1px dashed ${c.a3}`, backgroundColor: c.a04 }}
   848→        >
   849→          <Input
   850→            value={newStageTitle}
   851→            onChange={(e) => setNewStageTitle(e.target.value)}
   852→            onKeyDown={(e) => {
   853→              if (e.key === 'Enter') void addStage();
   854→              if (e.key === 'Escape') { setAddingStage(false); setNewStageTitle(''); }
   855→            }}
   856→            placeholder="Название этапа..."
   857→            className="bg-slate-900/80 border-slate-700/50 text-[11px] text-slate-200 placeholder:text-slate-600 h-7"
   858→            style={{ borderColor: c.a3 }}
   859→            autoFocus
   860→          />
   861→          <div className="flex items-center gap-1.5">
   862→            <button
   863→              onClick={() => void addStage()}
   864→              disabled={!newStageTitle.trim()}
   865→              className="text-[10px] px-2 py-1 rounded font-medium transition-all disabled:opacity-40"
   866→              style={{ color: '#fff', backgroundColor: c.a5 }}
   867→            >
   868→              + Добавить этап
   869→            </button>
   870→            <button
   871→              onClick={() => { setAddingStage(false); setNewStageTitle(''); }}
   872→              className="text-[10px] text-slate-500 hover:text-slate-300 px-2 py-1"
   873→            >
   874→              Отмена
   875→            </button>
   876→          </div>
   877→        </div>
   878→      ) : (
   879→        <button
   880→          onClick={() => setAddingStage(true)}
   881→          className="cp-add-btn mx-2 w-[calc(100%-1rem)] justify-center"
   882→        >
   883→          <Plus className="w-3 h-3" /> Добавить этап
   884→        </button>
   885→      )}
   886→    </div>
   887→  );
   888→}

   899→  total: number;

   993→            value={titleDraft}

   995→            onKeyDown={(e) => {
   996→              if (e.key === 'Enter') { e.preventDefault(); void saveTitle(); }
   997→              if (e.key === 'Escape') { setIsEditingTitle(false); setTitleDraft(stage.title); }

   999→            onBlur={() => void saveTitle()}

  1001→            style={{ border: `1.5px solid ${c.a5}` }}

  1374→            value={titleDraft}

  1377→              if (e.key === 'Enter') { e.preventDefault(); void saveTitle(); }

  1506→
  1507→function FlatSubtasksList({
  1508→  task, boardColor, c, members, reloadTasks,
  1509→}: {
  1510→  task: Task;
  1511→  boardColor: string;
  1512→  c: ColorSet;
  1513→  members: GroupMember[];
  1514→  reloadTasks: () => Promise<void>;
  1515→}) {
  1516→  const subtasks = task.children || [];
  1517→  const [expandedSubs, setExpandedSubs] = useState<Record<string, boolean>>({});
  1518→  const [addingSub, setAddingSub] = useState(false);
  1519→  const [newSubTitle, setNewSubTitle] = useState('');
  1520→
  1521→  const toggleExpand = (id: string) => setExpandedSubs(prev => ({ ...prev, [id]: !prev[id] }));
  1522→
  1523→  const updateSubtask = async (subId: string, patch: Record<string, unknown>) => {
  1524→    await fetch('/api/tasks', {
  1525→      method: 'PUT',
  1526→      headers: { 'Content-Type': 'application/json' },
  1527→      body: JSON.stringify({ id: subId, ...patch }),
  1528→    });
  1529→    await reloadTasks();
  1530→  };
  1531→
  1532→  const moveSubtask = async (idx: number, dir: -1 | 1) => {
  1533→    const target = idx + dir;
  1534→    if (target < 0 || target >= subtasks.length) return;
  1535→    const a = subtasks[idx];

  1636→function parseAssignees(raw: string | null): string[] {
  1637→  if (!raw) return [];

  1639→    const parsed = JSON.parse(raw);
  1640→    if (Array.isArray(parsed)) return parsed.filter(Boolean);
  1641→    return [raw];
  1642→  } catch {
  1643→    return [raw];
  1644→  }
  1645→}

  1648→  if (ids.length === 0) return null;

  1840→async function moveSubtask(stage: TaskChild, idx: number, dir: -1 | 1, reloadTasks: () => Promise<void>) {
  1841→  const subtasks = stage.children || [];
  1842→  const target = idx + dir;
  1843→  if (target < 0 || target >= subtasks.length) return;
  1844→  const a = subtasks[idx];
  1845→  const b = subtasks[target];
  1846→  await Promise.all([
  1847→    fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: a.id, hexR: b.hexR, hexQ: b.hexQ }) }),
  1848→    fetch('/api/tasks', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ id: b.id, hexR: a.hexR, hexQ: a.hexQ }) }),
  1849→  ]);
  1850→  await reloadTasks();
  1851→}
