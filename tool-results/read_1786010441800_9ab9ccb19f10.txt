   106→  const config = task.trackConfig ? JSON.parse(task.trackConfig) : null;

   422→/* ── Track Text / Lyrics Section ──────────────────────── */
   423→
   424→function TrackTextSection({
   425→  task,
   426→  boardColor,
   427→  onUpdate,
   428→}: {
   429→  task: Task;
   430→  boardColor: string;
   431→  onUpdate: (patch: Record<string, unknown>) => Promise<void>;
   432→}) {
   433→  const config = task.trackConfig ? JSON.parse(task.trackConfig) : null;
   434→  const trackText: string = config?.text || '';
   435→  const [isEditing, setIsEditing] = useState(false);
   436→  const [isExpanded, setIsExpanded] = useState(false);
   437→  const [draft, setDraft] = useState(trackText);
   438→
   439→  const save = async () => {
   440→    const newConfig = { ...config, text: draft.trim() };
   441→    await onUpdate({ trackConfig: JSON.stringify(newConfig) });
   442→    setIsEditing(false);
   443→  };
   444→
   445→  const cancel = () => {
   446→    setDraft(trackText);
   447→    setIsEditing(false);
   448→  };
   449→
   450→  return (
   451→    <div className="px-4 py-3">
   452→      <div className="flex items-center justify-between mb-2">
   453→        <span
   454→          className="text-[9px] uppercase tracking-widest font-medium"
   455→          style={{ color: hexToRgba(boardColor, 0.55) }}
   456→        >
   457→          Текст трека
   458→        </span>
   459→        <div className="flex items-center gap-1">
   460→          {!isEditing && trackText && (
   461→            <button
   462→              onClick={() => setIsExpanded(v => !v)}
   463→              className="text-[9px] px-1.5 py-0.5 rounded transition-colors text-slate-500 hover:text-slate-300"
   464→            >
   465→              {isExpanded ? 'Свернуть' : 'Читать'}
   466→            </button>
   467→          )}
   468→          {!isEditing && (
   469→            <button
   470→              onClick={() => { setDraft(trackText); setIsEditing(true); }}
   471→              className="text-[9px] px-1.5 py-0.5 rounded transition-colors flex items-center gap-1"
   472→              style={{ color: hexToRgba(boardColor, 0.5) }}
   473→            >
   474→              <Pencil className="w-2.5 h-2.5" /> Изменить
   475→            </button>
   476→          )}
   477→        </div>
   478→      </div>
   479→
   480→      {isEditing ? (
   481→        <div className="space-y-2">
   482→          <Textarea
   483→            value={draft}
   484→            onChange={(e) => setDraft(e.target.value)}
   485→            placeholder="Введите текст трека или лирику..."
   486→            className="bg-slate-900/80 text-[11px] text-slate-300 placeholder:text-slate-600 min-h-[120px] resize-none focus:outline-none rounded-md"
   487→            style={{ borderColor: hexToRgba(boardColor, 0.3) }}
   488→            autoFocus

   947→                </SelectTrigger>
   948→                <SelectContent className="bg-slate-900 border-slate-700 z-[60]">{PRIORITIES.map((p) => (<SelectItem key={p.value} value={p.value}><div className="flex items-center gap-1.5"><div className={cn('w-1.5 h-1.5 rounded-full', p.dot)} />{p.label}</div></SelectItem>))}</SelectContent>
   949→              </Select>

   959→
   960→        {/* Actions */}
   961→        <div className="flex gap-2 pt-2">
   962→          <button
   963→            onClick={() => void handleSave()}

   967→          >
   968→            <Save className="w-3 h-3" />{saving ? '...' : mode === 'edit' ? 'Сохранить' : 'Создать'}
   969→          </button>

   972→              onClick={() => void handleDelete()}

   976→            >

   978→            </button>
   979→          )}
   980→        </div>
   981→      </div>
   982→    </div>
   983→  );
   984→}

   988→function getProgress(children: { status: string }[]): number {
   989→  if (children.length === 0) return 0;
   990→  return Math.round((children.filter(c => c.status === 'done').length / children.length) * 100);
