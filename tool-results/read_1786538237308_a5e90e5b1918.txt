    98→    98→// Build a nested tree from flat comment list
    99→    99→interface CommentNode extends Comment {
   100→   100→  replies: CommentNode[];
   101→   101→}
   102→   102→
   103→   103→function buildCommentTree(comments: Comment[]): CommentNode[] {
   104→   104→  const map = new Map<string, CommentNode>();
   105→   105→  const roots: CommentNode[] = [];
   106→   106→  
   107→   107→  for (const c of comments) {
   108→   108→    map.set(c.id, { ...c, replies: [] });
   109→   109→  }
   110→   110→  for (const c of comments) {
   111→   111→    const node = map.get(c.id)!;
   112→   112→    if (c.parentId && map.has(c.parentId)) {
   113→   113→      map.get(c.parentId)!.replies.push(node);
   114→   114→    } else {
   115→   115→      roots.push(node);
   116→   116→    }
   117→   117→  }
   118→   118→  roots.sort((a, b) => a.timestampMs - b.timestampMs);
   119→   119→  for (const r of roots) {
   120→   120→    r.replies.sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   121→   121→  }
   122→   122→  return roots;
   123→   123→}

   144→   144→function formatTimestamp(ms: number): string {
   145→   145→  const totalSeconds = ms / 1000;
   146→   146→  const minutes = Math.floor(totalSeconds / 60);
   147→   147→  const seconds = Math.floor(totalSeconds % 60);
   148→   148→  const milliseconds = Math.floor((totalSeconds % 1) * 10);
   149→   149→  return `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}.${milliseconds}`;
   150→   150→}
   151→   151→
   152→   152→function formatDuration(seconds: number): string {
   153→   153→  const mins = Math.floor(seconds / 60);
   154→   154→  const secs = Math.floor(seconds % 60);
   155→   155→  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   156→   156→}

   158→   158→function getInitials(name?: string | null): string {
   159→   159→  if (!name) return '?';
   160→   160→  return name
   161→   161→    .trim()
   162→   162→    .split(' ')
   163→   163→    .map((n) => n[0])
   164→   164→    .join('')
   165→   165→    .toUpperCase()
   166→   166→    .slice(0, 2);
   167→   167→}
