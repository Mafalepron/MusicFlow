     8→     8→import {
     9→     9→  ChevronRight, ChevronLeft, Check, Music, Guitar, Mic2,

    13→    13→

    29→    29→  description: string;
    30→    30→  subtasks: EditableSubtask[];
    31→    31→  sourceKey?: string; // template key if added from template, undefined if custom
    32→    32→}
    33→    33→
    34→    34→const CUSTOM_ICONS: Record<string, typeof Music> = {

    42→    42→

    44→    44→  'Сонграйтинг': ['Написание текста', 'Демо'],

    46→    46→  'Редактура': ['Компинг', 'Тюнинг вокала', 'Тайминг'],

    48→    48→  'Мастеринг': ['Финальная громкость и лимитирование', 'Стерео-обработка'],
    49→    49→};

    51→    51→let _uid = 0;
    52→    52→const uid = () => `es_${++_uid}_${Date.now()}`;
    53→    53→
    54→    54→export default function TrackWizard() {
    55→    55→  const { boards, selectedBoardId, setBoardTasks, setIsTrackWizardOpen, setSelectedTaskId, setTrackWizardStep } = useKanbanStore();
    56→    56→  const boardColor = boards.find(b => b.id === selectedBoardId)?.color || '#00d9ff';
    57→    57→  const bc = boardColorStyles(boardColor);
    58→    58→  const [step, setStep] = useState(0); // 0=name, 1=instruments, 2=stages+configure
    59→    59→
    60→    60→  // Sync wizard step to store for onboarding
    61→    61→  useEffect(() => { setTrackWizardStep(step); return () => setTrackWizardStep(-1); }, [step, setTrackWizardStep]);
    62→    62→
    63→    63→  const [trackName, setTrackName] = useState('');
    64→    64→  const [trackDescription, setTrackDescription] = useState('');
    65→    65→  const [selectedInstruments, setSelectedInstruments] = useState<string[]>([]);
    66→    66→  const [customInstrument, setCustomInstrument] = useState('');
    67→    67→  const [wizardData, setWizardData] = useState<WizardData | null>(null);
    68→    68→  const [loading, setLoading] = useState(false);
    69→    69→  const [allInstruments, setAllInstruments] = useState<string[]>([]);
    70→    70→
    71→    71→  // Editable stages (shared state for merged step 2)
    72→    72→  const [editableStages, setEditableStages] = useState<EditableStage[]>([]);
    73→    73→  const [expandedStageId, setExpandedStageId] = useState<string | null>(null);
    74→    74→  const [addingSubtaskForStage, setAddingSubtaskForStage] = useState<string | null>(null);
    75→    75→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
    76→    76→  const [newSubtaskDesc, setNewSubtaskDesc] = useState('');
    77→    77→  const [newStageLabel, setNewStageLabel] = useState('');
    78→    78→  const [newStageDesc, setNewStageDesc] = useState('');
    79→    79→  const [showNewStageForm, setShowNewStageForm] = useState(false);
    80→    80→  const [showTemplates, setShowTemplates] = useState(true);
    81→    81→
    82→    82→  useEffect(() => {
    83→    83→    fetch('/api/tracks/wizard')
    84→    84→      .then(r => r.json())
    85→    85→      .then(data => {
    86→    86→        setWizardData(data);
    87→    87→        setAllInstruments([...data.instruments]);
    88→    88→      });
    89→    89→  }, []);
    90→    90→
    91→    91→  // --- Template helpers ---
    92→    92→  const isTemplateAdded = (key: string) => editableStages.some(s => s.sourceKey === key);
    93→    93→
    94→    94→  const toggleTemplate = (key: string) => {
    95→    95→    if (isTemplateAdded(key)) {
    96→    96→      // Remove template stage
    97→    97→      setEditableStages(prev => prev.filter(s => s.sourceKey !== key));
    98→    98→    } else {
    99→    99→      // Add template stage with default subtasks
   100→   100→      const tpl = wizardData?.stages.find(s => s.key === key);
   101→   101→      if (!tpl) return;
   102→   102→      const subtaskTitles = tpl.hasInstrumentBased && selectedInstruments.length > 0
   103→   103→        ? selectedInstruments.map(i => `Запись: ${i}`)
   104→   104→        : STAGE_SUBTASKS[key] || [];
   105→   105→      const newStage: EditableStage = {
   106→   106→        id: uid(),
   107→   107→        emoji: tpl.emoji,
   108→   108→        label: tpl.label,
   109→   109→        description: '',
   110→   110→        subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
   111→   111→        sourceKey: key,
   112→   112→      };
   113→   113→      setEditableStages(prev => [...prev, newStage]);
   114→   114→      setExpandedStageId(newStage.id);
   115→   115→    }
   116→   116→  };
   117→   117→
   118→   118→  const addAllTemplates = () => {
   119→   119→    if (!wizardData) return;
   120→   120→    const allAdded = wizardData.stages.every(s => isTemplateAdded(s.key));
   121→   121→    if (allAdded) {
   122→   122→      // Remove all template stages
   123→   123→      setEditableStages(prev => prev.filter(s => !s.sourceKey));
   124→   124→    } else {
   125→   125→      // Add all missing templates
   126→   126→      const newStages: EditableStage[] = [];
   127→   127→      for (const tpl of wizardData.stages) {
   128→   128→        if (!isTemplateAdded(tpl.key)) {
   129→   129→          const subtaskTitles = tpl.hasInstrumentBased && selectedInstruments.length > 0
   130→   130→            ? selectedInstruments.map(i => `Запись: ${i}`)
   131→   131→            : STAGE_SUBTASKS[tpl.key] || [];
   132→   132→          newStages.push({
   133→   133→            id: uid(),
   134→   134→            emoji: tpl.emoji,
   135→   135→            label: tpl.label,
   136→   136→            description: '',
   137→   137→            subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
   138→   138→            sourceKey: tpl.key,
   139→   139→          });
   140→   140→        }
   141→   141→      }
   142→   142→      setEditableStages(prev => [...prev, ...newStages]);
   143→   143→    }
   144→   144→  };
   145→   145→
   146→   146→  // Re-generate instrument-based subtasks when instruments change
   147→   147→  const updateInstrumentSubtasks = useCallback(() => {
   148→   148→    if (!wizardData) return;
   149→   149→    const recordStage = wizardData.stages.find(s => s.key === 'Запись');
   150→   150→    if (!recordStage || !isTemplateAdded('Запись')) return;
   151→   151→    const instSubtasks = selectedInstruments.map(i => `Запись: ${i}`);
   152→   152→    setEditableStages(prev => prev.map(s => {
   153→   153→      if (s.sourceKey !== 'Запись') return s;
   154→   154→      // Keep existing subtasks that still match an instrument, add new ones
   155→   155→      const existing = s.subtasks.filter(st => instSubtasks.includes(st.title));
   156→   156→      const finalSubs: EditableSubtask[] = instSubtasks.map(title => {
   157→   157→        const ex = existing.find(e => e.title === title);
   158→   158→        return ex || { id: uid(), title, description: '' };
   159→   159→      });
   160→   160→      return { ...s, subtasks: finalSubs };
   161→   161→    }));
   162→   162→  }, [wizardData, selectedInstruments]);
   163→   163→
   164→   164→  // When entering step 2, ensure instrument subtasks are synced
   165→   165→  const goToStep2 = () => {
   166→   166→    updateInstrumentSubtasks();
   167→   167→    setStep(2);
   168→   168→  };
   169→   169→
   170→   170→  // --- Instrument helpers ---
   171→   171→  const toggleInstrument = (inst: string) => {
   172→   172→    setSelectedInstruments(prev =>
   173→   173→      prev.includes(inst) ? prev.filter(i => i !== inst) : [...prev, inst]
   174→   174→    );
   175→   175→  };
   176→   176→
   177→   177→  const addCustomInstrument = () => {
   178→   178→    const trimmed = customInstrument.trim();
   179→   179→    if (!trimmed || allInstruments.includes(trimmed)) return;
   180→   180→    setAllInstruments(prev => [...prev, trimmed]);
   181→   181→    setSelectedInstruments(prev => [...prev, trimmed]);
   182→   182→    setCustomInstrument('');
   183→   183→  };
   184→   184→
   185→   185→  // --- Stage editing helpers ---
   186→   186→  const updateStage = (stageId: string, updates: Partial<Pick<EditableStage, 'label' | 'description' | 'emoji'>>) => {
   187→   187→    setEditableStages(prev => prev.map(s => s.id === stageId ? { ...s, ...updates } : s));
   188→   188→  };
   189→   189→
   190→   190→  const removeStage = (stageId: string) => {
   191→   191→    setEditableStages(prev => prev.filter(s => s.id !== stageId));
   192→   192→  };
   193→   193→
   194→   194→  const addNewStage = () => {
   195→   195→    if (!newStageLabel.trim()) return;
   196→   196→    const newStage: EditableStage = {
   197→   197→      id: uid(),
   198→   198→      emoji: DEFAULT_STAGE_EMOJIS[editableStages.length % DEFAULT_STAGE_EMOJIS.length],
   199→   199→      label: newStageLabel.trim(),
   200→   200→      description: newStageDesc.trim(),
   201→   201→      subtasks: [],
   202→   202→    };
   203→   203→    setEditableStages(prev => [...prev, newStage]);
   204→   204→    setNewStageLabel('');
   205→   205→    setNewStageDesc('');
   206→   206→    setShowNewStageForm(false);
   207→   207→    setExpandedStageId(newStage.id);
   208→   208→  };
   209→   209→
   210→   210→  const updateSubtask = (stageId: string, subtaskId: string, updates: Partial<Pick<EditableSubtask, 'title' | 'description'>>) => {
   211→   211→    setEditableStages(prev => prev.map(s => {
   212→   212→      if (s.id !== stageId) return s;
   213→   213→      return { ...s, subtasks: s.subtasks.map(st => st.id === subtaskId ? { ...st, ...updates } : st) };
   214→   214→    }));
   215→   215→  };
   216→   216→
   217→   217→  const removeSubtask = (stageId: string, subtaskId: string) => {
   218→   218→    setEditableStages(prev => prev.map(s => {
   219→   219→      if (s.id !== stageId) return s;
   220→   220→      return { ...s, subtasks: s.subtasks.filter(st => st.id !== subtaskId) };
   221→   221→    }));
   222→   222→  };
   223→   223→
   224→   224→  const addSubtask = (stageId: string) => {
   225→   225→    if (!newSubtaskTitle.trim()) return;
   226→   226→    setEditableStages(prev => prev.map(s => {
   227→   227→      if (s.id !== stageId) return s;
   228→   228→      return { ...s, subtasks: [...s.subtasks, { id: uid(), title: newSubtaskTitle.trim(), description: newSubtaskDesc.trim() }] };
   229→   229→    }));
   230→   230→    setNewSubtaskTitle('');
   231→   231→    setNewSubtaskDesc('');
   232→   232→    setAddingSubtaskForStage(null);
   233→   233→  };
   234→   234→
   235→   235→  // --- Create ---
   236→   236→  const handleCreate = useCallback(async () => {
   237→   237→    if (!trackName.trim() || !selectedBoardId) return;
   238→   238→    setLoading(true);
   239→   239→    try {
   240→   240→      const res = await fetch('/api/tracks/wizard', {
   241→   241→        method: 'POST',
   242→   242→        headers: { 'Content-Type': 'application/json' },
   243→   243→        body: JSON.stringify({
   244→   244→          title: trackName.trim(),
   245→   245→          description: trackDescription.trim(),
   246→   246→          instruments: selectedInstruments,
   247→   247→          boardId: selectedBoardId,
   248→   248→          customStages: editableStages.map(s => ({
   249→   249→            emoji: s.emoji,
   250→   250→            label: s.label,
   251→   251→            description: s.description,
   252→   252→            subtasks: s.subtasks.map(st => ({ title: st.title, description: st.description })),
   253→   253→          })),
   254→   254→        }),
   255→   255→      });
   256→   256→      const data = await res.json();
   257→   257→      const taskRes = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   258→   258→      const taskData = await taskRes.json();
   259→   259→      setBoardTasks(taskData.tasks);
   260→   260→      if (data.trackId) setSelectedTaskId(data.trackId);
   261→   261→      setIsTrackWizardOpen(false);
   262→   262→    } finally {
   263→   263→      setLoading(false);
   264→   264→    }
   265→   265→  }, [trackName, trackDescription, selectedInstruments, selectedBoardId, editableStages, setBoardTasks, setSelectedTaskId, setIsTrackWizardOpen]);
   266→   266→
   267→   267→  const canProceed = () => {
   268→   268→    if (step === 0) return trackName.trim().length > 0;
   269→   269→    if (step === 1) return selectedInstruments.length > 0;
   270→   270→    if (step === 2) return editableStages.length > 0;
   271→   271→    return true;
   272→   272→  };
   273→   273→
   274→   274→  const handleClose = () => setIsTrackWizardOpen(false);
   275→   275→
   276→   276→  const STEPS = ['Название', 'Инструменты', 'Этапы'];
   277→   277→
   278→   278→  return (
   279→   279→    <div className="flex flex-col h-full">
   280→   280→      {/* Header */}
   281→   281→      <div className="border-b border-slate-800/50 p-4 shrink-0">
   282→   282→        <div className="flex items-center justify-between mb-3">
   283→   283→          <div className="flex items-center gap-2">
   284→   284→            <div className="w-8 h-8 rounded-lg flex items-center justify-center" style={{ background: bc.gradient }}>
   285→   285→              <Music className="w-4 h-4 text-white" />
   286→   286→            </div>
   287→   287→            <div>
   288→   288→              <h3 className="text-xs font-bold text-white uppercase tracking-wider">Конструктор трека</h3>
   289→   289→              <p className="text-[10px] text-slate-500">Мастер создания пайплайна</p>
   290→   290→            </div>
   291→   291→          </div>
   292→   292→          <button onClick={handleClose} className="p-1 rounded hover:bg-slate-800 text-slate-500 hover:text-slate-300 transition-colors">
   293→   293→            <X className="w-4 h-4" />
   294→   294→          </button>
   295→   295→        </div>
   296→   296→
   297→   297→        {/* Progress steps */}
   298→   298→        <div className="flex items-center gap-1">
   299→   299→          {STEPS.map((s, i) => (
   300→   300→            <div key={s} className="flex items-center flex-1">
   301→   301→              <button
   302→   302→                onClick={() => i < step && setStep(i)}
   303→   303→                className={cn(
   304→   304→                  'flex items-center gap-1.5 text-[10px] font-medium px-2 py-1 rounded-md transition-all duration-200',
   305→   305→                  i < step && 'text-slate-400 hover:bg-slate-800/60 cursor-pointer',
   306→   306→                  i > step && 'text-slate-600',
   307→   307→                )}
   308→   308→                style={i === step ? { backgroundColor: bc.bg15, color: bc.text } : undefined}
   309→   309→              >
   310→   310→                {i < step ? <Check className="w-3 h-3" /> : <span className="w-4 h-4 rounded-full border text-center text-[9px] flex items-center justify-center" style={{
   311→   311→                  borderColor: i === step ? bc.text : i < step ? bc.text : '#334155',
   312→   312→                }}>{i + 1}</span>}
   313→   313→                <span className="hidden lg:inline">{s}</span>
   314→   314→              </button>
   315→   315→              {i < STEPS.length - 1 && (
   316→   316→                <div className="flex-1 h-px mx-1" style={{ backgroundColor: i < step ? bc.bg30 : '#1e293b' }} />
   317→   317→              )}
   318→   318→            </div>
   319→   319→          ))}
   320→   320→        </div>
   321→   321→      </div>
   322→   322→
   323→   323→      {/* Content */}
   324→   324→      <div className="flex-1 overflow-y-auto p-4">
   325→   325→        {/* Step 0: Name */}
   326→   326→        {step === 0 && (
   327→   327→          <div className="space-y-3">
   328→   328→            <div className="space-y-1.5">
   329→   329→              <label className="text-[11px] text-slate-400 font-medium">Название трека</label>
   330→   330→              <Input
   331→   331→                value={trackName}
   332→   332→                onChange={(e) => setTrackName(e.target.value)}
   333→   333→                onKeyDown={(e) => e.key === 'Enter' && canProceed() && setStep(1)}
   334→   334→                placeholder="Лирика, Эксперимент..."
   335→   335→                autoFocus
   336→   336→                className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 h-10"
   337→   337→                style={{ '--focus-border-color': bc.bg80 } as React.CSSProperties}
   338→   338→                onFocus={(e) => e.currentTarget.style.borderColor = bc.bg80}
   339→   339→                onBlur={(e) => e.currentTarget.style.borderColor = ''}
   340→   340→              />
   341→   341→            </div>
   342→   342→            <div className="space-y-1.5">
   343→   343→              <label className="text-[11px] text-slate-400 font-medium">Описание</label>
   344→   344→              <Textarea
   345→   345→                value={trackDescription}
   346→   346→                onChange={(e) => setTrackDescription(e.target.value)}
   347→   347→                placeholder="Краткое описание идеи трека, настроение, жанр..."
   348→   348→                rows={3}
   349→   349→                className="bg-slate-900/80 border-slate-700/50 text-sm text-slate-200 placeholder:text-slate-600 resize-none"
   350→   350→                onFocus={(e) => e.currentTarget.style.borderColor = bc.bg80}
   351→   351→                onBlur={(e) => e.currentTarget.style.borderColor = ''}
   352→   352→              />
   353→   353→            </div>
   354→   354→            <div className="rounded-lg bg-slate-900/40 border border-slate-800/30 p-3">
   355→   355→              <div className="flex items-center gap-2 mb-2">
   356→   356→                <Sparkles className="w-3.5 h-3.5" style={{ color: bc.text }} />
   357→   357→                <span className="text-[11px] font-medium" style={{ color: bc.text }}>Подсказка</span>
   358→   358→              </div>
   359→   359→              <p className="text-[10px] text-slate-500 leading-relaxed">
   360→   360→                После ввода названия и описания вы выберете инструменты и этапы производства.
   361→   361→                Система автоматически создаст структуру задач для вашего трека.
   362→   362→              </p>
   363→   363→            </div>
   364→   364→          </div>
   365→   365→        )}
   366→   366→
   367→   367→        {/* Step 1: Instruments */}
   368→   368→        {step === 1 && (
   369→   369→          <div className="space-y-3">
   370→   370→            <div className="flex items-center justify-between">
   371→   371→              <span className="text-[11px] text-slate-400 font-medium">
   372→   372→                Инструменты <span className="text-slate-600">({selectedInstruments.length} выбрано)</span>
   373→   373→              </span>
   374→   374→            </div>
   375→   375→            <div className="flex flex-wrap gap-1.5">
   376→   376→              {allInstruments.map((inst) => {
   377→   377→                const IconComp = CUSTOM_ICONS[inst];
   378→   378→                const isSelected = selectedInstruments.includes(inst);
   379→   379→                return (
   380→   380→                  <button
   381→   381→                    key={inst}
   382→   382→                    onClick={() => toggleInstrument(inst)}
   383→   383→                    className={cn(
   384→   384→                      'flex items-center gap-1.5 text-[11px] px-2.5 py-1.5 rounded-lg border transition-all duration-150',
   385→   385→                      !isSelected && 'bg-slate-900/60 border-slate-800/50 text-slate-400 hover:border-slate-700 hover:text-slate-300',
   386→   386→                    )}
   387→   387→                    style={isSelected ? {
   388→   388→                      backgroundColor: bc.bg15,
   389→   389→                      borderColor: bc.bg40,
   390→   390→                      color: bc.text,
   391→   391→                      boxShadow: `0 1px 2px 0 ${bc.bg20}`,
   392→   392→                    } : undefined}
   393→   393→                  >
   394→   394→                    {IconComp ? <IconComp className="w-3 h-3" /> : <Music className="w-3 h-3" />}
   395→   395→                    {inst}
   396→   396→                    {isSelected && <Check className="w-3 h-3 ml-0.5" />}
   397→   397→                  </button>
   398→   398→                );
   399→   399→              })}
   400→   400→            </div>
   401→   401→            <div className="flex gap-2 pt-1">
   402→   402→              <Input
