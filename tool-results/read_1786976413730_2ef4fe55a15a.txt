    90→  // Quick-access project IDs — same source as the home page's "Быстрый доступ"
    91→  // section. Read from localStorage so the header panel always shows the same
    92→  // projects the user pinned on the home page.

    94→  // so the value is re-read every time the panel opens. This is cheap

    96→  // lint rule that fires when you call setState inside a useEffect.

   101→    try {
   102→      const raw = localStorage.getItem('soundflow-quick-access');
   103→      return raw ? (JSON.parse(raw) as string[]) : [];

   106→

   108→  useEffect(() => {
   109→    if (!currentGroupId) return;
   110→    fetch(`/api/groups/${currentGroupId}/members`)
   111→      .then((r) => r.json())
   112→      .then((m) => setMemberCount(Array.isArray(m) ? m.length : 0))
   113→      .catch(() => {});
   114→  }, [currentGroupId]);
   115→

   117→  // Needed because quick-access IDs can point to either auto projects or

   121→    fetch('/api/tasks?parentId=null')
   122→      .then((r) => r.json())
   123→      .then((data) => {
   124→        const tasks: Task[] = Array.isArray(data) ? data : data.tasks || [];

   126→        useKanbanStore.getState().setProjects(tasks);
   127→      })
   128→      .catch(() => {});
   129→  }, []);
   130→

   132→  // Maps the localStorage `quickAccess` IDs to auto projects + kanban projects.

   135→    type Card =
   136→      | { kind: 'auto'; id: string; title: string; type: string; status: string; trackCount: number }
   137→      | { kind: 'kanban'; id: string; title: string; type: string; status: string; boardCount: number };

   139→    quickAccessIds.forEach((id) => {
   140→      const autoP = autoProjects.find((p) => p.id === id);
   141→      if (autoP) {
   142→        out.push({
   143→          kind: 'auto',
   144→          id: autoP.id,
   145→          title: autoP.title,
   146→          type: autoP.type,
   147→          status: autoP.status,
   148→          trackCount: tracks.filter((t) => t.projectId === autoP.id).length,
   149→        });
   150→        return;
   151→      }
   152→      const kanbanT = kanbanProjects.find((t) => t.id === id);
   153→      if (kanbanT) {
   154→        out.push({
   155→          kind: 'kanban',
   156→          id: kanbanT.id,
   157→          title: kanbanT.title,
   158→          type: kanbanT.projectType || 'general',
   159→          status: kanbanT.status,
   160→          boardCount: kanbanT.children?.length ?? 0,
   161→        });
   162→      }
   163→    });
   164→    return out;
   165→  }, [quickAccessIds, autoProjects, kanbanProjects, tracks]);
   166→

   174→      const target = e.target as Node | null;
   175→      if (!target) return;
   176→      if (
   177→        quickPanelRef.current && !quickPanelRef.current.contains(target) &&
   178→        quickStripeRef.current && !quickStripeRef.current.contains(target)
   179→      ) {
   180→        setQuickPanelOpen(false);
   181→      }
   182→    };

   184→      if (e.key === 'Escape') setQuickPanelOpen(false);
   185→    };

   187→    document.addEventListener('keydown', handleKey);
   188→    return () => {
   189→      document.removeEventListener('mousedown', handleClick);
   190→      document.removeEventListener('keydown', handleKey);
   191→    };

   205→          setNotifications(data);

   217→        setNotifOpen(false);
   218→      }
   219→    };

   221→    return () => document.removeEventListener('mousedown', handler);

   289→    setNotifications(notifications.map(n => ({ ...n, isRead: true })));

   372→          />

   374→
   375→        {/* Quick-access slide-down panel — rendered OUTSIDE the header (after

   407→        {/* Logo — mobile only (desktop has sidebar) */}

   410→            {/* Holographic double-ring icon */}

   488→            <TooltipTrigger asChild>
   489→              <Button
   490→                variant="ghost"
