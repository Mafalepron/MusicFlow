     1→'use client';
     2→
     3→import { useState, useEffect, useCallback } from 'react';
     4→import { useKanbanStore } from '@/store/kanban-store';
     5→import { useAuthStore } from '@/lib/store';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { Textarea } from '@/components/ui/textarea';
     9→import {
    10→  ChevronRight, ChevronLeft, Check, Music, Guitar, Mic2,
    11→  Sparkles, X, Loader2, Plus, Trash2, FileText,
    12→} from 'lucide-react';
    13→import { cn, boardColorStyles } from '@/lib/utils';
    14→import DeadlinePicker from '@/components/kanban/deadline-picker';
    15→
    16→interface WizardData {
    17→  instruments: string[];
    18→  stages: { key: string; label: string; emoji: string; hasInstrumentBased: boolean; defaultSubtaskCount: number }[];
    19→}
    20→
    21→interface EditableSubtask {
    22→  id: string;
    23→  title: string;
    24→  description: string;
    25→}
    26→
    27→interface EditableStage {
    28→  id: string;
    29→  emoji: string;
    30→  label: string;
    31→  description: string;
    32→  subtasks: EditableSubtask[];
    33→  sourceKey?: string; // template key if added from template, undefined if custom
    34→}
    35→
    36→const CUSTOM_ICONS: Record<string, typeof Music> = {
    37→  'Вокал (основной)': Mic2,
    38→  'Бэк-вокал': Mic2,
    39→  'Гитара (акустическая)': Guitar,
    40→  'Гитара (электро)': Guitar,
    41→};
    42→
    43→const DEFAULT_STAGE_EMOJIS = ['📁', '🎯', '⚙️', '🔧', '🎨', '📊', '💡', '📋', '🧩', '🚀'];
    44→
    45→const STAGE_SUBTASKS: Record<string, string[]> = {
    46→  'Сонграйтинг': ['Написание текста', 'Демо'],
    47→  'Аранжировка': ['Гармоническая структура', 'Тембровая расстановка'],
    48→  'Редактура': ['Компинг', 'Тюнинг вокала', 'Тайминг'],
    49→  'Сведение': ['Баланс и эквализация', 'Пространственная обработка', 'Эффекты и автоматизация'],
    50→  'Мастеринг': ['Финальная громкость и лимитирование', 'Стерео-обработка'],
    51→};
    52→
    53→let _uid = 0;
    54→const uid = () => `es_${++_uid}_${Date.now()}`;
    55→
    56→export default function TrackWizard() {
    57→  const { boards, selectedBoardId, setBoardTasks, setIsTrackWizardOpen, setSelectedTaskId, setTrackWizardStep } = useKanbanStore();
    58→  const user = useAuthStore((s) => s.user);
    59→  const boardColor = boards.find(b => b.id === selectedBoardId)?.color || '#00d9ff';
    60→  const bc = boardColorStyles(boardColor);
    61→  const [step, setStep] = useState(0); // 0=name, 1=instruments, 2=stages+configure
    62→
    63→  // Sync wizard step to store for onboarding
    64→  useEffect(() => { setTrackWizardStep(step); return () => setTrackWizardStep(-1); }, [step, setTrackWizardStep]);
    65→
    66→  const [trackName, setTrackName] = useState('');
    67→  const [trackDescription, setTrackDescription] = useState('');
    68→  const [trackDeadline, setTrackDeadline] = useState<string | null>(null);
    69→  const [selectedInstruments, setSelectedInstruments] = useState<string[]>([]);
    70→  const [customInstrument, setCustomInstrument] = useState('');
    71→  const [wizardData, setWizardData] = useState<WizardData | null>(null);
    72→  const [loading, setLoading] = useState(false);
    73→  const [allInstruments, setAllInstruments] = useState<string[]>([]);
    74→
    75→  // Editable stages (shared state for merged step 2)
    76→  const [editableStages, setEditableStages] = useState<EditableStage[]>([]);
    77→  const [expandedStageId, setExpandedStageId] = useState<string | null>(null);
    78→  const [addingSubtaskForStage, setAddingSubtaskForStage] = useState<string | null>(null);
    79→  const [newSubtaskTitle, setNewSubtaskTitle] = useState('');
    80→  const [newSubtaskDesc, setNewSubtaskDesc] = useState('');
    81→  const [newStageLabel, setNewStageLabel] = useState('');
    82→  const [newStageDesc, setNewStageDesc] = useState('');
    83→  const [showNewStageForm, setShowNewStageForm] = useState(false);
    84→  const [showTemplates, setShowTemplates] = useState(true);
    85→
    86→  useEffect(() => {
    87→    fetch('/api/tracks/wizard')
    88→      .then(r => r.json())
    89→      .then(data => {
    90→        setWizardData(data);
    91→        setAllInstruments([...data.instruments]);
    92→      });
    93→  }, []);
    94→
    95→  // --- Template helpers ---
    96→  const isTemplateAdded = (key: string) => editableStages.some(s => s.sourceKey === key);
    97→
    98→  const toggleTemplate = (key: string) => {
    99→    if (isTemplateAdded(key)) {
   100→      // Remove template stage
   101→      setEditableStages(prev => prev.filter(s => s.sourceKey !== key));
   102→    } else {
   103→      // Add template stage with default subtasks
   104→      const tpl = wizardData?.stages.find(s => s.key === key);
   105→      if (!tpl) return;
   106→      const subtaskTitles = tpl.hasInstrumentBased && selectedInstruments.length > 0
   107→        ? selectedInstruments.map(i => `Запись: ${i}`)
   108→        : STAGE_SUBTASKS[key] || [];
   109→      const newStage: EditableStage = {
   110→        id: uid(),
   111→        emoji: tpl.emoji,
   112→        label: tpl.label,
   113→        description: '',
   114→        subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
   115→        sourceKey: key,
   116→      };
   117→      setEditableStages(prev => [...prev, newStage]);
   118→      setExpandedStageId(newStage.id);
   119→    }
   120→  };
   121→
   122→  const addAllTemplates = () => {
   123→    if (!wizardData) return;
   124→    const allAdded = wizardData.stages.every(s => isTemplateAdded(s.key));
   125→    if (allAdded) {
   126→      // Remove all template stages
   127→      setEditableStages(prev => prev.filter(s => !s.sourceKey));
   128→    } else {
   129→      // Add all missing templates
   130→      const newStages: EditableStage[] = [];
   131→      for (const tpl of wizardData.stages) {
   132→        if (!isTemplateAdded(tpl.key)) {
   133→          const subtaskTitles = tpl.hasInstrumentBased && selectedInstruments.length > 0
   134→            ? selectedInstruments.map(i => `Запись: ${i}`)
   135→            : STAGE_SUBTASKS[tpl.key] || [];
   136→          newStages.push({
   137→            id: uid(),
   138→            emoji: tpl.emoji,
   139→            label: tpl.label,
   140→            description: '',
   141→            subtasks: subtaskTitles.map(t => ({ id: uid(), title: t, description: '' })),
   142→            sourceKey: tpl.key,
   143→          });
   144→        }
   145→      }
   146→      setEditableStages(prev => [...prev, ...newStages]);
   147→    }
   148→  };
   149→
   150→  // Re-generate instrument-based subtasks when instruments change
   151→  const updateInstrumentSubtasks = useCallback(() => {
   152→    if (!wizardData) return;
   153→    const recordStage = wizardData.stages.find(s => s.key === 'Запись');
   154→    if (!recordStage || !isTemplateAdded('Запись')) return;
   155→    const instSubtasks = selectedInstruments.map(i => `Запись: ${i}`);
   156→    setEditableStages(prev => prev.map(s => {
   157→      if (s.sourceKey !== 'Запись') return s;
   158→      // Keep existing subtasks that still match an instrument, add new ones
   159→      const existing = s.subtasks.filter(st => instSubtasks.includes(st.title));
   160→      const finalSubs: EditableSubtask[] = instSubtasks.map(title => {
   161→        const ex = existing.find(e => e.title === title);
   162→        return ex || { id: uid(), title, description: '' };
   163→      });
   164→      return { ...s, subtasks: finalSubs };
   165→    }));
   166→  }, [wizardData, selectedInstruments]);
   167→
   168→  // When entering step 2, ensure instrument subtasks are synced
   169→  const goToStep2 = () => {
   170→    updateInstrumentSubtasks();
   171→    setStep(2);
   172→  };
   173→
   174→  // --- Instrument helpers ---
   175→  const toggleInstrument = (inst: string) => {
   176→    setSelectedInstruments(prev =>
   177→      prev.includes(inst) ? prev.filter(i => i !== inst) : [...prev, inst]
   178→    );
   179→  };
   180→
   181→  const addCustomInstrument = () => {
   182→    const trimmed = customInstrument.trim();
   183→    if (!trimmed || allInstruments.includes(trimmed)) return;
   184→    setAllInstruments(prev => [...prev, trimmed]);
   185→    setSelectedInstruments(prev => [...prev, trimmed]);
   186→    setCustomInstrument('');
   187→  };
   188→
   189→  // --- Stage editing helpers ---
   190→  const updateStage = (stageId: string, updates: Partial<Pick<EditableStage, 'label' | 'description' | 'emoji'>>) => {
   191→    setEditableStages(prev => prev.map(s => s.id === stageId ? { ...s, ...updates } : s));
   192→  };
   193→
   194→  const removeStage = (stageId: string) => {
   195→    setEditableStages(prev => prev.filter(s => s.id !== stageId));
   196→  };
   197→
   198→  const addNewStage = () => {
   199→    if (!newStageLabel.trim()) return;
   200→    const newStage: EditableStage = {
   201→      id: uid(),
   202→      emoji: DEFAULT_STAGE_EMOJIS[editableStages.length % DEFAULT_STAGE_EMOJIS.length],
   203→      label: newStageLabel.trim(),
   204→      description: newStageDesc.trim(),
   205→      subtasks: [],
   206→    };
   207→    setEditableStages(prev => [...prev, newStage]);
   208→    setNewStageLabel('');
   209→    setNewStageDesc('');
   210→    setShowNewStageForm(false);
   211→    setExpandedStageId(newStage.id);
   212→  };
   213→
   214→  const updateSubtask = (stageId: string, subtaskId: string, updates: Partial<Pick<EditableSubtask, 'title' | 'description'>>) => {
   215→    setEditableStages(prev => prev.map(s => {
   216→      if (s.id !== stageId) return s;
   217→      return { ...s, subtasks: s.subtasks.map(st => st.id === subtaskId ? { ...st, ...updates } : st) };
   218→    }));
   219→  };
   220→
   221→  const removeSubtask = (stageId: string, subtaskId: string) => {
   222→    setEditableStages(prev => prev.map(s => {
   223→      if (s.id !== stageId) return s;
   224→      return { ...s, subtasks: s.subtasks.filter(st => st.id !== subtaskId) };
   225→    }));
   226→  };
   227→
   228→  const addSubtask = (stageId: string) => {
   229→    if (!newSubtaskTitle.trim()) return;
   230→    setEditableStages(prev => prev.map(s => {
   231→      if (s.id !== stageId) return s;
   232→      return { ...s, subtasks: [...s.subtasks, { id: uid(), title: newSubtaskTitle.trim(), description: newSubtaskDesc.trim() }] };
   233→    }));
   234→    setNewSubtaskTitle('');
   235→    setNewSubtaskDesc('');
   236→    setAddingSubtaskForStage(null);
   237→  };
   238→
   239→  // --- Create ---
   240→  const handleCreate = useCallback(async () => {
   241→    if (!trackName.trim() || !selectedBoardId) return;
   242→    setLoading(true);
   243→    try {
   244→      const res = await fetch('/api/tracks/wizard', {
   245→        method: 'POST',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({
   248→          title: trackName.trim(),
   249→          description: trackDescription.trim(),
   250→          deadline: trackDeadline,
   251→          instruments: selectedInstruments,
   252→          boardId: selectedBoardId,
   253→          userId: user?.id,
   254→          customStages: editableStages.map(s => ({
   255→            emoji: s.emoji,
   256→            label: s.label,
   257→            description: s.description,
   258→            subtasks: s.subtasks.map(st => ({ title: st.title, description: st.description })),
   259→          })),
   260→        }),
   261→      });
   262→      const data = await res.json();
   263→      const taskRes = await fetch(`/api/tasks?boardId=${selectedBoardId}&deep=true`);
   264→      const taskData = await taskRes.json();
   265→      setBoardTasks(taskData.tasks);
   266→      if (data.trackId) setSelectedTaskId(data.trackId);
   267→      setIsTrackWizardOpen(false);
   268→    } finally {
   269→      setLoading(false);
   270→    }
   271→  }, [trackName, trackDescription, selectedInstruments, selectedBoardId, editableStages, setBoardTasks, setSelectedTaskId, setIsTrackWizardOpen]);
   272→
   273→  const canProceed = () => {
   274→    if (step === 0) return trackName.trim().length > 0;
   275→    if (step === 1) return selectedInstruments.length > 0;
   276→    if (step === 2) return editableStages.length > 0;
   277→    return true;
   278→  };
   279→
   280→  const handleClose = () => setIsTrackWizardOpen(false);
   281→
   282→  const STEPS = ['Название', 'Инструменты', 'Этапы'];
   283→
   284→  return (
   285→    <div className="flex flex-col flex-1 min-h-0" style={{ background: 'linear-gradient(180deg, rgba(6,6,12,0.95), rgba(10,10,18,0.98))' }}>
   286→      {/* Header */}
   287→      <div className="p-4 shrink-0" style={{ borderBottom: '2px solid rgba(252, 238, 10, 0.15)', background: 'linear-gradient(90deg, rgba(252,238,10,0.03), transparent)' }}>
   288→        <div className="flex items-center justify-between mb-3">
   289→          <div className="flex items-center gap-2">
   290→            <div className="w-8 h-8 flex items-center justify-center" style={{ background: 'linear-gradient(135deg, #FCEE0A, #F1F100)', clipPath: 'polygon(0 0, calc(100% - 4px) 0, 100% 4px, 100% 100%, 4px 100%, 0 calc(100% - 4px))' }}>
   291→              <Music className="w-4 h-4 text-black" />
   292→            </div>
   293→            <div>
   294→              <h3 className="text-[10px] font-bold uppercase tracking-widest" style={{ color: '#FCEE0A', textShadow: '0 0 6px rgba(252,238,10,0.3)' }}>Конструктор трека</h3>
   295→              <p className="text-[9px] text-slate-600 font-mono">// мастер создания пайплайна</p>
   296→            </div>
   297→          </div>
   298→          <button onClick={handleClose} className="p-1.5 rounded transition-all" style={{ color: '#4a4a5e', border: '1px solid transparent' }}
   299→            onMouseEnter={(e) => { e.currentTarget.style.color = '#FCEE0A'; e.currentTarget.style.borderColor = 'rgba(252,238,10,0.3)'; e.currentTarget.style.background = 'rgba(252,238,10,0.06)'; }}
   300→            onMouseLeave={(e) => { e.currentTarget.style.color = '#4a4a5e'; e.currentTarget.style.borderColor = 'transparent'; e.currentTarget.style.background = 'transparent'; }}>
   301→            <X className="w-4 h-4" />
   302→          </button>
   303→        </div>
   304→
   305→        {/* Progress steps */}
   306→        <div className="flex items-center gap-1">
   307→          {STEPS.map((s, i) => (
   308→            <div key={s} className="flex items-center flex-1">
   309→              <button
   310→                onClick={() => i < step && setStep(i)}
   311→                className={cn(
   312→                  'flex items-center gap-1.5 text-[10px] font-bold px-2 py-1 transition-all duration-200',
   313→                  i < step && 'text-slate-400 cursor-pointer',
   314→                  i > step && 'text-slate-600',
   315→                )}
   316→                style={i === step ? { color: '#FCEE0A', textShadow: '0 0 4px rgba(252,238,10,0.3)' } : undefined}
   317→              >
   318→                {i < step ? <Check className="w-3 h-3" /> : <span className="w-4 h-4 rounded-full border text-center text-[9px] flex items-center justify-center" style={{
   319→                  borderColor: i === step ? '#FCEE0A' : i < step ? '#FCEE0A' : '#334155',
   320→                }}>{i + 1}</span>}
   321→                <span className="hidden lg:inline uppercase tracking-wider">{s}</span>
   322→              </button>
   323→              {i < STEPS.length - 1 && (
   324→                <div className="flex-1 h-px mx-1" style={{ backgroundColor: i < step ? 'rgba(252,238,10,0.3)' : '#1e293b' }} />
   325→              )}
   326→            </div>
   327→          ))}
   328→        </div>
   329→      </div>
   330→
   331→      {/* Content */}
   332→      <div className="flex-1 overflow-y-auto p-4">
   333→        {/* Step 0: Name + Description + Deadline */}
   334→        {step === 0 && (
   335→          <div className="space-y-3">
   336→            <div className="space-y-1.5">
   337→              <label className="text-[9px] uppercase tracking-widest font-bold" style={{ color: 'rgba(252,238,10,0.6)' }}>Название трека</label>
   338→              <Input
   339→                value={trackName}
   340→                onChange={(e) => setTrackName(e.target.value)}
   341→                onKeyDown={(e) => e.key === 'Enter' && canProceed() && setStep(1)}
   342→                placeholder="Лирика, Эксперимент..."
   343→                autoFocus
   344→                className="text-sm text-slate-100 placeholder:text-slate-600 h-10 rounded-md focus:outline-none"
   345→                style={{ background: 'rgba(8,8,16,0.9)', border: '1.5px solid rgba(252,238,10,0.2)', boxShadow: 'inset 0 0 6px rgba(252,238,10,0.02)' }}
   346→                onFocus={(e) => { e.currentTarget.style.borderColor = 'rgba(252,238,10,0.4)'; e.currentTarget.style.boxShadow = 'inset 0 0 6px rgba(252,238,10,0.05), 0 0 8px rgba(252,238,10,0.1)'; }}
   347→                onBlur={(e) => { e.currentTarget.style.borderColor = 'rgba(252,238,10,0.15)'; e.currentTarget.style.boxShadow = 'inset 0 0 6px rgba(252,238,10,0.02)'; }}
   348→              />
   349→            </div>
   350→            <div className="space-y-1.5">
   351→              <label className="text-[9px] uppercase tracking-widest font-bold" style={{ color: 'rgba(252,238,10,0.6)' }}>Описание</label>
   352→              <Textarea
   353→                value={trackDescription}
   354→                onChange={(e) => setTrackDescription(e.target.value)}
   355→                placeholder="Краткое описание идеи трека, настроение, жанр..."
   356→                rows={3}
   357→                className="text-sm text-slate-200 placeholder:text-slate-600 resize-none rounded-md focus:outline-none"
   358→                style={{ background: 'rgba(8,8,16,0.9)', border: '1.5px solid rgba(252,238,10,0.2)' }}
   359→                onFocus={(e) => { e.currentTarget.style.borderColor = 'rgba(252,238,10,0.4)'; }}
   360→                onBlur={(e) => { e.currentTarget.style.borderColor = 'rgba(252,238,10,0.15)'; }}
   361→              />
   362→            </div>
   363→            <div className="space-y-1.5">
   364→              <label className="text-[9px] uppercase tracking-widest font-bold" style={{ color: 'rgba(252,238,10,0.6)' }}>Дедлайн</label>
   365→              <DeadlinePicker value={trackDeadline} onChange={setTrackDeadline} size="md" />
   366→            </div>
   367→            <div className="rounded-md p-3" style={{ background: 'rgba(0,240,255,0.03)', border: '1px solid rgba(0,240,255,0.12)', clipPath: 'polygon(0 0, calc(100% - 4px) 0, 100% 4px, 100% 100%, 4px 100%, 0 calc(100% - 4px))' }}>
   368→              <div className="flex items-center gap-2 mb-1">
   369→                <Sparkles className="w-3.5 h-3.5" style={{ color: '#FCEE0A' }} />
   370→                <span className="text-[11px] font-medium" style={{ color: '#FCEE0A' }}>Подсказка</span>
   371→              </div>
   372→              <p className="text-[10px] text-slate-500 leading-relaxed">
   373→                После ввода названия и описания вы выберете инструменты и этапы производства.
   374→                Система автоматически создаст структуру задач для вашего трека.
   375→              </p>
   376→            </div>
   377→          </div>
   378→        )}
   379→
   380→        {/* Step 1: Instruments — blue default, yellow selected */}
   381→        {step === 1 && (
   382→          <div className="space-y-3">
   383→            <div className="flex items-center justify-between">
   384→              <span className="text-[9px] uppercase tracking-widest font-bold" style={{ color: 'rgba(0,229,255,0.7)' }}>
   385→                Инструменты <span style={{ color: '#5a7a9e' }}>({selectedInstruments.length})</span>
   386→              </span>
   387→            </div>
   388→            <div className="flex flex-wrap gap-1.5">
   389→              {allInstruments.map((inst) => {
   390→                const IconComp = CUSTOM_ICONS[inst];
   391→                const isSelected = selectedInstruments.includes(inst);
   392→                return (
   393→                  <button
   394→                    key={inst}
   395→                    onClick={() => toggleInstrument(inst)}
   396→                    className={cn(
   397→                      'flex items-center gap-1.5 text-[11px] font-medium px-2.5 py-1.5 transition-all duration-150',
   398→                    )}
   399→                    style={isSelected ? {
   400→                      color: '#000',
   401→                      background: 'linear-gradient(135deg, #FCEE0A, #F1F100)',
   402→                      border: '1.5px solid rgba(252,238,10,0.8)',
   403→                      boxShadow: '0 0 10px rgba(252,238,10,0.3), inset 0 1px 0 rgba(255,255,255,0.3)',
   404→                      clipPath: 'polygon(0 0, calc(100% - 3px) 0, 100% 3px, 100% 100%, 3px 100%, 0 calc(100% - 3px))',
   405→                    } : {
   406→                      color: '#8ab4d8',
   407→                      background: 'rgba(0,229,255,0.05)',
   408→                      border: '1.5px solid rgba(0,229,255,0.2)',
   409→                      clipPath: 'polygon(0 0, calc(100% - 3px) 0, 100% 3px, 100% 100%, 3px 100%, 0 calc(100% - 3px))',
   410→                    }}
   411→                    onMouseEnter={(e) => { if (!isSelected) { e.currentTarget.style.borderColor = 'rgba(0,229,255,0.5)'; e.currentTarget.style.background = 'rgba(0,229,255,0.1)'; e.currentTarget.style.boxShadow = '0 0 8px rgba(0,229,255,0.1)'; } }}
   412→                    onMouseLeave={(e) => { if (!isSelected) { e.currentTarget.style.borderColor = 'rgba(0,229,255,0.2)'; e.currentTarget.style.background = 'rgba(0,229,255,0.05)'; e.currentTarget.style.boxShadow = 'none'; } }}
   413→                  >
   414→                    {IconComp ? <IconComp className="w-3 h-3" /> : <Music className="w-3 h-3" />}
   415→                    {inst}
   416→                    {isSelected && <Check className="w-3 h-3 ml-0.5" />}
   417→                  </button>
   418→                );
   419→              })}
